\section{Quantitative confinement of the summand supports}
\label{app:confinement}

In this appendix we prove Proposition~\ref{prop:reduction}. We first establish
a quantitative stability refinement of Esseen's lattice moment inequality and
a two-shift smoothing lemma, and then apply them to the variational contact
equations of Lemma~\ref{lem:contact}.

\subsection{Quantitative stability in Esseen's moment inequality}

Throughout this subsection we write $c=\ce$, $p=\pe$, and $\tau=\te$
(compare \citet[Lemma~A.9]{HeCheng2026}). Let $Y$ be a centered,
nondegenerate random variable supported on a coset of $h\Z$ ($h>0$). Its
oriented Esseen moment deficit is
\[
 \delta_Y=\frac{c\E|Y|^3-\E Y^3-3h\E Y^2}{h\E Y^2}.
\]
Decomposing $\Law(Y)$ into a mixture of $\delta_0$ and centered two-point
laws $P_{u,v}$ on $\{-u,v\}$ ($u,v>0$) via the measure
$\lambda(du,dv)=(u+v)P_-(du)P_+(dv)/\E Y_+$ and passing to the
variance-weighted probability measure $\omega(du,dv)=uv\lambda(du,dv)/\E Y^2$,
let $n=(u+v)/h\in\{1,2,\ldots\}$ and $a=u/(u+v)\in(0,1)$. The algebraic
identity $c\{a^2+(1-a)^2\}-(1-2a)=3+2c(a-p)^2$ yields the exact representation
\begin{equation}\label{eq:lattice-deficit}
 \delta_Y=\int\bigl\{3(n-1)+2cn(a-p)^2\bigr\}\,d\omega.
\end{equation}
In particular $\delta_Y\ge0$, which recovers \eqref{eq:esseen-moment}.

When $\delta_Y<3$, some pair of support points has $n=1$; these must be the
unique two lattice points $-ah,(1-a)h$ of the coset straddling $0$, and they
carry $\omega$-weight at least $1-\delta_Y/3$. Since every other two-point
component has variance at least $h^2a(1-a)$, $\Law(Y)$ decomposes as
\begin{gather}
 \Law(Y)=(1-r_Y)\Law\bigl(h(B_a-a)\bigr)+r_YQ,\qquad r_Y\le\frac{\delta_Y}{3},\notag\\
 |a-p|\le\sqrt{\frac{\delta_Y}{2c(1-\delta_Y/3)}},\qquad
 h^2a(1-a)\le\E Y^2\le\frac{h^2a(1-a)}{1-\delta_Y/3},   \label{eq:nearest-law}
\end{gather}
and if $|Y|\le M$, then $W_1(\Law(Y),\Law(h(B_p-p)))\le\frac{2M}{3}\delta_Y+2h\sqrt{\delta_Y/[2c(1-\delta_Y/3)]}$.
Summing over a family of centered $h$-lattice variables $Y_i$ with
$V=\sum_i\E Y_i^2$, $B=\sum_i\E|Y_i|^3$, $K=\sum_i\E Y_i^3$, and aggregate
deficit $\Delta_L=(cB-K-3hV)/(hV)$ gives, by Cauchy--Schwarz,
\begin{equation}\label{eq:lattice-W1}
 \frac1V\sum_i\E Y_i^2 W_1\bigl(\Law(Y_i),\Law(h(B_p-p))\bigr)
 \le\left(\frac{5M}{3}+h\right)\Delta_L+h\sqrt{\frac3c}\sqrt{\Delta_L},
\end{equation}
and if $\Delta_L\le D$, then with $a_D=\sqrt{(1+D/3)D/(2c)}$,
$b_D=\tau D/3+D/c+(2/c)a_D$, and $k_D=D/(3c)+2a_D$,
\begin{equation}\label{eq:lattice-ratios}
 \left|\frac B{hV}-\tau\right|\le b_D,\qquad
 \left|\frac K{hV}-\frac1c\right|\le k_D,\qquad \frac B{hV}\ge\frac12.
\end{equation}

\subsection{Rounding to a lattice coset at the selected period}

For the selected maximizer of Lemma~\ref{lem:selection}, set $Y_i=X_i/\ell$,
so $|Y_i|\le8$, $\E Y_i^2\le9$, and $\ell^2\sum_i\E Y_i^2=\ell^2\sum_i\E|Y_i|^3=1$.
At a frequency $u\ne0$ with $h=2\pi/|u|$, round each $Y_i$ to the nearest
point of the lattice coset aligned with $\arg\E e^{\ii uY_i}$ and center the
rounded variable to obtain $Q_i$. Using $|v|\le(\pi/2)|e^{\ii v}-1|$ on
$[-\pi,\pi]$, we have
\begin{equation}\label{eq:rounding}
 e:=\ell^2\sum_{i=1}^n\E|Y_i-Q_i|^2\le \frac{h^2g(u)}{8},\qquad |Q_i|\le8+h=:M,
\end{equation}
and Cauchy--Schwarz bounds the moment differences by
\begin{align}
 \ell^2\sum_{i=1}^n|\E Q_i^2-\E Y_i^2|&\le2\sqrt e+e,\notag\\
 \ell^2\sum_{i=1}^n\bigl|\E|Q_i|^3-\E|Y_i|^3\bigr|,\
 \ell^2\sum_{i=1}^n|\E Q_i^3-\E Y_i^3|
                &\le\frac{3M}{2}\sqrt e(2+\sqrt e).           \label{eq:rounding-moments}
\end{align}

\subsection{Support spacing from two small smoothing losses}

\begin{lemma}\label{lem:support-spacing}
Let $F$ be the distribution function of a finite positive measure $\nu$, and
let $J=F*\operatorname{Unif}[-h/2,h/2]$ ($h>0$). Suppose $J(x)$ differs from
an affine function of slope $q>0$ by at most $\varepsilon$ on
$[-D_0-2h,3h]$. Put $H(a)=J(a+h/2)-F(a)=\int_{(a,a+h)}(1-(t-a)/h)\,\nu(dt)\ge0$.
If $0\le d\le D_0$ and $H(0),H(-d)\le s$, then for any $0<r_0\le h/4$ and
$K_0=\lceil D_0/h\rceil$,
\begin{equation}\label{eq:spacing-bound}
 \dist(d,h\Z)\le3r_0+
 \frac{4s+(5+4K_0)\varepsilon+4K_0h\varepsilon/r_0}{q}.
\end{equation}
\end{lemma}
\begin{proof}
For $0<d<h$, the trapezoid kernel $T_d(t)=h^{-1}\int_{h/2-d}^{h/2-d/2}\ind_{\{|x-t|<h/2\}}\,dx$
satisfies $\min\{1,2(h-d)/d\}T_d(t)\le w_0(t)+w_{-d}(t)$ pointwise, where
$w_a(t)=(1-(t-a)/h)\ind_{(a,a+h)}(t)$. Integrating against $\nu$ and using
$J(h/2-d/2)-J(h/2-d)\ge qd/2-2\varepsilon$ gives
\begin{equation}\label{eq:two-shifts}
       q\min(d/2,h-d)\le H(0)+H(-d)+2\varepsilon.
\end{equation}
For general $d=kh+d_0$ ($0\le k\le K_0$, $0\le d_0<h$), averaging $H$ over an
interval of length $r_0$ and using $|\overline H_{r_0}(a+h)-\overline H_{r_0}(a)|\le2\varepsilon(1+h/r_0)$
transfers the bound $H(-d)\le s$ across $k$ periods to $-d_0$, where
\eqref{eq:two-shifts} yields \eqref{eq:spacing-bound}.
\end{proof}

\subsection{Pinning the lattice span $h$ and threshold $z$}

For the selected maximizer, put $v_i=\E Y_i^2$, $M_i=\E[Y_i|Y_i|]$, and
$\theta=\ell^{-1}\sum_i\E X_i^3\in[-1,1]$. Let $\rho,r,T,\eps,h$ be as in
Proposition~\ref{prop:jitter}, and define the explicit error quantities
\begin{equation}\label{eq:confinement-errors}
\begin{gathered}
\begin{aligned}
 u&=\rho+\ell/2,& D&=16u+256r,\\
 H_h&=72r+5b_D,& Z^2&=10(u+18r),\\
 S_0&=2\rho+18r+460\ell^2,&
 A_0&=2\rho+24\{4\ell^2+(Z+32\ell)^2/2+\ell/5\},\\
 \gamma&=300\sqrt{A_0}+11S_0+131A_0,&
 W_0&=21D+3\sqrt D+42r,
\end{aligned}\\
\begin{aligned}
 E_P&=14000(u+18r+\eta)+1100Z^2+4000\ell+4000\ell^2+1100\eta,\\
 E_{\rm end}&=E_P+6100\gamma+700\gamma^2+500H_h,
\end{aligned}
\end{gathered}
\end{equation}
which under \eqref{eq:reduction-parameters} satisfy
\begin{equation}\label{eq:confinement-gates}
 \begin{gathered}
 r\le10^{-6},\qquad D\le10^{-3},\qquad A_0\le\frac9{64},\qquad
 \gamma\le\frac1{100},\\
 E_P\le10^{-4},\qquad E_{\rm end}\le\frac1{2000},\qquad W_0\le\frac14.
 \end{gathered}
\end{equation}

Since $h=0$ would give $R\le\phz/6+\rho<\be$ by \eqref{eq:jitter}, we have
$h>0$. Applying \eqref{eq:rounding}--\eqref{eq:rounding-moments} first with
$h\le128\pi$ gives $h<5/2$, and reapplying them with $h<5/2$ gives
\begin{equation}\label{eq:rounded-data}
 \sqrt e\le r,\qquad|V_Q-1|\le3r,\qquad
 |B_Q-1|\le32r,\qquad|K_Q-\theta|\le32r,
\end{equation}
so Esseen's inequality \eqref{eq:esseen-moment} for $Q_i$ implies
\begin{equation}\label{eq:span-envelope}
                3h+|\theta|\le c+256r.
\end{equation}
Conversely, monotonicity $F_S(z)\le F_{S+\ell U_h}(z+h\ell/2)$ and
\eqref{eq:jitter} give
\begin{equation}\label{eq:global-envelope}
 R\le\phi(z)\left\{\frac h2+\frac\theta6(1-z^2)\right\}+u,
\end{equation}
which together with $R>\be$ and \eqref{eq:span-envelope} implies
\begin{equation}\label{eq:near-max-envelope}
 3h+|\theta|>c-16u,\qquad R\le\be+u+18r.
\end{equation}
Thus the aggregate Esseen deficit of $(Q_i)$ is at most $D$, so
\eqref{eq:lattice-ratios} yields
\begin{equation}\label{eq:near-esseen}
 |h-\he|\le H_h<\frac1{40},\qquad
 \bigl||\theta|-1/\sqrt{10}\bigr|\le34r+3k_D+\frac{H_h}{6}<\frac1{10},
\end{equation}
and hence
\begin{equation}\label{eq:span-range}
                  \frac{19}{10}<h<2,\qquad\theta>\frac15,
\end{equation}
while $\be<(\be+18r)e^{-z^2/2}+u$ gives
\begin{equation}\label{eq:threshold-bound}
           |z|\le Z,\qquad0\le K-\be\le u+18r+\eta.
\end{equation}

\subsection{Deleted jitter and pairwise support spacing}

Deleting summand $i$ reduces $g(u)$ by at most $\ell^2$, so the deleted
sum $S-X_i$ convolved with the same jitter $\ell U_h$ has distribution
function $J_i$ satisfying, with $\sigma_i^2=1-s_i\ge3/4$,
$\kappa_-=\sum_{j\ne i}\E X_j^3$, and
$G_i^*(x)=\Phi(x/\sigma_i)+\frac{\kappa_-}{6\sigma_i^3}(1-(x/\sigma_i)^2)\phi(x/\sigma_i)$,
\begin{equation}\label{eq:deleted-jitter}
 \|J_i-G_i^*\|_\infty
 \le6000\ell^2+\frac{24r\ell}{\delta^2}
       +2\log(T/\delta)e^{-\eps^2/(2\ell^2)}+\frac{20\ell}{T}
 \le2\rho\ell.
\end{equation}
At any support point $y=\ell t\in\supp X_i$, the contact equality $D_i(y)=0$
and \eqref{eq:gaussian-global} give
\begin{equation}\label{eq:contact-flat}
 |G_i(z-y)-\Phi((z-y)/\sigma_i)-R\ell|\le450\ell^3,
\end{equation}
which combined with \eqref{eq:span-envelope} and \eqref{eq:deleted-jitter}
bounds the one-sided smoothing loss by
\begin{equation}\label{eq:smoothing-loss}
 0\le\frac{J_i(z-y+h\ell/2)-G_i(z-y)}{\ell}\le S_0.
\end{equation}
Applying Lemma~\ref{lem:support-spacing} to $a\mapsto\ell^{-1}J_i(z-x+\ell a)$
around any $x\in\supp X_i$ shows that every pair $x,y\in\supp X_i$ satisfies
\begin{equation}\label{eq:arithmetic-spacing}
                   \dist\bigl((y-x)/\ell,h\Z\bigr)\le\gamma.
\end{equation}

\subsection{The cubic contact polynomial and two-cluster decomposition}

Dividing \eqref{eq:deletion} by $\phz\ell^3/6$ and using \eqref{eq:gaussian-local}
and \eqref{eq:threshold-bound} shows that every $t\in\supp Y_i$ satisfies
\begin{equation}\label{eq:support-polynomial}
 P_i(t):=t^3-c|t|^3+\frac{3c}{2}t^2+3(cM_i-v_i)t\ge-E_P.
\end{equation}
For the extreme points $-a,b$ of $\supp Y_i$, combining $P_i(-a),P_i(b)\ge-E_P$
with $v_i\le ab$ and $|M_i|\le v_i$ gives $a+b<7/2<2h$.

Consequently, by \eqref{eq:arithmetic-spacing}, if $a+b<h/2$, then
$a+b\le\gamma$ and $|Y_i|\le\gamma$; we relabel these summands as
$Z_1,\ldots,Z_s$. If $a+b\ge h/2$, then $|(a+b)-h|\le\gamma$ and every point
of $\supp Y_i$ lies within $2\gamma$ of $-a$ or $b$. Conditioning on the
indicator $B_i$ of the upper cluster $[(-a+b)/2,b]$ yields
\begin{equation}\label{eq:cluster-representation}
 Y_i=h_i(B_i-p_i)+U_i,\qquad\E(U_i\mid B_i)=0,\qquad
 |h_i-h|\le5\gamma,\qquad |U_i|\le2\gamma,
\end{equation}
and substituting the cluster endpoints into \eqref{eq:support-polynomial}
forces $p_i\in[1/4,3/4]$. Finally, \eqref{eq:lattice-W1} and the rounding
bound \eqref{eq:rounded-data} give
\begin{equation}\label{eq:original-W1}
 \sum_{i=1}^n\ell^2v_i W_1\bigl(\Law(Y_i),\Law(h(B_{\pe}-\pe))\bigr)\le W_0,
\end{equation}
so the small-diameter summands $Z_j$ carry total original variance at most
$W_0/(2h\pe\qe-\gamma)\le2W_0$, while the two-cluster summands have count $m$
satisfying
\begin{equation}\label{eq:core-count}
 \frac{1-2W_0}{2\ell^2}\le m\le\frac2{\ell^2},
                         \qquad m\ge\frac1{4\ell^2}\ge N,
\end{equation}
and $\sum_{j=1}^s\Var Z_j/(mh^2)<3W_0\le\lambda$. Checking the numerical
gates \eqref{eq:confinement-gates} under \eqref{eq:reduction-parameters}
completes the proof of Proposition~\ref{prop:reduction}.
