\section{Quantitative Fourier and smoothing estimates}\label{app:fourier}

Write $\widehat\nu(t)=\int_\R e^{\ii tx}\,\nu(dx)$ and $\sinc u=(\sin u)/u$
(with $\sinc 0=1$). This appendix establishes the explicit signed-smoothing,
low-frequency Edgeworth, Poisson-binomial count, and conditional Fourier
bounds used in Sections~\ref{sec:jitter} and~\ref{sec:local}.

\subsection{Signed smoothing and low-frequency Edgeworth expansion}

Let $F$ be a probability distribution function with finite first absolute
moment, and let $G$ be the distribution primitive of a finite signed measure
of total mass one, finite first absolute moment, and Lipschitz constant $M_G$.
By the signed smoothing inequality of \citet[Lemma~2.1 and equation~(71)]{HeCheng2026},
for every $A>0$,
\begin{equation}\label{eq:signed-smoothing}
 \|F-G\|_\infty\le\frac14\int_{-A}^A
       \frac{|\widehat F(t)-\widehat G(t)|}{|t|}\,dt+\frac{24M_G}{A}.
\end{equation}
(Indeed, convolving with the shifted Fej\'er-type kernel $k(x)=3\sinc(x/4)^4/(8\pi)$,
whose Fourier transform is supported on $[-1,1]$ and which satisfies
$\varepsilon_k=\int_{16}^\infty k(x)\,dx<1/8$ and $\int_{-\infty}^{16}(16-x)k(x)\,dx<18$,
gives coefficients $1/[2\pi(1-2\varepsilon_k)]\le1/4$ and $18/(1-2\varepsilon_k)\le24$;
in particular, the weaker coefficients $1/\pi$ and $40$ also hold.)

In the setting of Proposition~\ref{prop:jitter}, with $v_i=\E X_i^2$,
$\kappa=\sum_{i=1}^n\E X_i^3$, and $f_i(t)=\E e^{\ii tX_i}$, we have on
$|t|\le1/(64\ell)$ that $|f_i(t)-1|\le v_it^2/2\le1/128$,
$\sum_i\E X_i^4\le8\ell^2$, and $\sum_i v_i^2\le64\ell^2$. Expanding
$\sum_i\log f_i(t)=-t^2/2+\kappa(\ii t)^3/6+w(t)$ with $|w(t)|\le(49/3)\ell^2t^4\le t^2/6$
bounds the difference against $\widehat\nu(t)=e^{-t^2/2}(1+\kappa(\ii t)^3/6)$ by
\begin{equation}\label{eq:log-error}
 |\widehat F(t)-\widehat\nu(t)|\le\ell^2e^{-t^2/3}\left\{\frac{49}{3}t^4+\frac{t^6}{72}\right\}.
\end{equation}
Including the jitter factor $\sinc(h\ell t/2)$, integrating over
$|t|\le1/(64\ell)$ for $h\le128\pi$, adding the Gaussian tail beyond
$a=1/(64\ell)$, and applying Lemma~\ref{lem:period} with the smoothing
constants $(1/\pi,40)$ and $M_G\le1/2$ yields \eqref{eq:jitter-error}.

For the local comparison in Section~\ref{sec:local}, telescoping the product
directly against $e^{A_i}$ with $A_i=-v_it^2/2+\E X_i^3(\ii t)^3/6$ gives
the tighter low-frequency bound
\begin{equation}\label{eq:small-low-frequency}
 \int_{|t|\le1/(64\ell)}
 \frac{|\sinc(h\ell t/2)\widehat F(t)-\widehat\nu(t)|}{|t|}\,dt
                         \le\left(73+\frac{h^2}{8}\right)\ell^2.
\end{equation}

\subsection{Local Gaussian approximation for Poisson-binomial counts}

\begin{lemma}\label{lem:count-estimate}
Let $K=\sum_{i=1}^m B_i$ be a sum of independent Bernoulli variables with
mean $\mu=\E K$ and variance $V=\Var K\ge1/2$. Then for every $k\in\Z$,
\begin{equation}\label{eq:count-estimate}
 \left|\Pp(K=k)-\frac{\phi((k-\mu)/\sqrt V)}{\sqrt V}\right|
                   \le\frac6V,\qquad
 \sup_{k\in\Z}\Pp(K=k)<\frac1{\sqrt V}.
\end{equation}
In particular, if $p_i\in[\alpha,1-\alpha]$ for all $i$, $\beta=\alpha(1-\alpha)$,
$\gamma=\phi(M/\sqrt\beta)$, and $m\ge144/(\beta\gamma^2)$, then
$\Pp(K=k)\ge\gamma/\sqrt m$ whenever $|k-\mu|\le M\sqrt m$.
\end{lemma}
\begin{proof}
For $Z_i=B_i-p_i$ and $v_i=p_i(1-p_i)$, on $t\in[-\pi,\pi]$ we have
$|\E e^{\ii tZ_i}|\le e^{-2v_i\sin^2(t/2)}\le e^{-2v_it^2/\pi^2}$ and
$|\E e^{\ii tZ_i}-e^{-v_it^2/2}|\le v_i(|t|^3/6+t^4/32)$. Telescoping the
product with deleted decay $e^{-2(V-v_i)t^2/\pi^2}\le e^{-Vt^2/\pi^2}$
(since $v_i\le1/4\le V/2$) and
integrating over $[-\pi,\pi]$ gives an error at most
$(\pi^3/12+\pi^4/64+1/\pi^2)/V<6/V$.
\end{proof}

\subsection{Conditioning an independent product on a Bernoulli count}

\begin{lemma}\label{lem:conditional-fourier}
Let $(B_i,R_i)$ ($1\le i\le m$) be independent pairs with $B_i\sim\Bern(r_i)$,
$r_i\in[a,1-a]$, $\E R_i=0$, and $|R_i|\le d_0$, and let $Z_j$ ($1\le j\le s$)
be additional independent centered variables with $|Z_j|\le d_0$. Put
\[
 K=\sum_{i=1}^m B_i,\qquad R=\sum_{i=1}^mR_i+\sum_{j=1}^sZ_j,\qquad
 \E K=k\in\Z,\qquad V=\Var K\ge1000,\qquad L=\Var R>0,
\]
and assume $\Cov(K,R)=0$. Set $\beta=a(1-a)$. If $g:\R\to\R$ is continuous
and integrable with integrable Fourier transform supported on $[-T,T]$, and
$d_0T\le\beta/162$, then for $Z\sim N(0,1)$,
\begin{align}
 &\sup_{y\in\R}\bigl|\E[g(R-y)\mid K=k]-\E g(\sqrt L Z-y)\bigr|\notag\\
 &\qquad\le\frac{\|g\|_1}{\sqrt L}
 \left\{\frac{32}{\sqrt V}+\frac{8d_0}{\sqrt L}
                 +2T\sqrt{VL}e^{-\beta m/162}\right\}.    \label{eq:conditional-fourier-error}
\end{align}
\end{lemma}
\begin{proof}
Write $\chi(s,t)=\E e^{\ii s(K-k)+\ii tR}$. On $|s|\le1/4$ and $|t|\le T$,
each centered phase $X=s(B_i-r_i)+tR_i$ or $tZ_j$ satisfies $|X|\le1/2$,
$|\E e^{\ii X}|\le e^{-\E X^2/3}$, and $|\E e^{\ii X}-e^{-\E X^2/2}|<\E|X|^3/4$.
Since $\Cov(K,R)=0$, the total phase variance is $Vs^2+Lt^2$ and the sum of
third absolute moments is at most $4(V|s|^3+d_0L|t|^3)$, so telescoping
gives
\[
 |\chi(s,t)-e^{-Vs^2/2-Lt^2/2}|
 \le2(V|s|^3+d_0L|t|^3)e^{-(Vs^2+Lt^2)/3},
\]
whose double integral over $|s|\le1/4$, $|t|\le T$ is at most
$56(VL)^{-1/2}(V^{-1/2}+d_0L^{-1/2})$. On $1/4\le|s|\le\pi$, each Bernoulli
factor has modulus at most $1-\beta/81$ and the residual perturbation changes
it by at most $Td_0\le\beta/162$, giving $|\chi(s,t)|\le e^{-\beta m/162}$.
Dividing the joint Fourier inversion formula by $\Pp(K=k)\ge\frac12(2\pi V)^{-1/2}$
(from Lemma~\ref{lem:count-estimate} since $V\ge1000$) yields
\eqref{eq:conditional-fourier-error}.
\end{proof}

\begin{lemma}[Bandlimited interval minorant]\label{lem:minorant}
For an interval $(a,b)$ with midpoint $w_0=(a+b)/2$ and half-width
$d=(b-a)/2$, fix $L_0>0$, $C_0\ge0$, and choose
\[
 A_g\ge C_0+\frac{|w_0|+d/2}{\sqrt{L_0}},\qquad
 \omega=\frac2d e^{A_g^2/6},\qquad T=4\omega.
\]
Then the function
\[
 g(w)=\left(1-\frac{(w-w_0)^2}{d^2}\right)
                      \sinc\bigl(\omega(w-w_0)\bigr)^4
\]
satisfies $g\le\ind_{(a,b)}$, has integrable Fourier transform supported on
$[-T,T]$, and obeys
\[
 \|g\|_1\le\frac4\omega,\qquad
 \E g(\sqrt L Z-y)\ge\frac{\phi(A_g)}{4\omega\sqrt L}
            \quad(L\ge L_0,\ |y|\le C_0\sqrt L).
\]
Consequently, if the bracket in \eqref{eq:conditional-fourier-error} is at
most $\phi(A_g)/32$, then $\Pp(R-y\in(a,b)\mid K=k)\ge\phi(A_g)/(8\omega\sqrt L)$.
\end{lemma}
\begin{proof}
On $|w-w_0|\le1/\omega\le d/2$, we have $g(w)\ge(3/4)(5/6)^4>1/4$ and
Gaussian density at least $\phi(A_g)/\sqrt L$, while the negative part of $g$
outside $(a,b)$ has integral at most $2/(d^3\omega^4)=\phi(A_g)/(4\omega\phz)$
and Gaussian density at most $\phz/\sqrt L$; subtracting gives the lower bound
on $\E g(\sqrt L Z-y)$.
\end{proof}

\subsection{Jitter on the rescaled count lattice}

To establish \eqref{eq:local-jitter} in Section~\ref{sec:local} (compare
\citet[Lemma~A.4]{HeCheng2026}), set $g(t)=\sum_i(1-|f_i(t)|^2)$ (over all
$m+s$ summands) and $g_0(t)=2V(1-\cos t)$. Because $\sum_{i=1}^m v_ia_i=0$
and $\E(U_i\mid B_i)=0$, the first-order terms in the Taylor expansion of
$1-\cos(td)$ around the unperturbed differences $D_0=B_i-B_i'$ vanish, giving
for $\max_i|a_i|,\eps_0\le1/4$:
\begin{align}
 |g(t)-g_0(t)|&\le\Lambda t^2,\notag\\
 |g'(t)-g_0'(t)|&\le\Lambda(2|t|+2t^2),\notag\\
 |g''(t)-g_0''(t)|&\le\Lambda(2+8|t|+4t^2).                    \label{eq:resonance-derivatives}
\end{align}
When $T\ge10$ and $\xi(T+1)^2\le1/56$, each resonance interval
$I_j=[2\pi j-1/4,2\pi j+1/4]$ intersecting $[-T,T]$ contains a unique minimum
$t_j$ of $g$ with $|t_j-2\pi j|\le4\Lambda(T+1)^2/V$ and
$\prod_i|f_i(t)|\le e^{-V(t-t_j)^2/4}$ on $I_j$, while outside $\bigcup_j I_j$
we have $\prod_i|f_i(t)|\le e^{-V/50}$. Since $\sinc(t/2)$ vanishes at
$2\pi j$ and is $(1/4)$-Lipschitz, summing over $j\ne0$ gives
\begin{equation}\label{eq:resonance-integral}
 \sum_{j\ne0}\int_{I_j\cap[-T,T]}
       \frac{|\sinc(t/2)\prod_i f_i(t)|}{|t|}\,dt
 \le(1+\log T)\left\{\frac2{5V}
        +\frac{4\sqrt\pi}{5}\frac{\Lambda(T+1)^2}{V\sqrt V}\right\}.
\end{equation}
Combining \eqref{eq:small-low-frequency} on $|u|\le1/128$,
\eqref{eq:resonance-integral} on the nonzero resonance intervals, the
exponential decay $e^{-V/50}$ outside $\bigcup_j I_j$, and the smoothing
bound \eqref{eq:signed-smoothing} at cutoff $BT$ yields \eqref{eq:local-jitter}
with error \eqref{eq:local-error}.
