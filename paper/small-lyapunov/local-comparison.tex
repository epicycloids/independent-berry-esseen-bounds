\section{Local comparison near a common-diameter Bernoulli array}\label{sec:local}

\begin{theorem}[Explicit local comparison for two-cluster arrays]\label{thm:local}
Set
\begin{equation}\label{eq:local-constants}
 \mathcal A=38e^{43},\qquad\lambda=e^{-\mathcal A},
                   \qquad N=\lceil e^{\mathcal A}\rceil.
\end{equation}
Every independent array of the form \eqref{eq:clusters}--\eqref{eq:extra-bounds}
with $m\ge N$ satisfies $\mathcal R\le\be$, where $\mathcal R=\Delta/\ell$ is
computed from the total variance and the sum of third absolute moments of all
$m+s$ summands.
\end{theorem}

Following the two-regime strategy of \citet[Section~3]{HeCheng2026}, we prove
Theorem~\ref{thm:local} by comparing the array against the common-diameter
Bernoulli reference array with the same label probabilities $p_1,\ldots,p_m$,
handling small and large total residual variance $\Lambda$ by complementary
arguments that are uniform in unequal probabilities $p_i$, unequal cluster
diameters $h_i$, and the additional small-diameter summands $Z_j$.

\subsection{Normalization and conditional moment estimates}

Set $v_i=p_i(1-p_i)$ and define the variance-weighted mean diameter
\[
 \bar h=\frac{\sum_{i=1}^m v_i h_i}{\sum_{i=1}^m v_i}\in[(1-\lambda)h,(1+\lambda)h].
\]
Dividing every summand by $\bar h$ (and retaining the notation $h_i,U_i,Z_j$
for the rescaled quantities), we write
\begin{gather*}
 V=\sum_{i=1}^m v_i,\qquad\mu=\sum_{i=1}^m p_i,
 \qquad h_i=1+a_i,\qquad\sum_{i=1}^m v_i a_i=0,\\
 K_0=\sum_{i=1}^mB_i,\qquad A=\sum_{i=1}^m a_i(B_i-p_i),\qquad
 U=\sum_{i=1}^mU_i+\sum_{j=1}^s Z_j,\qquad W=A+U,\qquad S=K_0+W,\\
 \Lambda_A=\sum_{i=1}^m v_i a_i^2,\qquad
 \Lambda_U=\sum_{i=1}^m\E U_i^2+\sum_{j=1}^s\E Z_j^2,\qquad
 \Lambda=\Lambda_A+\Lambda_U,\qquad B^2=V+\Lambda.
\end{gather*}
Here the uncentered sum $S$ has mean $\mu$, $\Cov(K_0,W)=0$, and with
$\eps_0=\max(\max_i\|U_i\|_\infty,\max_j\|Z_j\|_\infty)$ and $\xi=\Lambda/V$,
the hypotheses \eqref{eq:cluster-bounds}--\eqref{eq:extra-bounds} imply
\begin{equation}\label{eq:local-domains}
 \frac{3m}{16}\le V\le \frac m4,\qquad\max_{1\le i\le m}|a_i|\le3\lambda,\qquad
 \eps_0\le2\lambda,\qquad\xi\le8\lambda,\qquad\Lambda\le2\lambda m.
\end{equation}

Let $T_X$ and $\kappa$ denote the sums of the absolute and signed third
central moments of all $m+s$ summands, and let
\[
 T_0=\sum_{i=1}^m v_i\bigl(p_i^2+(1-p_i)^2\bigr),\qquad
 \kappa_0=\sum_{i=1}^m v_i(1-2p_i)
\]
be the corresponding moments of the unit-diameter reference array
$(B_1-p_1,\ldots,B_m-p_m)$. Because $\E(U_i\mid B_i)=0$ and each cluster preserves
its sign ($p_i,1-p_i\ge1/4>7/32>\eps_0$), expanding $|(1+a_i)(B_i-p_i)+U_i|^3$
gives
\begin{align}
 |T_X-T_0|,\ |\kappa-\kappa_0|
      &\le2\sqrt{m\Lambda_A}+4\Lambda,                  \label{eq:moment-budget}\\
 T_X\ge \frac V3,\qquad\frac14\le \frac{T_X}{B^2}&\le2,\qquad
 B^2\le m,\qquad \frac B2\le P:=\frac{B^3}{T_X}\le4B.   \label{eq:prefactor}
\end{align}

For $m\ge10^{12}$ and $|k-\mu|\le6\sqrt m$, Appendix~\ref{app:conditional}
establishes the count-conditioned gap-perturbation bounds
\begin{align}
 |m_k|:=|\E(A\mid K_0=k)|&\le10^4\frac{\sqrt{\Lambda_A}}{\sqrt m},
                                                        \label{eq:conditional-mean}\\
 |\Var(A\mid K_0=k)-\Lambda_A|&\le40000\frac{\Lambda_A}{\sqrt m},\notag\\
 \E[(A-m_k)^4\mid K_0=k]&\le10^7\Lambda_A^2,             \label{eq:conditional-gap}
\end{align}
together with the relative conditional label-probability bound
\begin{equation}\label{eq:conditional-label}
 \left|\frac{\Pp(B_i=b\mid K_0=k)}{\Pp(B_i=b)}-1\right|
                         \le\frac{500}{\sqrt m}\qquad(b\in\{0,1\}).
\end{equation}
For $\Lambda>0$, define
\begin{equation}\label{eq:DH}
 D=\Lambda^2+\eps_0^2\Lambda_U,\qquad
 H=\max\left\{\sqrt{\Lambda_A},\,
                   \frac\Lambda{\sqrt{\Lambda+\eps_0^2}}\right\}.
\end{equation}
Conditioning first on the full vector of labels $(B_1,\ldots,B_m)$ and
combining \eqref{eq:conditional-mean}--\eqref{eq:conditional-label} with the
moment interpolation $\E|Y|\ge(\E Y^2)^{3/2}/(\E Y^4)^{1/2}$ yields, for
$|k-\mu|\le6\sqrt m$,
\begin{gather}
 \E(W\mid K_0=k)=m_k,\qquad
 \frac\Lambda2\le\Var(W\mid K_0=k)\le2\Lambda,\notag\\
 \E[(W-m_k)^4\mid K_0=k]\le10^8D,\qquad
 \E(|W-m_k|\mid K_0=k)\ge \frac{H}{30000},                       \label{eq:full-conditional}\\
 \E(W^4\mid K_0=k)\le10^9D,\qquad\E W^4\le400D.\notag
\end{gather}

\subsection{Jittered Edgeworth approximation and central threshold reduction}

Let $J$ denote the distribution function of $S$ plus an independent
$\operatorname{Unif}[-1/2,1/2]$ variable. For $T\ge10$ with $\xi(T+1)^2\le1/56$,
set
\begin{align}
 E_m(T)={}&\frac{401(1+\log T)}{\sqrt m}
       +(1+\log T)\xi(T+1)^2\notag\\
       &+60000(1+\log T)\sqrt m\,e^{-m/262144}+\frac{56}{T}.       \label{eq:local-error}
\end{align}
In Appendix~\ref{app:fourier} we prove that for $z=(x-\mu)/B$,
\begin{equation}\label{eq:local-jitter}
 \sqrt m\left\|J(x)-\Phi(z)
             -\frac\kappa{6B^3}(1-z^2)\phi(z)\right\|_\infty\le E_m(T).
\end{equation}
Choosing the explicit parameters
\[
 a_0=e^{43},\qquad\mathcal C=\frac{49}{4}a_0,\qquad
 \mathcal B=\frac{25}{2}a_0,
 \qquad q=e^{-\mathcal C},\qquad T=e^{\mathcal B},
\]
direct exponent comparison shows that for every $m\ge N=\lceil e^{\mathcal A}\rceil$,
\begin{equation}\label{eq:local-numerical}
 \xi(T+1)^2<\frac1{56},\qquad E_m(T)<\frac q{1000},\qquad
 m^{-1/2}<\frac q{1000},\qquad200\lambda<\frac q{1000}.
\end{equation}
Shifting the uniform jitter by $\pm1/2$ bounds either normalized one-sided
discrepancy at $z=(x-\mu)/B$ by
\begin{equation}\label{eq:local-envelope}
 \phi(z)\left\{\frac{B^2}{2T_X}
      +\frac{|\kappa|}{6T_X}|1-z^2|\right\}+4E_m(T)+m^{-1/2}.
\end{equation}
For $|z|\ge4$, the main term in \eqref{eq:local-envelope} is at most
$(3+z^2)\phi(z)<1/100$, so the normalized discrepancy for $|z|\ge4$ is less
than $1/24<\be$. It therefore suffices to consider central thresholds
$|z|\le4$.

\subsection{Regime I: small residual variance ($0<\Lambda\le e^{-43}$)}\label{subsec:small-residual}

When $\Lambda=0$, all two-cluster summands have diameter $1$ and $U_i=0,Z_j=0$,
so \eqref{eq:finite-input} gives $\mathcal R\le\be$. Suppose now that
$0<\Lambda\le e^{-43}$. In Appendix~\ref{app:conditional} we prove the
following one-sided clipping inequality (compare \citet[Lemma~3.2]{HeCheng2026}):
if $\E Y=0$, $\E Y^4\le1/160000$, $f(0)=0$, and $3/4\le f'\le5/4$ on
$[-7/10,7/10]$, then for $|u|\le3/5$,
\begin{align}
 \Pp(Y>u)+f(u)&\ge\frac38\E|Y|-176000\E Y^4,\notag\\
 \Pp(Y<u)-f(u)&\ge\frac38\E|Y|-176000\E Y^4.             \label{eq:one-sided}
\end{align}

By Lemma~\ref{lem:count-estimate}, the count probabilities satisfy
\begin{equation}\label{eq:central-count}
 b_k:=\Pp(K_0=k)\ge \frac{e^{-100}}{\sqrt m}
                         \qquad(|k-\mu|\le6\sqrt m).
\end{equation}
Write any threshold with $|z|\le4$ as $x=k+u$, where $k$ is a nearest
integer to $x$ and $|u|\le1/2$, so that $|k-\mu|\le5\sqrt m$. Setting
$f_k(u)=[\Phi((k-\mu+u)/B)-\Phi((k-\mu)/B)]/b_k$, we have $3/4\le f_k'\le5/4$
on $[-1,1]$. Decomposing the event $\{S\le k+u\}$ by the value of $K_0$ gives
\begin{align}
 F_S(k+u)&=F_{K_0}(k)-b_k\Pp(W>u\mid K_0=k)+e^+_{k,u},\notag\\
 F_S^-(k+u)&=F_{K_0}(k-1)+b_k\Pp(W<u\mid K_0=k)+e^-_{k,u},
                                                        \label{eq:layers}
\end{align}
where Markov's inequality and \eqref{eq:full-conditional} bound the off-layer
tail sums by
\[
 |e^\pm_{k,u}|\le\sum_{j\ne k}b_j\Pp\bigl\{|W|\ge|j-k|-1/2\bigm| K_0=j\bigr\}
                         \le2\cdot10^{11}\frac{D}{\sqrt m}.
\]
Applying \eqref{eq:one-sided} conditionally on $K_0=k$ to $Y=W-m_k$ (with
$10^8D\le2\cdot10^8e^{-86}<1/160000$) and using $Pb_k\ge\frac25\phi(4.01)>e^{-11}$
and $(3/8)(H/30000)e^{-11}>e^{-23}H$, while comparing $P=B^3/T_X$ to
$V^{3/2}/T_0$ via \eqref{eq:moment-budget}, bounds either normalized
discrepancy branch at $x=k+u$ by
\begin{equation}\label{eq:small-comparison}
 R_0-e^{-23}H+e^{40}
           \left\{D+\frac{\sqrt{\Lambda_A}+\Lambda}{\sqrt m}\right\}
 <R_0\le\be,
\end{equation}
where $R_0=\mathcal R(B_1-p_1,\ldots,B_m-p_m)\le\be$ by \eqref{eq:finite-input}
and the error bracket in \eqref{eq:small-comparison} is strictly smaller
than $e^{-23}H$ since $D/H\le(\Lambda+\eps_0^2)^{3/2}\le3e^{-129/2}$ and
$m\ge e^{\mathcal A}$.

\subsection{Regime II: larger residual variance ($\Lambda\ge e^{-43}$)}
\label{subsec:large-residual}

Suppose now that $\Lambda\ge e^{-43}$ and $|z|\le4$. For every integer $k$
with $|k-x|\le\max(1,\sqrt\Lambda)$ (at least $\sqrt\Lambda$ integers, all
satisfying $|k-\mu|\le6\sqrt m$), tilt the Bernoulli probabilities
exponentially as in Appendix~\ref{app:conditional} so that the tilted count
mean equals $k$, and project $A$ orthogonally to $K_0-k$. Conditional on
$K_0=k$, we have $W=m_k^0+R_k$, where under the tilted measure $R_k$ is a sum
of independent centered pair residuals and $Z_j$'s satisfying
\begin{equation}\label{eq:tilted-residual}
 L:=\Var_\theta R_k\in[\Lambda/2,2\Lambda],\qquad
 \Var_\theta K_0\ge \frac m{10},\qquad |m_k^0|\le\sqrt\Lambda,
 \qquad\Cov_\theta(K_0,R_k)=0.
\end{equation}
Applying Lemmas~\ref{lem:conditional-fourier} and~\ref{lem:minorant} to both
quarter-intervals $(1/4,1/2)$ and $(-1/2,-1/4)$ with $L_0=\frac12e^{-43}$,
$C_0=4+2e^{43/2}$, $A_g=4e^{43/2}$, $\omega=16e^{A_g^2/6}$, and $T_g=4\omega$
shows that conditional on $K_0=k$, $R_k-(x-k-m_k^0)$ falls in each
quarter-interval with probability at least $\phi(A_g)/(8\omega\sqrt L)$.
Multiplying by $b_k\ge e^{-100}/\sqrt m$ and summing over the chosen integers
$k$ yields
\begin{equation}\label{eq:quarter-mass}
 \Pp\{S-x\in(1/4,1/2)\}\ge \frac q{\sqrt m},\qquad
 \Pp\{S-x\in(-1/2,-1/4)\}\ge \frac q{\sqrt m}.
\end{equation}
Consequently, the one-sided jitter smoothing losses satisfy
$J(x+1/2)-F_S(x)\ge q/(2\sqrt m)$ and $F_S^-(x)-J(x-1/2)\ge q/(2\sqrt m)$.
Multiplying by $P\ge B/2\ge\sqrt{3m}/8$ subtracts at least $q/10$ from the
main term of \eqref{eq:local-envelope} (which is at most $\be+200\lambda$ by
Esseen's Bernoulli inequality $3V+|\kappa_0|\le\ce T_0$ and
\eqref{eq:moment-budget}), bounding either branch below
$\be-q/10+200\lambda+4E_m(T)+m^{-1/2}<\be-q/20$. This completes the proof of
Theorem~\ref{thm:local}.
