\section{Conditional moment bounds and the one-sided clipping inequality}
\label{app:conditional}

This appendix proves the count-conditioned moment estimates
\eqref{eq:conditional-mean}--\eqref{eq:conditional-label} and the one-sided
clipping inequality \eqref{eq:one-sided} used in Section~\ref{sec:local}.

\subsection{Exponential tilting and orthogonal projection}

Let $p_i\in[1/4,3/4]$, $v_i=p_i(1-p_i)$, $\sum_{i=1}^m v_i a_i=0$,
$\Lambda_A=\sum_{i=1}^m v_i a_i^2>0$, $m\ge10^{12}$, and $|k-\mu|\le6\sqrt m$.
Choose the unique tilt parameter $\theta\in[-\log2,\log2]$ so that the tilted
Bernoulli variables $B_i\sim\Bern(r_i)$ with
\[
 r_i=\frac{p_i e^\theta}{1-p_i+p_i e^\theta}\in\left[\frac18,\frac78\right],\qquad
 w_i=r_i(1-r_i)\ge\frac7{64}>\frac1{10},\qquad V_\theta=\sum_{i=1}^m w_i
\]
satisfy $\sum_{i=1}^m r_i=k$. Then
\begin{equation}\label{eq:tilt-quantities}
 |\theta|\le\frac{60}{\sqrt m},\qquad \left|\frac{w_i}{v_i}-1\right|\le2|\theta|,
       \qquad |r_i-p_i-\theta v_i|\le\theta^2v_i.
\end{equation}
Because the exponential factor $e^{\theta K_0}$ is constant on the event
$\{K_0=k\}$, the conditional distribution given $K_0=k$ is identical under
the original and tilted measures. Setting
\[
 c_\theta=\frac{\sum_{i=1}^mw_i a_i}{V_\theta},\qquad b_i=a_i-c_\theta,
 \qquad L=\sum_{i=1}^m b_i(B_i-r_i),\qquad Q=\sum_{i=1}^m b_i^2,
 \qquad Q_\theta=\sum_{i=1}^mw_i b_i^2,
\]
we have $A=m_k^0+L$ on $\{K_0=k\}$, where $\Cov_\theta(K_0,L)=\sum_{i=1}^mw_ib_i=0$
and
\begin{equation}\label{eq:tilted-gap-data}
 |m_k^0|=\left|\sum_{i=1}^m a_i(r_i-p_i)\right|
      \le1800\frac{\sqrt{\Lambda_A}}{\sqrt m},\qquad
 |Q_\theta-\Lambda_A|\le180\frac{\Lambda_A}{\sqrt m},
 \qquad Q\le16\Lambda_A.
\end{equation}

\subsection{Fourier inversion for count-conditioned moments}

For $|t|\le1/4$, let $z_i(t)=r_ie^{\ii t}/(1-r_i+r_ie^{\ii t})$ and let
$c_{i,1}=z_i-r_i$, $c_{i,2}=z_i(1-z_i)$, $c_{i,3}=z_i(1-z_i)(1-2z_i)$, and
$c_{i,4}=z_i(1-z_i)(1-6z_i(1-z_i))$ be the first four cumulants of $B_i-r_i$
twisted by $e^{\ii t(B_i-r_i)}$. Because $\sum_{i=1}^mw_ib_i=0$, the twisted
cumulant sums $M_j(t)=\sum_{i=1}^mb_i^jc_{i,j}(t)$ satisfy
\begin{gather}
 |M_1(t)|\le10\sqrt{mQ}\,t^2,\qquad
 |M_2(t)-Q_\theta|\le20Q|t|,\qquad |M_2(t)|\le6Q,\notag\\
 |M_3(t)|\le20Q^{3/2},\qquad |M_4(t)|\le100Q^2.              \label{eq:complex-moments}
\end{gather}
With $F_\theta(t)=\E_\theta e^{\ii t(K_0-k)}$, we have $|F_\theta(t)|\le e^{-mt^2/50}$
on $[-\pi,\pi]$ and $\Pp_\theta(K_0=k)\ge1/(3\sqrt m)$ by Lemma~\ref{lem:count-estimate}.
Dividing $(2\pi)^{-1}\int_{-\infty}^\infty e^{-mt^2/50}|t|^j\,dt$ by
$\Pp_\theta(K_0=k)$ gives the upper bounds
\begin{equation}\label{eq:gaussian-moment-table}
             7,\quad 25m^{-1/2},\quad 175m^{-1},\quad
                          13125m^{-2},\quad 3\cdot10^8m^{-4}
\end{equation}
for $j=0,1,2,4,8$, while on $1/4\le|t|\le\pi$ the remaining product of at
least $m-4$ Bernoulli factors is bounded by $e^{-m/810}$. Integrating the
exact Fourier integrands $F_\theta M_1$, $F_\theta(M_1^2+M_2)$, and
$F_\theta(M_1^4+6M_1^2M_2+3M_2^2+4M_1M_3+M_4)$ over $[-\pi,\pi]$ yields
\begin{equation}\label{eq:projected-moments}
 \begin{gathered}
 |\E(L\mid K_0=k)|\le2000\frac{\sqrt Q}{\sqrt m},\\
 |\Var(L\mid K_0=k)-Q_\theta|\le2000\frac{Q}{\sqrt m},
 \qquad \E(L^4\mid K_0=k)\le2000Q^2.
 \end{gathered}
\end{equation}
Combining \eqref{eq:tilted-gap-data} and \eqref{eq:projected-moments} proves
\eqref{eq:conditional-mean}--\eqref{eq:conditional-gap}, and comparing the
one-coordinate deleted count mass to $\Pp_\theta(K_0=k)$ via
Lemma~\ref{lem:count-estimate} and \eqref{eq:tilt-quantities} proves
\eqref{eq:conditional-label}.

\subsection{Proof of the one-sided clipping inequality \eqref{eq:one-sided}}

Following \citet[Lemma~3.2 and equation~(83)]{HeCheng2026}, suppose first
that $\E V_0=0$ and $|V_0|\le a:=1/10$. For $u\ge0$,
$\E(V_0)_+\le u+a\Pp(V_0>u)$ and $f(u)\ge3u/4$ imply
$\Pp(V_0>u)+f(u)\ge\frac34\E(V_0)_+=\frac38\E|V_0|$. For $u=-r$ with
$0<r\le a$, putting $t=\Pp(V_0>-r)$, $\E V_0=0$ gives $0\le at-r(1-t)$, so
$r\le2at$ and $\E|V_0|=2\E(V_0)_+\le2at$, whence
$t+f(-r)\ge t-5r/4\ge3t/4\ge\frac38\E|V_0|$. For $a<r\le13/20$,
$\Pp(V_0>-r)=1$ and $1+f(-r)\ge3/16\ge\frac38\E|V_0|$.

For general centered $Y$ with $M_4=\E Y^4\le1/160000$, truncate $Y$ at
$\pm1/20$ to obtain $T$, and set $b=\E T$ and $V_0=T-b$. By Markov's
inequality, $|b|\le8000M_4\le1/20$, $\Pp(T\ne Y)\le160000M_4$,
$|V_0|\le1/10$, and $\E|V_0|\ge\E|Y|-16000M_4$. Applying the bounded case to
$V_0$ at $u-b\in[-13/20,13/20]$ gives the first inequality in
\eqref{eq:one-sided} with error $(3/8)16000+160000+(5/4)8000=176000$, and
applying it to $-Y$ and $-f(-\cdot)$ gives the second.
