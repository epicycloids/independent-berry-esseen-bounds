\section{Retained smoothing losses and the Esseen core}\label{sec:shape}

In this section we establish the second assertion of Theorem~\ref{thm:near}:
any sequence of arrays satisfying \eqref{eq:near-assumption} concentrates in
$W_3$ around a padded i.i.d.\ Esseen array. Normalize total variance to
$B=1$ and omit the row index $n$, so that $R=L\to0$. For each summand $X_i$,
put $v_i=\E X_i^2$, $\sigma_i=\sqrt{v_i}$, $\rho_i=\E|X_i|^3$,
$\kappa_i=\E X_i^3$, $b_i=\E|X_i|$, and define the scalar moment deficit
\[
 \delta_i=\sqrt{10}\rho_i-|\kappa_i|-3b_iv_i.
\]

\begin{lemma}[Retained moment, tail, and orientation losses]\label{lem:retained}
Under \eqref{eq:near-assumption}, as $R\to0$,
\begin{equation}\label{eq:retained-losses}
 \sum_{i=1}^N\delta_i=o(R),\qquad
 \sum_{i=1}^N\E\bigl[|X_i|^3\ind_{\{|X_i|>R^{1/6}\}}\bigr]=o(R),\qquad
 \sum_{i=1}^N|\kappa_i|-\left|\sum_{i=1}^N\kappa_i\right|=o(R),
\end{equation}
and $\sum_{i=1}^N\sigma_i^3/R\to1/\be$. Moreover, after reflecting the array
if necessary so that $\sum_{i=1}^N\kappa_i\ge0$, let $P_i^*=\Law(\sigma_i W)$
be the positively oriented Esseen two-point distribution of variance $v_i$.
Then
\begin{equation}\label{eq:shape-W3}
                  \sum_{i=1}^N W_3^3(X_i,P_i^*)=o(R).
\end{equation}
\end{lemma}
\begin{proof}
We first establish quantitative stability in Shevtsova's scalar moment
inequality $\delta(X)\ge0$ \citep[Theorem~3.1, pp.~256--262]{Shevtsova2012}.
Put $s=\sqrt{\pe\qe}$, $a=\sqrt{\qe/\pe}$, $b=\sqrt{\pe/\qe}$, and $t=\te$.
For a centered unit-variance variable $X$ oriented to have $\E X^3\ge0$, set
$Q(x)=-6sx^2+6tx+6s$ and $f(x)=x^3-\sqrt{10}|x|^3+3|x|$. Direct factorization
gives
\[
 Q(x)-f(x)=
 \begin{cases}
 (\sqrt{10}-1)(x-a)^2(x+2s),&x\ge0,\\
 (\sqrt{10}+1)(-x-b)^2(-x+2s),&x\le0.
 \end{cases}
\]
Since $\E Q(X)=0$ and $\E f(X)=-\delta(X)$, we have
$\delta(X)=\E[Q(X)-f(X)]\ge c_0\E[\dist(X,\{-b,a\})^3]$ with
$c_0=2\pe(\sqrt{10}-1)>0$. Rounding $X$ to $a$ on $[0,\infty)$ and to $-b$ on
$(-\infty,0)$ at cubic cost $e\le\delta(X)/c_0$ and then matching the atom
probabilities to $\pe,\qe$ gives $W_3^3(X,W)\le4e+4(a+b)^2e^{1/3}$. Rescaling
each summand by $\sigma_i$ and applying H\"older's inequality yields, for
individually oriented Esseen targets $P_i^{\rm oriented}$,
\begin{equation}\label{eq:shape-stability}
 \frac{\sum_{i=1}^N W_3^3(X_i,P_i^{\rm oriented})}{R}
 \le\frac4{c_0}\bar\delta+
       4(a+b)^2c_0^{-1/3}\bar\delta^{1/3},\qquad
 \bar\delta=\frac1R\sum_{i=1}^N\delta_i.
\end{equation}

Next, we retain the nonnegative moment and truncation losses in the smoothing
proof of \citet[pp.~289--296]{Shevtsova2012}. With truncation level
$U=R^{1/6}$, set $Y_i=X_i\ind_{\{|X_i|\le U\}}$, $Z_i=Y_i-\E Y_i$, and
$T_i=\E[|X_i|^3\ind_{\{|X_i|>U\}}]$. By \citet[Lemma~4.3 and $I_{14}$--$I_{18}$, pp.~272--295]{Shevtsova2012},
$\Delta\le\phz(I_{16}+R/2)+O(R^{7/6})$, where
$I_{16}=\frac{\sqrt{10}}{6}R-\frac16\sum_i\delta(Z_i)+I_{17}-I_{18}$,
$I_{17}=\frac{\sqrt{10}}{6}\sum_i(\E|Z_i|^3-\E|Y_i|^3)$, and
$I_{18}=\frac{\sqrt{10}-1}{6}\sum_iT_i$. Since $|\E Y_i|\le T_i/U^2$ gives
$\E|Z_i|^3-\E|Y_i|^3\le\frac13T_i$ for $R\le1/4096$, we obtain
\[
 \Delta\le\ce R-\frac{\phz}{6}\sum_{i=1}^N\delta(Z_i)
            -\frac{\phz(\sqrt{10}-1)}{12}\sum_{i=1}^NT_i+O(R^{7/6}).
\]
Under \eqref{eq:near-assumption}, both sums must be $o(R)$, which proves the
first two relations in \eqref{eq:retained-losses}.

To prove the third relation in \eqref{eq:retained-losses}, write
$K_3=\sum_i\kappa_i$, $\nu=1+\sum_i b_iv_i/R\in[1,2]$, and $T=2\pi/(\nu R)$.
On every compact frequency interval, $\chi(u)=\E e^{\ii uS}=e^{-u^2/2}(1-\ii K_3u^3/6)+o(R)$,
so dominated convergence in the low-frequency Prawitz integral of
\citet[Lemma~4.1 and (4.4)--(4.7), pp.~263--267]{Shevtsova2012} replaces
$\sum_i|\kappa_i|$ by $|K_3|$:
\begin{equation}\label{eq:signed-loss}
 \Delta\le\phz\left\{\frac R2+\frac12\sum_{i=1}^N b_iv_i
                                  +\frac{|K_3|}{6}\right\}+o(R)
 =\ce R-\frac{\phz}{6}\left\{\sum_{i=1}^N\delta_i+
                  \sum_{i=1}^N|\kappa_i|-|K_3|\right\}+o(R),
\end{equation}
forcing $\sum_i|\kappa_i|-|K_3|=o(R)$ and proving \eqref{eq:shape-W3} via
\eqref{eq:shape-stability}. Finally, Theorem~4.14 of \citet[pp.~300--303]{Shevtsova2012}
gives $\Delta\le cR+K(c)\sum_i\sigma_i^3+O_c(R^{4/3})$ for every fixed
$c>2\phz/3$, where $K(\ce)=0$ and $K'(\ce)=-\be$; varying $c$ on both sides
of $\ce$ forces $\sum_i\sigma_i^3/R\to1/\be$.
\end{proof}

\begin{lemma}[High-frequency Fourier window and subset variance control]\label{lem:window}
For every fixed $\epsilon>0$ and sufficiently small fixed $t_0>0$, set
\[
 T'=(1+\epsilon)\frac{2\pi}{\nu R},\qquad
 H=2\int_{t_0}^1|K(u)|\,|\chi(T'u)|\,du,
\]
where $K(u)$ is Prawitz's smoothing kernel. Under \eqref{eq:near-assumption},
$\nu\to\nus$ and
\begin{equation}\label{eq:window}
 \liminf_{n\to\infty} \frac{H}{R}\ge\frac{\phz\nus\epsilon}{2(1+\epsilon)}>0.
\end{equation}
Consequently, there exist $\eta>0$ and $e_n\to0$ such that for every subset
of indices $A\subseteq\{1,\ldots,N_n\}$,
\begin{equation}\label{eq:subset-inverse}
             \sum_{i\in A}v_i\le\eta^{-1}
                          \frac{\sum_{i\in A}\rho_i}{R}+e_n,
\end{equation}
and $\sum_{i=1}^{N_n}W_2^2(X_i,P_i^*)\to0$.
\end{lemma}
\begin{proof}
Applying Prawitz's smoothing inequality at the enlarged cutoff $T'$ bounds
the low-frequency integral by $\phz|K_3|/6+o(R)$ and the Gaussian cutoff term
by $\phz\nu R/[2(1+\epsilon)]+O(R^2)$; comparing with $\Delta\sim\ce R$ yields
\eqref{eq:window}. Since $|\chi_A(u)|\le\exp(-V_Au^2/2+2R_A|u|^3/3)$ for
$V_A=\sum_{i\in A}v_i$ and $R_A=\sum_{i\in A}\rho_i$, on $u\in[a/R,C/R]$ with
$\eta\le3/(8C)$ any subset with $R_A/V_A\le\eta R$ has
$|\chi(u)|\le\exp(-a^2V_A/(4R^2))$, which combined with \eqref{eq:window}
gives \eqref{eq:subset-inverse}. Applying \eqref{eq:subset-inverse} to the
exceptional set $\{i:W_3^3(X_i,P_i^*)>\zeta\rho_i\}$ and H\"older's inequality
on its complement proves $\sum_i W_2^2(X_i,P_i^*)\to0$.
\end{proof}

\begin{proposition}[Concentration at a single variance scale]\label{prop:core}
The i.i.d.\ Esseen approximation of Theorem~\ref{thm:near} holds. Moreover,
in gap units $h=\nus R$, there exists a subset $I\subseteq\{1,\ldots,N_n\}$
of cardinality $M=|I|$ carrying $1-o(1)$ of both total variance and $R$ such
that
\begin{equation}\label{eq:raw-core}
 M R^2\longrightarrow\be^2,\qquad
 \sup_{i\in I}W_3\bigl(\Law(X_i/h),\Law(B_{\pe}-\pe)\bigr)\longrightarrow0,
\end{equation}
and in these units the rescaled total variance $V$ and third-moment sums
$P,\kappa$ satisfy $V\sim\pe\qe M$, $P/V\to\re$, and $\kappa/V\to\te$.
\end{proposition}
\begin{proof}
Consider the variance-weighted probability measures
$\mu_n=\sum_i v_i\delta_{\sigma_i/R}$, whose means $\sum_i\sigma_i^3/R$
converge to $s_0=1/\be$, and which have no limiting mass at $0$ by
\eqref{eq:subset-inverse}. For each $\epsilon>0$, \eqref{eq:window} supplies
a frequency $u_n=w_n/R$ with $\limsup w_n\le2\pi(1+\epsilon)/\nus$ and
$|\chi(u_n)|\ge c_\epsilon R$, so $\sum_i(1-|\chi_i(u_n)|^2)=O_\epsilon(\log(1/R))$.
Transferring to the Esseen targets $P_i^*$ via $\sum_i W_2^2(X_i,P_i^*)\to0$
shows that any weak limit $\mu$ of $\mu_n$ satisfies
$\int\sin^2((a+b)zw_\epsilon/2)\,d\mu(z)=0$, so $\supp\mu\subset[s_0/(1+\epsilon),\infty)$.
Letting $\epsilon\downarrow0$ gives $\supp\mu\subset[s_0,\infty)$, which
together with $\int z\,d\mu\le s_0$ forces $\mu=\delta_{s_0}$ and
$\int|z-s_0|\,d\mu_n\to0$. Restricting to a slowly shrinking neighborhood of
$s_0$ and discarding indices with large $W_3(\Law(X_i/h),\Law(B_{\pe}-\pe))$
yields \eqref{eq:raw-core}.
\end{proof}

\section{First nonzero Fourier peak and integer lifts}\label{sec:alias}

\begin{lemma}\label{lem:first-alias}
At a spacing $h_n$ asymptotic to $\nus R_n/B_n^2=\nus L_nB_n$ (so $h_n\sim\nus R_n$
under the normalization $B_n=1$), the rescaled summands
$Y_i=X_i/h_n$ decompose as
\[
 Y_i=J_i+Z_i,\qquad J_i=K_i-\E K_i,\qquad K_i\in\Z,\qquad
 \E Z_i=0,\qquad |Z_i|\le1,
\]
where the pairs $(K_i,Z_i)$ are independent across $i$, $\sup_{i\in I}W_3(K_i,\Bern(\pe))\to0$
(up to integer shifts) on the core $I$ of size $M$ from Proposition~\ref{prop:core},
$\sum_{i\notin I}(\Var K_i+\E|J_i|^3)=o(M)$, and for $K=\sum_i K_i$ and
$T=\sum_i Z_i$,
\begin{equation}\label{eq:initial-lifts}
 Q=\Var T=O(1),\qquad \Cov(K,T)=o(\sqrt M),\qquad
 A=\Var K\sim\pe\qe M.
\end{equation}
\end{lemma}
\begin{proof}
Taking $0<\epsilon<1/2$ in Lemma~\ref{lem:window}, the frequency window in
gap units lies inside $(0,4\pi)$, where the core product $\prod_{i\in I}|\E e^{\ii sY_i}|$
is exponentially small outside a neighborhood of $2\pi$ and attains its local
maximum at $s_0\to2\pi$ with $\prod_{i=1}^N|\E e^{\ii s_0Y_i}|\ge c>0$.
Rescaling $h_n$ by $2\pi/s_0\to1$ places this core maximum at $2\pi$.

Aligning each summand by $\theta_i=(2\pi)^{-1}\arg\E e^{2\pi\ii Y_i}$, write
$Y_i=\theta_i+K_i+E_i$ with $K_i\in\Z$ and $E_i\in[-1/2,1/2]$. Then
$\E\sin(2\pi E_i)=0$, $r_i=\E\cos(2\pi E_i)>0$, and
$\sum_i\E E_i^2\le\frac18\sum_i(1-r_i)=O(1)$. Setting $Z_i=E_i-\E E_i$ gives
$Q=\sum_i\E Z_i^2=O(1)$. Stationarity of the core log-modulus at $2\pi$ is
$\sum_{i\in I}\E[Y_i\sin(2\pi E_i)]/r_i=0$; since $r_i\to1$ uniformly on $I$
and $|\sin(2\pi E_i)-2\pi E_i|\le K|E_i|^3$, this implies
$\Cov(K,T)=\Cov(\sum_i Y_i,T)-Q=o(\sqrt M)$, proving \eqref{eq:initial-lifts}.
\end{proof}
