\section{Binomial phase and the optimal number of zero coordinates}
\label{sec:iid-phase}

In this section we analyze the $m$-summand i.i.d.\ binomial maxima $B_m$ and
their padded envelope $P_N:=\max_{1\le m\le N}B_m$. By Theorem~\ref{thm:common},
$P_N=C_N^{\mathrm{com}}$ is the exact $N$-summand common-diameter maximum.

Put $s=\sqrt{10}$ and define the algebraic constants
\begin{equation}\label{eq:phase-constants}
\begin{gathered}
 p=\frac{4-s}2,\qquad q=1-p,\qquad t=s-3,\qquad u=p-t,
 \qquad c_*=\frac{20+s}{72},\\
 Q_*=\frac{\phz(867+287s)}{25920},\qquad
 \alpha=\frac{\phz(60+19s)}{30},\qquad
 b=\frac{1010-397s}{12960},\\
 \delta=c_*-b=\frac{2590+577s}{12960},\qquad
 \eta=\frac{Q_*}\alpha=\frac{2510-747s}{8640},\qquad
 \omega=\frac{7773-2447s}{8640}.
\end{gathered}
\end{equation}
For a general Bernoulli parameter $v\in(0,1)$, write $a(v)=v(1-v)$,
$r(v)=v^2+(1-v)^2$, and $\tau(v)=1-2v$; for $x\in\R$, write
$\langle x\rangle=x-\operatorname{round}(x)$.

\begin{theorem}[Asymptotic zero padding and record gaps in $P_N$]\label{thm:iid-padding}
For every sufficiently large integer $N$, $P_N=\max_{1\le m\le N}B_m$ is
attained at a unique active count $m_N$, and the number of zero coordinates
$j_N=N-m_N$ belongs to $\{0,1,2,3,4\}$. As $N\to\infty$, the limiting
frequencies of $j_N=0,1,2,3,4$ exist and equal
\[
 p-\omega,\qquad p-\omega,\qquad t,\qquad\omega,\qquad\omega,
\]
respectively. Moreover, the gaps between successive record active counts
$m_N$ eventually belong to $\{2,3,5\}$, and each of the three gaps $2,3,5$
occurs infinitely often with positive asymptotic density.
\end{theorem}

\subsection{Uniform Edgeworth--Euler--Maclaurin expansion at an integer threshold}

Let $F_{m,v}$ be the distribution function of $\Bin(m,v)$. For an integer
$k$ and $v\in(0,1)$, set $c=mv-k$, $d=(ma(v))^{-1/2}$, and
$D_{m,k}(v)=F_{m,v}(k)-\Phi(-cd)$.

\begin{lemma}\label{lem:phase-uniform}
For every compact interval $J\subset(0,1)$ and $M>0$, uniformly for
$|c|\le M$ and $v=(k+c)/m\in J$,
\begin{equation}\label{eq:phase-cdf}
 D_{m,k}(v)=\phz\bigl[d(1/2+\tau/6)+d^3C(\tau,-c)
                         +d^5T_5(\tau,-c)\bigr]+O(d^7),
\end{equation}
and the same asymptotic expansion holds upon differentiating with respect to
$c$ at fixed $m,k$ (and extends to any fixed order in $d$). Here
\begin{align*}
C(\tau,\beta)={}&\frac{-1080(1+\tau)\beta^2
 -(90\tau^2+1080\tau+630)\beta
 +\tau^3-45\tau^2-189\tau-135}{4320},\\
T_5(\tau,\beta)={}&\frac{3+5\tau}{48}\beta^4
 +\frac{27+60\tau+25\tau^2}{288}\beta^3
 +\frac{15+31\tau+25\tau^2+\tau^3}{192}\beta^2\\
&+\frac{549+1320\tau+1030\tau^2+120\tau^3+5\tau^4}{23040}\beta\\
 &+\frac{189+927\tau+126\tau^2+234\tau^3+21\tau^4-25\tau^5}{193536}.
\end{align*}
\end{lemma}

\begin{proof}
By Fourier inversion on $[-\pi,\pi]$ with $|1-v+ve^{\ii\xi}|^m=(1-4a(v)\sin^2(\xi/2))^{m/2}\le e^{-2z^2/\pi^2}$
($z=\xi\sqrt{ma(v)}$), expanding $\log\E e^{\ii\xi(\Bin(m,v)-mv)}$ on
$|z|\le m^\varepsilon$ gives a uniform local Hermite expansion for each point
mass $\Pp(\Bin(m,v)=j)$ and its $c$-derivatives (using
$\partial_c(1-v+ve^{\ii\xi})^m=(e^{\ii\xi}-1)(1-v+ve^{\ii\xi})^{m-1}$) to
error $O(d^{12})$. Summing over $j\le k$ via Euler--Maclaurin summation
(with lattice corrections $df/2+d^2f'/12-d^4f'''/720$ and Bernoulli cumulant
ratios $\tau,1-6a,\tau(1-12a),1-30a+120a^2,\tau(1-60a+360a^2)$) leaves only
odd powers of $d$ at $-cd$ and yields \eqref{eq:phase-cdf} (compare
\citet[Remark~4.3, p.~25]{Schulz2016} for the cubic polynomial $C(\tau,\beta)$).
\end{proof}

Dividing \eqref{eq:phase-cdf} by $r(v)d$ gives the branch ratio expansion
\begin{equation}\label{eq:phase-efficiency}
 R_{m,+,k}(v)=E(v)-\frac{Q(v,c)}m-\frac{H(v,c)}{m^2}+O(m^{-3}),
\end{equation}
where
\[
 E(v)=\phz\frac{3+\tau(v)}{3(1+\tau(v)^2)},\qquad
 Q(v,c)=-\frac{\phz C(\tau(v),-c)}{r(v)a(v)},\qquad
 H(v,c)=-\frac{\phz T_5(\tau(v),-c)}{r(v)a(v)^2}.
\]
For each fixed $v$, the quadratic function $c\mapsto Q(v,c)$ attains its
minimum at $c_0(v)=(\tau(v)^2+12\tau(v)+7)/[24(1+\tau(v))]$, with minimum
value
\begin{equation}\label{eq:phase-qmin}
 Q_{\min}(v)=\frac{\phz\{345+72\tau-498\tau^2-8\tau^3-23\tau^4\}}
 {4320(1+\tau)(1-\tau^4)}.
\end{equation}

\subsection{Second-order expansion of $B_m$ and the five-state phase envelope}

By Proposition~6.1 and pp.~29--31 of \citet{Schulz2016}, noncentral thresholds
and parameters $v\notin[1/6,5/6]$ cannot compete with $\ce-O(m^{-1})$. After
reflecting to $v\le1/2$, $E(v)$ has its unique maximum $\ce$ at $v=p$ with
$E(v)=\ce-\alpha(v-p)^2+O((v-p)^3)$, and the positive closed central
threshold dominates the left-limit central threshold (whose leading term has
$1/2-\tau/6<1/2+\tau/6$). Comparing \eqref{eq:phase-efficiency} with the
trial choice $v=(k+c_*)/m$ (for $k=\operatorname{round}(mp-c_*)$) gives
$v-p=O(m^{-1})$ and $c-c_0(v)=O(m^{-1/2})$, and branch stationarity
$\partial_c R_{m,+,k}=0$ refines this to $c-c_0(v)=O(m^{-1})$ and
$v=(k+c_*)/m+O(m^{-2})$. Setting $e=k+c_*-mp$ and completing the square in
$\alpha e^2+Q_p(p,c_*)e+H(p,c_*)=\alpha(e-b)^2+H_*$ yields the uniform
second-order expansion
\begin{equation}\label{eq:phase-bm}
 B_m=\ce-\frac{Q_*}m-\frac{H_*+
 \alpha\operatorname{dist}(mp-\delta,\Z)^2}{m^2}+O(m^{-3}).
\end{equation}

Because $B_m<\ce$ for all $m$ and $B_m\to\ce$, any maximizing active count
$m_N$ for $P_N$ satisfies $m_N\to\infty$, and comparing \eqref{eq:phase-bm}
at $m_N$ and $N$ forces $j_N=N-m_N=O(1)$. With
$\theta_N=\operatorname{round}(Np-\delta)-(Np-\delta)\in[-1/2,1/2]$, we have
for bounded $j\ge0$:
\[
 N^2(B_{N-j}-B_N)=\alpha\bigl\{\theta_N^2-
 \operatorname{dist}(\theta_N+jp,\Z)^2-\eta j\bigr\}+O(N^{-1}).
\]
Maximizing $B_{N-j}$ therefore minimizes $f_j(\theta)=\eta j+\operatorname{dist}(\theta+jp,\Z)^2$.
Define the five transition points in $[-1/2,1/2]$ by
\begin{equation}\label{eq:phase-boundaries}
\begin{aligned}
 T_{13}&=-\frac{p+u}2+\frac\eta t,& T_{30}&=-\frac u2-\frac{3\eta}{2u},\\
 T_{02}&=\frac t2+\frac\eta t,&T_{24}&=\frac{3t}2+\frac\eta t,\\
 T_{41}&=\frac{q+2t}2-\frac{3\eta}{2u}.
\end{aligned}
\end{equation}
On $[-1/2,1/2]$, the lower envelope $\min_{j\ge0}f_j(\theta)$ is achieved on
six intervals by $j\in\{0,1,2,3,4\}$:
\begin{center}
\begin{tabular}{cc}
\toprule
Phase interval for $\theta_N$&Optimal zero count $j$\\
\midrule
$[-1/2,T_{13}]$&1\\
$[T_{13},T_{30}]$&3\\
$[T_{30},T_{02}]$&0\\
$[T_{02},T_{24}]$&2\\
$[T_{24},T_{41}]$&4\\
$[T_{41},1/2]$&1\\
\bottomrule
\end{tabular}
\end{center}
Every $j\ge5$ satisfies $f_j(\theta)\ge5\eta$, whereas $\min_{0\le j\le2}f_j(\theta)\le2\eta+p^2/4<5\eta-59/10000$
on $[-1/2,1/2]$. Hence $j_N\in\{0,1,2,3,4\}$ for all large $N$, and by Weyl
equidistribution of the irrational rotation $\theta_N$, the limiting
frequencies of $j_N=0,1,2,3,4$ equal the lengths
$p-\omega,p-\omega,t,\omega,\omega$ of the corresponding phase intervals.

\subsection{$O(N^{-1})$ boundary drift and quadratic-field separation}

To resolve ties near the five transition points \eqref{eq:phase-boundaries},
carrying Lemma~\ref{lem:phase-uniform} to order $d^7$ gives the third-order
expansion on a winning branch:
\begin{equation}\label{eq:phase-cubic}
 B_m=\ce-\frac{Q_*}m-\frac{F_2(e)}{m^2}+\frac{G(e)-J_*}{m^3}+O(m^{-4}),
\end{equation}
where $F_2(e)=\alpha e^2+Q_pe+H_0$, $K(e)=(2\alpha+Q_{pc})e+Q_p+H_c$, and
$G(e)=E_3e^3-\frac12Q_{pp}e^2-H_pe+K(e)^2/(2Q_{cc})$. Expanding
$B_{N-j}$ in powers of $N^{-1}$ shows that the crossing between parabolas
$i$ and $j$ occurs at $\theta=T+D_{ij}/N+O(N^{-2})$, where explicit evaluation
gives
\begin{equation}\label{eq:phase-drift}
\begin{gathered}
 D_{13}=D_{02}=D_{24}=D_A
   =\frac{129918319}{529079040}
       +\frac{191267395}{1693052928}s,\\
 D_{30}=D_{41}=D_B
   =-\frac{663979357}{2351462400}
       -\frac{20072514361}{141087744000}s.
\end{gathered}
\end{equation}
For each of the five transition points $T$, writing $h_N=\theta_N-T=(A_N+B_Ns)/L_0$
with $L_0=25920$ and $A_N,B_N\in\Z$, the field norm
$\nu_N=(A_N^2-10B_N^2)/L_0^2\in L_0^{-2}\Z$ is bounded whenever
$|h_N|\le C/N$, and $Nh_N=-\nu_N/s+O(N^{-1})$. Because both $D_A$ and $D_B$
have nonzero rational parts in $\mathbb Q(s)$, $-\nu_N/s$ is bounded away
from $D_A$ and $D_B$, so $|\theta_N-T-D_{ij}/N|\ge\varepsilon/N$ for all
large $N$. This dominates the $O(N^{-2})$ remainder and proves that $m_N$ is
unique for all large $N$. Finally, the exact translation identities
$T_{02}-p=T_{13}$, $T_{13}+q=T_{24}$, and $T_{30}+q=T_{41}$ preserve the
drift classes $D_A$ and $D_B$, so under $\theta_{N+1}=\langle\theta_N-p\rangle$
the transitions of $j_N$ follow the directed graph
\[
 0\longrightarrow1,\qquad1\longrightarrow0\text{ or }2,\qquad
 2\longrightarrow0\text{ or }3,\qquad3\longrightarrow4,\qquad
 4\longrightarrow0,
\]
whose return cycles to $j_N=0$ have lengths $2,3,5$ with positive limiting
frequencies $u-\omega,t-\omega,\omega$. This completes the proof of
Theorem~\ref{thm:iid-padding}.
