\section{Support geometry and weighted variance setup}\label{sec:weighted-shape}

We now assume the third-moment upper bound
\begin{equation}\label{eq:weighted-resource-range}
                       \beta\le\beta_*:=\frac{18481}{25000}=0.73924
\end{equation}
and prove Theorem~\ref{thm:weighted-variance} and
Corollary~\ref{cor:weighted-regularity}. Throughout
Sections~\ref{sec:weighted-shape}--\ref{sec:weighted-iteration}, we consider
a maximizing representation with at most three support points per summand as
in Theorem~\ref{thm:structure} and threshold data \eqref{eq:max-data}.
Omitting deterministic zero summands, write
\[
 H_i=\diam(\supp X_i),\qquad v_i=\E X_i^2,\qquad
 V(h)=\sum_{H_i\le h}v_i,\qquad
 W_s=\sum_{i\ge1} v_iH_i^{-s}.
\]

\begin{lemma}[Strict scalar curvature margin]\label{lem:weighted-scalar}
Let $C\ge\ce$, $D=C\beta$, and define
\[
 L_0=\frac{15D+\phi'''(z)}4,\qquad
 c_\varepsilon=3C+\frac{\varepsilon\phi''(z)}{2}\qquad
 (\varepsilon\in\{-1,1\}).
\]
Under \eqref{eq:weighted-resource-range}, $c_\varepsilon>0$ and
$c_\varepsilon^2>3DL_0$ for both $\varepsilon\in\{-1,1\}$.
\end{lemma}
\begin{proof}
With $r=486833/500000>\phi(0)/\ce=6/(3+\sqrt{10})$ and $x=|z|$,
$(c_\varepsilon^2-3DL_0)/C^2$ is bounded below by
\[
 \left[3-\frac r2|x^2-1|e^{-x^2/2}\right]^2
 -\frac{3\beta_*r}{4}|3x-x^3|e^{-x^2/2}
 -\frac{45}{4}\beta_*^2.
\]
On $0\le x\le1/2$, replacing $e^{-x^2/2}$ by $T(x)=1-x^2/2+x^4/8-x^6/48+x^8/384$
gives a degree-$20$ polynomial $P(x)$ whose Bernstein coefficients on the
five dyadic subintervals $[0,1/8]$, $[1/8,3/16]$, $[3/16,7/32]$, $[7/32,1/4]$,
and $[1/4,1/2]$ exceed $1/40$, $1/1000$, $1/30000$, $1/1000$, and $9/1000$,
respectively. For $x\ge1/2$, $|\phi''(x)|/\phi(0)<2/3$ and
$|\phi'''(x)|/\phi(0)<7/5$ give the lower bound
$(3-r/3)^2-\frac{21}{20}\beta_*r-\frac{45}{4}\beta_*^2=35773617637/140625000000>0$.
\end{proof}

\begin{lemma}[Balanced extreme support points for three-point summands]\label{lem:weighted-shape}
There exists $\kappa>0$ such that every sufficiently late nondegenerate
three-point summand $X_i$, with extreme support points $-a_i,b_i$ ($a_i,b_i>0$),
satisfies $\min(a_i,b_i)\ge\kappa H_i$. Consequently, for all $i\ge1$,
\begin{equation}\label{eq:weighted-shape-bound}
                         H_i\E|X_i|\le K v_i.
\end{equation}
\end{lemma}
\begin{proof}
For a three-point summand with support $x<y<t$ in $[-h,h]$, varying its raw
mean and second moment on the fixed support gives the second variation
\[
 (-3D-Cr_{\mu\mu})e^2+
   (\phi''(z)+6Ck+3Cr_\mu)ef+
   (-L_0+3Cr_v)f^2\le0,
\]
where $k=[u|u|]_{x,y,t}\in[-1,1]$ is the second divided difference of $u|u|$
and $r_{\mu},r_v,r_{\mu\mu}=O(h)$. As $h\to0$, nonpositivity of this
quadratic form requires $(3Ck+\phi''(z)/2)^2\le3DL_0$, which by
Lemma~\ref{lem:weighted-scalar} forces $|k|$ to stay bounded away from $1$.
Since $k=1-2a^2/[(a+y)(a+b)]$ for $y\ge0$ and $k=-1+2b^2/[(a+b)(b-y)]$ for
$y\le0$, bounding $|k|$ away from $1$ forces $\min(a,b)\ge\kappa H$, which
immediately implies $v_i\ge\frac12\min(a_i,b_i)\E|X_i|\ge\frac12\kappa H_i\E|X_i|$.
\end{proof}

\section{Third-moment growth under grouping and normal comparison}
\label{sec:weighted-grouping}

When disjoint summands are grouped into a partial sum, the sum law $S$ and
total variance are unchanged, while the sum of third absolute moments
increases by the \emph{grouping cost}.

\begin{lemma}[Variance and annular grouping bounds]\label{lem:weighted-grouping}
For any independent centered family $(X_i)$ with total variance $V$,
\begin{equation}\label{eq:weighted-variance-cost}
 0\le\E\left|\sum_i X_i\right|^3-\sum_i\E|X_i|^3\le2V^{3/2}.
\end{equation}
More precisely, if $X_i=Y_i-\E Y_i$ with $M=\sum_i\E|Y_i|$ and
$A_-=\sum_i\E(Y_i)_-$, then the grouping cost is at most $2M^3+6A_-V$.
Consequently, if the summands are partitioned into $B$ classes with first-moment
and variance sums $M_j,V_j$, then for every integer $J\ge1$ there exists a
partition into at most $8J+6B$ groups whose total grouping cost is at most
\begin{equation}\label{eq:weighted-annular-cost}
 \frac{18M^3}{J^2}+\frac{9S^2}{J},
 \qquad M=\sum_{j=1}^B M_j,\qquad S=\sum_{j=1}^B\sqrt{M_jV_j}.
\end{equation}
\end{lemma}
\begin{proof}
Inequality \eqref{eq:weighted-variance-cost} follows by summing the
incremental bound $\E|S_{<i}+X_i|^3-\E|S_{<i}|^3-\E|X_i|^3\le3v_i\sqrt{V_{<i}}$
(compare \citet[eq.~(5)]{Pinelis2013}). The refined bound $2M^3+6A_-V$
follows from the exact identity
$\E|X+Y|^3-\E|X|^3-\E|Y|^3=12\int_\R k_X(t)k_Y(-t)\,dt$, where
$k_X(t)=\E(X-t)_+-(-t)_+\ge0$ has $\int_\R k_X(t)\,dt=v_X/2$,
$\sup_{t\in\R}k_{X_i}(t)\le M_i$, and $k_{X_i}(-t)\le(M_i-t)_++A_i$ for
$t\ge0$. Greedy subdivision within each class $j$ with budget
$J_j=\lceil J(M_j/M+\sqrt{M_jV_j}/S)\rceil$ yields \eqref{eq:weighted-annular-cost}.
\end{proof}

\begin{lemma}[Quadratic majorant for a grouped block]\label{lem:weighted-majorant}
Let $U$ be the sum of a subcollection of summands of variance
$\sigma^2=\E U^2>0$ and maximum summand radius $H$, independent of the
remainder with distribution function $G$. There exists a quadratic polynomial
$P(u)=a+bu+cu^2$ such that
\begin{equation}\label{eq:weighted-majorant}
 Q_U(u):=P(u)+C|u|^3-G(z-u)\ge0\quad(u\in\R),\qquad
                        \E Q_U(U)\le2C\sigma^3.
\end{equation}
As $\sigma\to0$ with $H/\sigma\to0$, its coefficients satisfy
\begin{equation}\label{eq:weighted-coefficients}
 |a|+|b|+|c|\le K,\qquad b+\phi(z)=O(\sigma),\qquad c+A=O(\sigma).
\end{equation}
\end{lemma}
\begin{proof}
By strong moment duality at $(0,\sigma^2)$, the supporting plane $P(u)$ to
the concave value function of $\sup\{\E[G(z-Y)-C|Y|^3]:\E Y=m,\E Y^2=s\}$
majorizes $G(z-u)-C|u|^3$ and satisfies $\E Q_U(U)\le C(\E|U|^3-\beta_I)\le2C\sigma^3$
by \eqref{eq:weighted-variance-cost}. Testing two-sided bounded-density
perturbations $(1+t\psi)\Law(U)$ orthogonal to $1,U$ gives
\begin{equation}\label{eq:weighted-affine-deletion}
             \E|G(z-U)-F(z)+\phi(z)U|\le K\sigma^2
\end{equation}
and $\E|G(z-U)-C|U|^3-a_0-b_0U+AU^2|=O(\sigma^3)$ by $L^1$ quotient duality,
from which \eqref{eq:weighted-coefficients} follows.
\end{proof}

\begin{lemma}[Weighted normal comparison via Cram\'er moderate deviations]\label{lem:weighted-normal}
In Lemma~\ref{lem:weighted-majorant}, if $(H/\sigma)\log^{3/2}(1/\sigma)\to0$,
then for $Z\sim N(0,1)$,
\begin{equation}\label{eq:weighted-normal-slack}
                         \E Q_U(\sigma Z)\le K(\sigma^3+H).
\end{equation}
If $H/\sigma=O(\sigma^\zeta)$ for some fixed $\zeta>1$, then for every
$2<q<\min(\zeta+1,3)$ and fixed $T>0$,
\begin{equation}\label{eq:weighted-normal-window}
 \sup_{|t|\le T\sigma}
 \bigl|F(z+t)-F(z)-\phi(z)t+At^2\bigr|=O_{q,T}(\sigma^q).
\end{equation}
\end{lemma}
\begin{proof}
For a variance-one sum $W=U/\sigma$ of independent centered variables bounded
by $\epsilon=H/\sigma$, Cram\'er exponential tilting (see also
\citet[Theorem~2.1]{BeknazaryanSangXiao2019}) gives
\begin{equation}\label{eq:weighted-MD}
 |\Pp(W\le x)-\Phi(x)|+|\Pp(W<x)-\Phi(x)|
       \le K\epsilon(1+x^2)\phi(x)\qquad(|x|\le L).
\end{equation}
Integrating $Q_\sigma(x)=Q_U(\sigma x)\ge0$ by parts against $(1+x^2)\phi(x)$
on $[-L,L]$ ($L=\sqrt{20\log(1/\sigma)}$) via \eqref{eq:weighted-MD} proves
\eqref{eq:weighted-normal-slack}, and applying the same bound to
$Q_\sigma(x-t)$ gives
\begin{equation}\label{eq:weighted-shifted-slack}
 \sup_{|t|\le T}\E Q_U(U-\sigma t)
                         \le\sigma^{-o(1)}(\sigma^3+H).
\end{equation}
Multiplying the individual contact equations by $X_i$ and summing refines
$b+\phi(z)=O(\sigma^2\sqrt{\log(1/\sigma)})$, which together with
\eqref{eq:weighted-shifted-slack} proves \eqref{eq:weighted-normal-window}.
\end{proof}

\subsection{Randomized lattice phase identity and small-diameter Fourier bound}
\label{subsec:weighted-phase}

For $N\ge3$ independent groups rounded unbiasedly to a lattice of mesh
$\ell$ with uniform phases $u_1,\ldots,u_{N-1}$ and $u_N\equiv z-\sum_{i<N}u_i\pmod\ell$,
the phase-averaged rounded distribution function at $z$ satisfies the exact
identity
\begin{equation}\label{eq:weighted-phase-identity}
 \E_{\boldsymbol u}F_{L_{\boldsymbol u}}(z)
       =\E F(z-W_\ell),\qquad
 W_\ell=T_1+\cdots+T_{N-1}-U_\ell,
\end{equation}
where $T_i$ have independent triangular densities $\ell^{-1}(1-|x|/\ell)_+$
and $U_\ell\sim\operatorname{Unif}[0,\ell]$, so $\E W_\ell=-\ell/2$ and
$\E W_\ell^2=(N+1)\ell^2/6$, while the total rounding variance
$v_{\boldsymbol u}$ satisfies
\begin{equation}\label{eq:weighted-phase-variance}
 \E v_{\boldsymbol u}=\frac{N\ell^2}{6},\qquad
 \Var(v_{\boldsymbol u})\le \frac{N\ell^4}{64},\qquad
 |\operatorname{Cov}(v_{\boldsymbol u},F_{L_{\boldsymbol u}}(z))|
                                                    \le\frac{\sqrt N\ell^2}{16}
\end{equation}
(using Hoeffding's inequality \citep{Hoeffding1963} to control tails of
$W_\ell$). Consequently, we obtain:

\begin{lemma}[Phase-rounding exclusion at a normal window]\label{lem:weighted-window-phase}
Suppose \eqref{eq:weighted-normal-window} holds with $2<q<3$ along
$\sigma\downarrow0$, and the summands are partitioned into $N\to\infty$
groups with grouping cost $w\ge0$. Then maximality is violated whenever, with
$L=\log(1/\sigma)$,
\begin{equation}\label{eq:weighted-window-costs}
 N\sigma^{2q-2}L\longrightarrow0,\qquad
       \frac{w\sqrt{NL}}{\sigma}\longrightarrow0,\qquad wN^{2/3}\longrightarrow0.
\end{equation}
\end{lemma}

\begin{lemma}[Fourier exclusion of slow small-diameter variance decay]\label{lem:weighted-Fourier}
No maximizing representation can satisfy $V(r)\ge cr^p$ for all sufficiently
small $r>0$ with an exponent $p<2$.
\end{lemma}
\begin{proof}
If $V(r)\ge cr^p$ with $p<2$, then $w(t)=\exp\{-t^2V(1/(8t))/12\}$ has finite
moments of all orders on $[1,\infty)$. Rounding the first $k$ summands and
interpolating the tail's accompanying Poisson second-moment measure to
$\ell_k\Z$ ($\ell_k=\sqrt{\sum_{i>k}v_i^2/k}$) yields lattice approximations
$Z_k$ with $F_{Z_k}(x)=F(x)+\frac12\ell_k\phi(z)+o(\ell_k)$ at a lattice
point within $\ell_k/2$ of $z$ and variance and third-moment changes
$o(\ell_k)$, producing a positive objective gain $\frac12\phi(z)\ell_k+o(\ell_k)>0$.
\end{proof}

\section{Initial weighted variance moments ($W_{1/2}$ and $W_{3/4}$)}\label{sec:weighted-first}

\begin{lemma}\label{lem:weighted-subunit}
Under \eqref{eq:weighted-shape-bound}, both $W_{1/2}$ and $W_{3/4}$ are
finite.
\end{lemma}
\begin{proof}
By Tonelli's theorem,
\begin{equation}\label{eq:weighted-Tonelli}
 \sum_{H_i\le1}v_iH_i^{-s}
                 =V(1)+s\int_0^1 V(u)u^{-s-1}\,du.
\end{equation}
If $W_{1/2}=\infty$, then points with $V(u)\ge u^a$ ($a=13/24$) occur
arbitrarily close to $0$, while Lemma~\ref{lem:weighted-Fourier} supplies
$r\downarrow0$ with $V(r)\le r^p$ ($p=15/8$). Taking the last crossing
$h<r$ where $V(h)=h^a$ and $V(u)<u^a$ on $(h,r]$ and applying
Lemma~\ref{lem:weighted-window-phase} at $\sigma\sim h^{1/3}$, $q=23/8$, and
$t=h^{37/72}$ yields a contradiction. Once $W_{1/2}<\infty$, the annular
grouping criterion
\begin{equation}\label{eq:weighted-annular-window}
 M\sigma^{q-4/3}L^{2/3}\to0,\qquad
 S^2\sigma^{q-2}L\to0,\qquad
 \frac{V^3L}{\sigma^2}\to0,\qquad B\sigma^{2q-2}L\to0
\end{equation}
at $a=7/8$, $p=15/8$, $Q_0=15/7$, $q=29/14$, and $t=\sqrt h$ similarly
rules out $W_{3/4}=\infty$.
\end{proof}

\section{Compound-Poisson intensity bounds on dyadic shells}\label{sec:weighted-packets}

Set $d(u)=-Q(u)=F(z)-\phi(z)u-Au^2+C|u|^3-F(z-u)\ge0$, and for $m>3$ define
the monotone envelope
\begin{equation}\label{eq:weighted-envelope}
 E_m(h)=h^3+\sup_{r\ge h}\left(\frac hr\right)^m\sup_{|u|\le r}d(u).
\end{equation}

\begin{lemma}[Pairwise contact defect bound]\label{lem:weighted-pair}
Let two distinct summands with modal states translated to $0$ have laws
$K_i=(1-\theta_i)\delta_0+\eta_i$ ($\theta_i\le1/4$), diameters at most $Lh$,
and same-sign states $x=\varepsilon u$, $y=\varepsilon v$ with
$\kappa h\le u,v\le Lh$. Under Lemma~\ref{lem:weighted-scalar},
\[
 d(x)\le K\theta_1E_m(h),\qquad
 d(y)\le K\theta_2E_m(h),\qquad
 d(x+y)\le K(\theta_1+\theta_2)E_m(h).
\]
\end{lemma}
\begin{proof}
Inverting $K_i$ via its Neumann series and testing the $2\times2$ probability
Hessian on $(1+t,-1)$ with $t=c_\varepsilon(u-v)/(3D)$ gives
$uv(u-v)^2\le K(\theta_1+\theta_2)E_m(h)$ by Lemma~\ref{lem:weighted-scalar};
testing on $(1,-1)$ then bounds $d(x+y)$.
\end{proof}

\begin{lemma}[Shell nonmodal intensity bound]\label{lem:weighted-intensity}
Fix $\kappa,L>0$. For finitely many summands with modal state $0$, diameters
at most $Lh$, and modal deficits $\theta_i\le\epsilon$, let submeasures
$\nu_i$ of their nonmodal states in $\varepsilon[\kappa h,Lh]$ have total
mass $\lambda=\sum_i\nu_i(\R)\in(0,1]$. Then for small $h,\epsilon$,
\begin{equation}\label{eq:weighted-intensity}
 \epsilon E_m(h)\ge c\lambda h^4\min(\lambda,h),\qquad
 \lambda\le K\bigl\{\sqrt{\epsilon E_m(h)}\,h^{-2}
                              +\epsilon E_m(h)h^{-5}\bigr\}.
\end{equation}
\end{lemma}
\begin{proof}
With $\mu=\lambda^{-1}\sum_i\nu_i$, $a=\int x\,d\mu\ge\kappa h$,
$s=2c_\varepsilon a/(3D)$, and $\alpha=2-s$, adjoin one compound-Poisson-type
summand $\varepsilon(\sum_{j=1}^{N_q}Z_j-2qa)$ whose count $N_q$ has
probability generating function
\begin{equation}\label{eq:weighted-compound-pgf}
 B_q(t)=e^{\alpha q(t-1)}
                         \bigl[1+sq(t-1)+q(t-1)^2\bigr],
\end{equation}
while withdrawing fraction $\alpha q/\lambda$ of each $\nu_i$ to its mode $0$.
In the weighted convolution norm $\|\xi\|_m=\int(1+|u|/h)^m\,|\xi|(du)$, the
relative convolution operator $\mathcal P(q)=\prod_i(I-\alpha qA_i)e^{\alpha qD_\mu}[I+q(sD_\mu+D_\mu^2)]$
satisfies
\begin{equation}\label{eq:weighted-relative-error}
 \|\mathcal P(q)-I-q\mathcal P'(0)\|_m
                                      \le K\frac{\epsilon q^2}{\lambda}.
\end{equation}
Evaluating \eqref{eq:objective} at $q=c_0\min(\lambda,h)$ and using
Lemma~\ref{lem:weighted-pair} and $c_\varepsilon^2>3DL_0$ yields
\eqref{eq:weighted-intensity}.
\end{proof}

\section{Bootstrap iteration and proof of Theorem~\ref{thm:weighted-variance}}
\label{sec:weighted-iteration}

\begin{lemma}[Transfer from grouping rates to weighted variance moments]
\label{lem:weighted-rate-transfer}
Suppose that after deleting any finite set of summands, the remaining
summands can for every large $J$ be partitioned into at most $KJ$ groups with
grouping cost $w\le KJ^{-1-\theta}$ ($\theta>0$ fixed). Then
\[
 V(h)=o(h^p)\quad\text{and}\quad W_s<\infty
 \qquad\text{for all }0<s,p<\frac{1+2\theta}{1+\theta}.
\]
\end{lemma}
\begin{proof}
If $V(h)\ge ch^p$ along a sequence with $1<p<(1+2\theta)/(1+\theta)$, replace
a block of summands with $H_i\le h$ and variance $\sigma^2\asymp h^p$ by
$N(0,\sigma^2)$, group the remaining summands into $N\le KJ$ groups, and
apply \eqref{eq:weighted-phase-identity} with $\ell=h^\lambda$ and
$J=\lceil h^{-j}\rceil$ ($\frac{p(1+\theta)}{1+2\theta}<\lambda<1$,
$\frac{\lambda}{1+\theta}<j<2\lambda-p$) to obtain a positive objective gain
$\frac12\phi(z)\ell-o(\ell)>0$.
\end{proof}

\begin{lemma}[Initial polynomial grouping rate]\label{lem:weighted-initial-rate}
Under Lemma~\ref{lem:weighted-shape}, the hypothesis of
Lemma~\ref{lem:weighted-rate-transfer} holds with $\theta=1/80$, so
$W_s<\infty$ for all $s<82/81$ and $\sum_{i\ge1}\E|X_i|<\infty$.
\end{lemma}
\begin{proof}
Applying Lemma~\ref{lem:weighted-intensity} with $E_m(h)\le K$ to the shells
of $\{i:v_i\le\delta\}$ above $r_\delta=\delta^{1/20}$ and using $W_{3/4}<\infty$
below $r_\delta$ gives $\sum_{v_i\le\delta}v_i/\sqrt{H_i}\le K\delta^{1/80}$,
which via Lemma~\ref{lem:weighted-grouping} yields $\theta=1/80$.
\end{proof}

\begin{lemma}[Linear envelope bound on Variance-rich shells]\label{lem:weighted-rich-shell}
If $W_s<\infty$ for some $s>1$ and a shell $h<H_i\le2h$ contains a block of
variance $\sigma^2\ge ch^{2-\eta}$ ($\eta>0$), then $E_m(h)\le Kh$ for every
fixed $m>\max(3,2/\eta)$.
\end{lemma}
\begin{proof}
Applying Lemma~\ref{lem:weighted-normal} to the block gives $d(t)\le Kh$ for
$|t|\le\sigma/\sqrt{160\log(1/h)}$, while $(h/r)^md(u)=o(h)$ for larger $r$.
\end{proof}

\begin{lemma}[Sublevel sums for $v_i$ and $p_iH_i$]\label{lem:weighted-sublevels}
If $1<s<12/7$ and $W_s<\infty$, then for $q\in\{0,1/2\}$,
\begin{equation}\label{eq:weighted-variance-sublevels}
             \sum_{v_i\le\delta}v_iH_i^{-q}
                              \le K_s\delta^{(s-q)/6}.
\end{equation}
If all sufficiently late summands are two-point variables
$X_i=\varepsilon_iH_i(B_{p_i}-p_i)$ ($p_i\le1/2$) and $1<s<11/6$ with
$W_s<\infty$, then
\begin{equation}\label{eq:weighted-mean-sublevels}
             \sum_{p_iH_i\le\delta}p_iH_i
                              \le K_s\delta^{(s-1)/5}.
\end{equation}
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:weighted-rich-shell} and \eqref{eq:weighted-intensity}, any
shell with nonmodal intensity $\ge h^{-\eta}$ has $E_m(h)\le Kh$, which
forces $\delta\ge ch^6$ in the three-point case (and $\delta\ge ch^5$ in the
two-point case). Splitting at $r_1=L_1'\delta^{1/6}$ (or $L_1'\delta^{1/5}$)
and $r_0=L_0'\delta^{1/7}$ (or $L_0'\delta^{1/6}$) proves
\eqref{eq:weighted-variance-sublevels} and \eqref{eq:weighted-mean-sublevels}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:weighted-variance} and Corollary~\ref{cor:weighted-regularity}]
In the general three-point case, \eqref{eq:weighted-variance-sublevels} with
$\delta=J^{-6/(6-s)}$ gives grouping exponent $\theta_s=(2s-1)/(6-s)$, so
Lemma~\ref{lem:weighted-rate-transfer} maps any available $s\in(1,12/7)$ to
every $\gamma<T(s)=(3s+4)/(s+5)$. Iterating $T$ from $s>1$ (supplied by
Lemma~\ref{lem:weighted-initial-rate}) reaches every $\gamma$ below the positive
fixed point $\sqrt5-1$ of $T$. In the eventually two-point case, grouping by
the sign of the nonmodal displacement has $A_-=0$ in Lemma~\ref{lem:weighted-grouping},
so \eqref{eq:weighted-mean-sublevels} with $\delta=J^{-5/(6-s)}$ gives
$\theta_s^{\mathrm B}=(2s+3)/(6-s)$ and iteration map
$T_{\mathrm B}(s)=(3s+12)/(s+9)$, whose positive fixed point is
$\sqrt{21}-3$. Finally, applying \eqref{eq:weighted-phase-identity} with
$\ell=J^{-a}$ ($\frac{1+\alpha}{2\alpha}<a<1+\theta$) proves
Corollary~\ref{cor:weighted-regularity}.
\end{proof}
