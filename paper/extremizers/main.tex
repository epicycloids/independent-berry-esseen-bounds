\begin{abstract}
We study the variational structure of distributions maximizing the
Berry--Esseen ratio for independent centered real random variables. First,
at any fixed-$n$ or compactified maximizer, a second-order probability
variation preserving mass and mean proves that every summand is supported on
at most four points; at a global maximum, any four-point summand lies on a
flat probability segment whose endpoint replaces it by a three-point summand
with the same variance and third absolute moment. By the strict
Cox--Kemperman third-moment inequality $\E|U+W|^3>\E|U|^3+\E|W|^3$ for
nonconstant independent centered variables, every summand of an unrestricted
maximizer is convolution-indecomposable. Second, if the universal independent
Berry--Esseen constant $\ci$ exceeds Esseen's constant
$\ce=(3+\sqrt{10})/(6\sqrt{2\pi})$, the supremum is attained in a
L\'evy--Khintchine compactification consisting of countably many independent
summands plus an infinitely divisible remainder. For any three-point
maximizing representation, we prove that both the Gaussian and L\'evy jump
components vanish, the summand support diameters tend to zero, and any
infinitely active sum is nonatomic. Under the additional third-moment bound
$\beta\le18481/25000$ at unit variance, multi-scale grouping and randomized
lattice rounding establish weighted variance summability
$\sum_i v_i\diam(\supp X_i)^{-\gamma}<\infty$ for all $\gamma<\sqrt5-1$ (and
all $\gamma<\sqrt{21}-3$ when all but finitely many summands are two-point),
ruling out two-sided $C^{1,1}$ regularity of the distribution function at a
maximizing threshold as well as separated finite blocks and dense or
nonoverlapping rare Bernoulli tails.
\end{abstract}

\section{Introduction and main results}\label{sec:statements}

\subsection{Formulation of the problem}

For a centered real random variable $S$ of variance one, write
\[
 \Delta(S)=\sup_{x\in\R}|\Pp(S\le x)-\Phi(x)|,
\]
where $\Phi$ and $\phi$ denote the standard normal distribution function and
density. The universal Berry--Esseen constant for independent summands is
\[
 \ci=\sup\frac{\Delta(\sum_{i=1}^nX_i)}
                        {\sum_{i=1}^n\E|X_i|^3},
 \qquad
 \ce=\frac{3+\sqrt{10}}{6\sqrt{2\pi}}=0.4097321837\ldots,
\]
where the supremum is taken over all finite independent centered arrays
$(X_1,\ldots,X_n)$ with $\sum_{i=1}^n\E X_i^2=1$ and $\sum_{i=1}^n\E|X_i|^3<\infty$.
Esseen's two-point construction gives $\ci\ge\ce$ \citep{Esseen1956,Shevtsova2013}.

A companion paper \citep[Theorem~1.1]{BellSmall2026}, extending the method of
\citet{HeCheng2026} to non-identically distributed summands via the
common-diameter theorem of \citet{BellFinite2026}, establishes the sharp
bound in the small-Lyapunov regime: with $\ell_0=\exp(-\exp(97))$,
\begin{equation}\label{eq:small-input}
 \sum_{i=1}^n\E|X_i|^3\le\ell_0
 \quad\Longrightarrow\quad
 \Delta\!\left(\sum_{i=1}^nX_i\right)\le\ce\sum_{i=1}^n\E|X_i|^3.
\end{equation}
Consequently, any sequence of unit-variance arrays whose Berry--Esseen ratios
exceed $\ce$ must have third-moment sums bounded below by $\ell_0>0$. This
uniform lower bound prevents maximizing sequences from degenerating to a
pure Gaussian limit and allows us to study the extremal problem at positive
Lyapunov ratio via compactification and variational calculus.

\subsection{Compactification and structural theorems}

When passing to a limit of finite independent arrays $(X_{n,1},\ldots,X_{n,k_n})$
ordered by decreasing variance, individual summands whose variances remain
bounded away from zero converge weakly to countable summands $X_i$, while
vanishingly small summands can coalesce into an independent infinitely
divisible remainder $Y$ by the classical accompanying-Poisson limit theorem
\citep{Levy1937}.

\begin{definition}\label{def:representation}
An \emph{admissible representation} is a decomposition
\begin{align}
 S&=\sum_{i\ge1}X_i+Y,\notag\\
 \log\E e^{\ii tY}
   &=-\frac{at^2}{2}
     +\int_{\R\setminus\{0\}}(e^{\ii ty}-1-\ii ty)\,\nu(dy),
                                                        \label{eq:LK}\\
 \sum_{i\ge1}\Var(X_i)+a+\int_{\R\setminus\{0\}} y^2\,\nu(dy)&=1,\qquad
 \beta=\sum_{i\ge1}\E|X_i|^3+\int_{\R\setminus\{0\}}|y|^3\,\nu(dy)<\infty,         \label{eq:profile-moments}
\end{align}
where $(X_i)_{i\ge1}$ are independent centered random variables, $Y$ is a
centered infinitely divisible random variable independent of $(X_i)_{i\ge1}$,
$a\ge0$ is the Gaussian variance, and $\nu$ is a L\'evy measure on
$\R\setminus\{0\}$. We call $\beta$ the \emph{additive third-moment
functional} of the representation. An admissible representation is
\emph{maximizing} if $\beta>0$ and $\Delta(S)=\ci\beta$.
\end{definition}

Note that the Gaussian variance $a$ contributes to the unit-variance
constraint in \eqref{eq:profile-moments} but contributes zero to $\beta$,
because a normal variable $N(0,a)$ is a weak limit of sums of $N$ symmetric
two-point variables of total variance $a$ whose third-moment sums
$a^{3/2}/\sqrt N$ tend to zero (Lemma~\ref{lem:realization}). Thus $\beta$
depends on the representation $(X_i,a,\nu)$ rather than on the marginal law
of $S$ alone.

Our first main theorem establishes the existence of maximizing
representations when $\ci>\ce$, proves the unconditional four-point and
three-point support restrictions (stated for both finite-$n$ and compactified
maximizers in Theorem~\ref{thm:original-support} below), and eliminates both
the Gaussian and L\'evy jump components.

\begin{theorem}[Support reduction and elimination of infinitely divisible components]\label{thm:structure}
If $\ci>\ce$, there exists a maximizing admissible representation. Every
maximizing representation has at most four support points per discrete
summand $X_i$, and admits an equally maximizing representation with at most
three support points per summand that preserves each individual variance
$\Var(X_i)$, each third absolute moment $\E|X_i|^3$, and the remainder $Y$.
Moreover, every maximizing representation with at most three support points
per summand satisfies
\begin{equation}\label{eq:main-structure}
 a=0,\qquad \nu=0,\qquad
 \max\{|u|:u\in\supp X_i\}\longrightarrow0\quad(i\to\infty).
\end{equation}
If infinitely many $X_i$ are nonconstant, then the distribution of $S$ is
nonatomic and
\[
 \sum_{i\ge1}\bigl(1-\max_{u\in\R}\Pp(X_i=u)\bigr)=\infty.
\]
\end{theorem}

After reflecting $S\mapsto-S$ if necessary, a maximizing representation
attains its discrepancy on the positive closed branch at some threshold
$z\in\R$:
\begin{equation}\label{eq:max-data}
 F(z)-\Phi(z)=D=C\beta>0,
 \qquad C=\ci,\qquad A=\frac{3D+z\phi(z)}2,
 \qquad F(x)=\Pp(S\le x).
\end{equation}

Our second main theorem establishes weighted variance summability and
excludes smooth local expansions of $F$ at $z$ whenever $\beta$ satisfies
the upper bound $\beta\le18481/25000=0.73924$.

\begin{theorem}[Weighted variance summability]\label{thm:weighted-variance}
Suppose $\ci>\ce$, and let a maximizing representation with at most three
support points per summand satisfy $\beta\le18481/25000$. For each
nonconstant summand $X_i$, set $H_i=\diam(\supp X_i)$ and $v_i=\Var(X_i)$.
Then
\[
 \sum_{i\ge1}\frac{v_i}{H_i^\gamma}<\infty
       \quad(0<\gamma<\sqrt5-1),\qquad
                         \sum_{i\ge1}\E|X_i|<\infty.
\]
If all but finitely many nonconstant summands are supported on two points,
then $\sum_{i\ge1}v_iH_i^{-\gamma}<\infty$ holds for the larger range
$0<\gamma<\sqrt{21}-3$.
\end{theorem}

\begin{corollary}[Excluded local regularity at a maximizing threshold]
\label{cor:weighted-regularity}
Under the hypotheses of Theorem~\ref{thm:weighted-variance}, a maximizing
threshold $z$ satisfying \eqref{eq:max-data} cannot satisfy the two-sided
expansion
\[
             F(z+t)=F(z)+\phi(z)t+O(t^2)\qquad(t\to0).
\]
If all but finitely many nonconstant summands have two support points, $z$
cannot satisfy the one-sided left expansion
\[
       F(z)-F(z-h)=\phi(z)h+O(h^{1+\alpha})\qquad(h\downarrow0)
\]
for any $\alpha>(\sqrt{21}-3)/6$.
\end{corollary}

Our third main theorem restricts the combinatorial structure of support
configurations that can reach the maximizing threshold $z$. For any finite
index set $I$, put $T_I=S-\sum_{i\in I}X_i$ and define the \emph{threshold
completion set}
\begin{equation}\label{eq:completion}
 \mathcal A_I=
 \left\{(x_i)_{i\in I}\in\prod_{i\in I}\supp X_i:\
       z-\sum_{i\in I}x_i\in\supp T_I\right\},
\end{equation}
where $\supp T_I$ is the topological support of the independent remainder
$T_I$.

\begin{theorem}[Full affine rank of threshold completions]\label{thm:rank}
Fix a maximizing representation with at most three support points per
summand as in Theorem~\ref{thm:structure}, and a threshold $z$ satisfying
\eqref{eq:max-data}. There exists $k_0\ge1$ such that for every finite subset
$I\subset\{k_0,k_0+1,\ldots\}$, if functions $f_i:\supp X_i\to\R$ ($i\in I$)
satisfy
\[
             \sum_{i\in I}f_i(x_i)=c
             \qquad\text{for all }(x_i)_{i\in I}\in\mathcal A_I,
\]
then every $f_i$ is constant on $\supp X_i$. Equivalently, fixing a base point
$u_i^0\in\supp X_i$ for each $i\in I$, the indicator vectors
$(\ind_{\{x_i=u\}})_{i\in I,\,u\in\supp X_i\setminus\{u_i^0\}}$ for
$(x_i)_{i\in I}\in\mathcal A_I$ span an affine subspace of full dimension
$\sum_{i\in I}(|\supp X_i|-1)$.
\end{theorem}

\begin{corollary}[Exclusion of separated finite blocks]\label{cor:separated}
A maximizing representation as in Theorem~\ref{thm:rank} cannot consist of a
finite-support prefix followed by infinitely many nonconstant finite blocks
of finite-support summands whose block diameters and minimum nonzero block-sum
spacings
\[
 H_k=\sum_{i\text{ in block }k}\diam(\supp X_i),\qquad
 \delta_k=\min_{u\ne v\in\supp(\sum_{i\text{ in block }k}X_i)}|u-v|
\]
satisfy
\begin{equation}\label{eq:separation}
              \sum_k H_k<\infty,
              \qquad \delta_k\ge\sum_{l>k}H_l.
\end{equation}
\end{corollary}

In Section~\ref{sec:rare-tails} we further show that a maximizing
representation cannot contain an infinite sequence of rare two-point summands
$h_j(p_j-B_j)$ ($p_j\to0$) whose gap decrements satisfy $h_j-h_{j+1}=O(h_j^\nu)$
with $\nu>1+(\sqrt5-1)/2$ (Theorem~\ref{thm:dense-rare-gaps}) or whose gaps
are nonoverlapping in the sense that $\sum_{k>j}h_k\le h_j$ and
$h_{j+1}\ge ch_j$ (Theorem~\ref{thm:nonoverlapping-tail}).

\subsection{Open questions and overview of the proofs}

The theorems above place stringent constraints on any hypothetical violation
of $\ci=\ce$: a maximizing representation must be a pure discrete sum
$S=\sum_i X_i$ of indecomposable three-point (or two-point) summands with no
Gaussian or Poisson part, shrinking supports, nonatomic sum law when infinite,
full completion rank, and (for $\beta\le18481/25000$) finite weighted variance
moments $W_\gamma<\infty$ and rough distribution function at $z$. To close
the conjecture $\ci=\ce$ along this variational route, what remains in the
range $\ell_0<\beta<1/\ce$ is to rule out nonatomic infinite convolutions with
shrinking supports and finite representations (other than common-diameter
two-point arrays, which satisfy $\Delta(S)\le\ce\beta$ by \citet{BellFinite2026}),
including finite arrays with genuine three-point summands or with two-point
summands of unequal support diameters.

Let us outline the methods used across the paper:
\begin{itemize}[leftmargin=2em]
\item \textbf{Compactification and variational calculus
(Sections~\ref{sec:compactness}--\ref{sec:variation}):} Section~\ref{sec:compactness}
establishes compactness of finite arrays in the space of admissible
representations via accompanying Poisson laws \citep{Levy1937}.
Section~\ref{sec:variation} derives the first- and second-order variational
conditions for probabilities and L\'evy intensities, proves the unconditional
four-point and three-point support reductions (Theorem~\ref{thm:original-support})
by analyzing the rank of $(1,u,u^2,|u|^3)$ in the stationary second
logarithmic derivative, proves convolution indecomposability via the
Cox--Kemperman inequality \citep{CoxKemperman1983,Ruzankin2014}, and bounds
point masses of infinite sums via Sperner's theorem \citep{Sperner1928}.
\item \textbf{Elimination of L\'evy jumps, nonshrinking supports, Gaussian
components, and atoms (Sections~\ref{sec:propagation}--\ref{sec:nonatomic}):}
Section~\ref{sec:propagation} shows that any L\'evy support point or
nonshrinking sequence of summand supports $u_n\to x\ne0$ generates an
infinite arithmetic progression of contacts $Q(nx)=0$ ($n\ge1$) by testing
the $2\times2$ Hessian between the old probability/intensity direction and a
single adjoined three-point summand $(1-(1+\eps)t)\delta_0+t\delta_x+\eps t\delta_y$,
contradicting boundedness of the contact set. Section~\ref{sec:gaussian}
rules out $a>0$ by combining a Stein--Ornstein--Uhlenbeck zero-bias bound
($a<1/10$), a binomial Edgeworth lattice replacement ($\beta<\frac{17}{20}\sqrt a$),
and a Fourier inversion bound using the Korolev--Shevtsova compound-Poisson
estimate \citep{KorolevShevtsova2012}. Section~\ref{sec:nonatomic} rules out
atoms in infinitely active sums by coupling a single pivotal mass transfer
with a collective product-law variation across the tail.
\item \textbf{Completion rank, weighted variance bootstrap, and rare tails
(Sections~\ref{sec:rank}--\ref{sec:rare-tails}):} Section~\ref{sec:rank}
proves Theorem~\ref{thm:rank} by combining support translations that hold
$\mathcal A_I$ fixed with probability tilts. Sections~\ref{sec:weighted-shape}--\ref{sec:weighted-iteration}
prove Theorem~\ref{thm:weighted-variance} via a bootstrap iteration: under
$\beta\le18481/25000$, scalar curvature margins force every late three-point
summand to have balanced extreme points ($H_i\E|X_i|\le Kv_i$). Combining
Cram\'er moderate deviations for quadratic majorants, randomized lattice
phase rounding, and compound-Poisson intensity variations converts any
available weighted variance moment $W_s<\infty$ ($s>1$) into $W_\gamma<\infty$
for all $\gamma<T(s)=(3s+4)/(s+5)$ (whose fixed point is $\sqrt5-1$), or
$\gamma<T_{\mathrm B}(s)=(3s+12)/(s+9)$ (whose fixed point is $\sqrt{21}-3$)
for two-point tails. Section~\ref{sec:rare-tails} applies Neumann-series
inversion of $(qI+pT_h)$ to exclude dense and nonoverlapping rare Bernoulli
tails.
\end{itemize}

\section{Compactification and attainment above $\ce$}\label{sec:compactness}

\begin{lemma}[Moment-face three-point replacement]\label{lem:threepoint}
For every $c>0$, any finite independent unit-variance array satisfying
$\Delta(\sum_{i=1}^nX_i)>c\sum_{i=1}^n\E|X_i|^3$ can be replaced by one with
the same number of summands and the same individual variances, each supported
on at most three points, satisfying the same strict inequality.
\end{lemma}
\begin{proof}
Fix the attaining threshold $z$ on the positive closed branch (after
reflection if needed). Holding all summands except $X_i$ fixed, maximize
$\int\{G_i(z-u)-c|u|^3\}\,P(du)$ over probability measures $P$ with
$\int u\,dP=0$ and $\int u^2\,dP=v_i$. By the extreme-point principle for
three-constraint moment sets \citep{Winkler1988,BellFinite2026}, the maximum
is attained at an extreme point supported on at most three atoms; replacing
each summand successively preserves the strict inequality.
\end{proof}

\begin{lemma}[Sequential compactness in admissible representations]\label{lem:compactness}
Let $(X_{n,1},\ldots,X_{n,k_n})$ be finite independent centered arrays with
total variance one and third-moment sums $\beta_n\le M$. After passing to a
subsequence, their sums converge weakly to an admissible representation $S$
satisfying
\[
                      \beta\le\liminf_{n\to\infty}\beta_n.
\]
If each $X_{n,i}$ is supported on at most three points, then so is each
limiting discrete summand $X_i$.
\end{lemma}
\begin{proof}
Pass to a subsequence along which $\beta_n$ converges to its liminf, and
order each row by decreasing variance $v_{n,1}\ge v_{n,2}\ge\cdots$ (padding
with zeros). Since $v_{n,k}\le1/k$ and $\E[X_{n,k}^2\ind_{\{|X_{n,k}|>L\}}]\le M/L$,
a diagonal subsequence gives $X_{n,k}\Rightarrow X_k$ with $\E X_k=0$,
$v_k=\E X_k^2=\lim_n v_{n,k}$, and $r_k=\E|X_k|^3\le b_k:=\lim_n\E|X_{n,k}|^3$.
Choosing $K_n\to\infty$ slowly enough that $\sum_{k\le K_n}X_{n,k}\Rightarrow\sum_{k\ge1}X_k$
and $\sum_{k\le K_n}|v_{n,k}-v_k|\to0$, define the quadratic tail measures
\[
 \Lambda_n(E)=\sum_{k>K_n}\E\bigl[X_{n,k}^2\ind_{\{X_{n,k}\in E\}}\bigr],
\]
whose total masses converge to $v_0=1-\sum_{k\ge1}v_k$ and whose tails outside
$[-L,L]$ are at most $M/L$. Along a further subsequence, $\Lambda_n\Rightarrow\Lambda$
with $\Lambda(\R)=v_0$. Because $|f_{n,k}(t)-1|\le v_{n,k}t^2/2$ and
$\max_{k>K_n}v_{n,k}\le1/(K_n+1)\to0$, telescoping gives
\begin{equation}\label{eq:poissonization}
 \left|\prod_{k>K_n}f_{n,k}(t)-\exp\Bigl\{\sum_{k>K_n}(f_{n,k}(t)-1)\Bigr\}\right|
                  \le\frac{t^4}{8}\sum_{k>K_n}v_{n,k}^2\longrightarrow0.
\end{equation}
Since $\sum_{k>K_n}(f_{n,k}(t)-1)=\int q_t(y)\,\Lambda_n(dy)$ with the bounded
continuous integrand $q_t(y)=(e^{\ii ty}-1-\ii ty)/y^2$ ($q_t(0)=-t^2/2$),
the tail sums converge weakly to $Y$ with $a=\Lambda(\{0\})$ and
$\nu(dy)=y^{-2}\Lambda(dy)$ on $\R\setminus\{0\}$. Lower semicontinuity of
$\int|y|\,\Lambda_n(dy)\le\beta_n-\sum_{k\le K}\E|X_{n,k}|^3$ as $n\to\infty$
followed by $K\to\infty$ yields $\beta\le\lim_n\beta_n$.
\end{proof}

\begin{lemma}[Approximation of admissible representations by finite arrays]\label{lem:realization}
Every admissible representation $S$ is a weak limit of finite independent
centered unit-variance arrays whose third-moment sums converge to $\beta$.
Consequently,
\begin{equation}\label{eq:extended-bound}
                         \Delta(S)\le\ci\beta.
\end{equation}
\end{lemma}
\begin{proof}
Truncate $(X_i)_{i\ge1}$ to a finite prefix, approximate $\nu$ by a finite
atomic measure $\sum_j\lambda_j\delta_{y_j}$, replace each Poisson atom
$\lambda_j\delta_{y_j}$ by $N$ independent variables $y_j(B-\lambda_j/N)$
($B\sim\Bern(\lambda_j/N)$), and replace the remaining variance $b\to a$ by
$N$ independent symmetric two-point variables of variance $b/N$ (whose
third-moment sum is $b^{3/2}/\sqrt N\to0$). Continuity of $\Phi$ implies
$\Delta(S_N)\to\Delta(S)$, giving \eqref{eq:extended-bound}.
\end{proof}

\begin{proposition}[Attainment of $\ci$ when $\ci>\ce$]\label{prop:attainment}
If $\ci>\ce$, there exists a maximizing admissible representation whose
discrete summands $X_i$ are each supported on at most three points.
\end{proposition}
\begin{proof}
Choose $c_n\uparrow\ci$ with $c_n>\ce$, and apply Lemma~\ref{lem:threepoint}
to obtain finite three-point unit-variance arrays with $\Delta(S_n)>c_n\beta_n$.
By \eqref{eq:small-input} and $\Delta(S_n)\le1$, we have
$\ell_0<\beta_n<1/c_n$. Passing to a subsequence with $\beta_n\to r\in[\ell_0,1/\ci]$
via Lemma~\ref{lem:compactness} gives a limiting representation $S$ with
$\ci r=\Delta(S)\le\ci\beta\le\ci r$, so $\beta=r>0$ and $\Delta(S)=\ci\beta$.
\end{proof}

\section{Variational identities, support size, and indecomposability}\label{sec:variation}

Fix a maximizing representation with $C=\ci>\ce$ and threshold $z$ satisfying
\eqref{eq:max-data}. For any admissible perturbation of the uncentered
summand laws with total mean $m$, central variance $B$, recentered third-moment
functional $R_*$, and uncentered sum distribution function $F_*$ at threshold
$w$, Lemma~\ref{lem:realization} implies
\begin{equation}\label{eq:objective}
       J=B^{3/2}\left\{F_*(w)-\Phi\!\left(\frac{w-m}{\sqrt B}\right)\right\}
                    -C R_*\le0,
\end{equation}
with $J=0$ at the maximizing representation and $w=z$.

\begin{lemma}[First-order contact inequalities]\label{lem:contacts}
Let $G_i$ be the distribution function of $S-X_i$, and put $v_i=\E X_i^2$,
$\rho_i=\E|X_i|^3$, and $M_i=\E[X_i|X_i|]$. Then for all $u,y\in\R$,
\begin{align}
 G_i(z-u)-F(z)+\phi(z)u+A(u^2-v_i)
   -C\{|u|^3-\rho_i-3M_i u\}&\le0,                    \label{eq:coord-contact}\\
 Q(y):=F(z-y)-F(z)+\phi(z)y+Ay^2-C|y|^3&\le0,       \label{eq:Q}
\end{align}
with equality in \eqref{eq:coord-contact} at every $u\in\supp X_i$ and
$Q(y)=0$ at every $y\in\supp\nu\setminus\{0\}$. In particular, all supports
$\supp X_i$ and $\supp\nu$ are uniformly bounded.
\end{lemma}
\begin{proof}
Replacing the raw law of $X_i$ by $(1-t)\Law(X_i)+t\delta_u$ at fixed raw
threshold $w=z$ (so the centered threshold is $w-m=z-tu$, while mean, variance,
and central third absolute moment have derivatives $u$, $u^2-v_i$, and
$|u|^3-\rho_i-3M_iu$) gives \eqref{eq:coord-contact} (compare
\citet[Lemma~4.3]{HeCheng2026}), with equality on $\supp X_i$ by upper
semicontinuity. Similarly, appending an independent compensated Poisson
variable $y(N_t-t)$ at the centered threshold $z-ty$ (where the raw CDF is
$\Pp(S+yN_t\le z)=\E F(z-yN_t)$ and variance and $\beta$ increase by $ty^2$
and $t|y|^3$) gives \eqref{eq:Q}, and varying $\nu$ by bounded signed
densities yields $Q(y)=0$ on $\supp\nu\setminus\{0\}$. Since $-C|u|^3$
dominates as $|u|\to\infty$, all supports are uniformly bounded.
\end{proof}

For a mass transfer from $v$ to $u$ within $X_i$, write $d=u-v$, $b=u^2-v^2$,
and $g=G_i(z-u)-G_i(z-v)$. The first and second derivatives of $\E|X_i-\E X_i|^3$
are
\begin{align}
 r'&=|u|^3-|v|^3-3d\E[X_i|X_i|],\notag\\
 r''&=-6d(u|u|-v|v|)+6d^2\E|X_i|.                  \label{eq:transfer-moments}
\end{align}
With $B_0=[3D+(3z+z^3)\phi(z)]/4$ and $E_0=(1+z^2/2)\phi(z)$, the second
derivatives of \eqref{eq:objective} with respect to mass transfers in
coordinates $i$ and $j$ are
\begin{align}
 H_{ii}&=3b_i g_i+B_0b_i^2+2E_0b_i d_i-3Dd_i^2-Cr_i'',\label{eq:prob-hessian}\\
 H_{ij}&=g_{ij}+\frac32(b_i g_j+b_j g_i)+B_0b_i b_j
      +E_0(b_i d_j+b_j d_i)+z\phi(z) d_i d_j\qquad(i\ne j),\notag
\end{align}
where $g_{ij}$ is the mixed second difference of the raw distribution function.

\subsection{Support size of every summand at a probability-local maximum}

The next theorem applies unconditionally to any finite-array or
admissible-representation maximizer (without requiring $\ci>\ce$).

\begin{theorem}[Four-point support bound and moment-preserving three-point replacement]
\label{thm:original-support}
Let $(X_i)$ be the independent centered summands of a finite array or of an
admissible representation with total variance $V>0$ and third-moment
functional $R>0$. Suppose that at a threshold $y$ with
$D=\Pp(S\le y)-\Phi(y/\sqrt V)>0$, the ratio $C=V^{3/2}D/R$ is locally
maximal with respect to all two-sided bounded-density probability
perturbations of each $X_i$ that preserve its mass and mean. Then every
summand $X_i$ is supported on at most four points.

Furthermore, if the array or representation is a global maximizer of the
full Berry--Esseen ratio (at a fixed finite number of summands $n$, or over
all admissible representations) and the selected branch attains the full
Kolmogorov distance, then there exists an equally maximizing representation
in which every summand is supported on at most three points and has the exact
same variance $\Var(X_i)$ and third absolute moment $\E|X_i|^3$ as before.
\end{theorem}

\begin{proof}
Fix a summand $X$ with law $P$, and let $\delta$ be a compactly supported
signed measure with bounded density relative to $P$ satisfying
$\int d\delta=\int u\,d\delta=0$. For small $|\eps|$, $P+\eps\delta$ is a
centered probability measure with variance change $\eps v_1=\eps\int u^2\,d\delta$,
third-moment change $\eps r_1=\eps\int|u|^3\,d\delta$, and affine raw CDF
$F_\eps(y)=\int\Pp(S-X\le y-u)\,(P+\eps\delta)(du)$. Stationarity $C'_0=0$
gives $D'_0/D=r_1/R-3v_1/(2V)$, and with $z=y/\sqrt V$ and
$q=z\phi(z)(3-z^2)$, the second logarithmic derivative of
$C_\eps=(V+\eps v_1)^{3/2}D_\eps/(R+\eps r_1)$ at $\eps=0$ is
\begin{equation}\label{eq:original-rank-hessian}
 \frac{C''_0}{C}=-\frac{15+q/D}{4}\left(\frac{v_1}{V}\right)^2
                       +\frac{3r_1v_1}{RV}\le0.
\end{equation}
Consequently, the subspace of pairs $(v_1,r_1)$ realizable by mean-zero,
mass-zero perturbations $\delta$ cannot be all of $\R^2$. By Rolle's theorem
applied twice to the second derivative $6|u|-2c$, the four functions
$1,u,u^2,|u|^3$ are linearly independent on any five distinct real points.
If $\supp P$ contained five points, signed combinations of the restrictions
of $P$ to small disjoint neighborhoods of those points would realize every
pair $(v_1,r_1)\in\R^2$, violating \eqref{eq:original-rank-hessian}. Thus
$|\supp P|\le4$.

If $P$ has four distinct atoms $u_1,u_2,u_3,u_4$ of positive masses $p_j$,
then $(1,u,u^2,|u|^3)$ has rank $3$ on $\{u_1,u_2,u_3,u_4\}$, so any nonzero
vector $(d_1,d_2,d_3,d_4)$ annihilating $1,u,u^2$ also annihilates $|u|^3$.
Moving along $p_j+td_j$ preserves mass, mean, variance, and $\E|X|^3$ while
making $F(y)$ affine in $t$; stationarity at $t=0$ forces $F(y)$ to be
constant along the segment up to an endpoint where one atom mass vanishes,
yielding a three-point summand with the exact same variance and third
absolute moment.
\end{proof}

\subsection{Convolution indecomposability of maximizing summands}

By the $p=3$ case of the inequality of \citet[Theorem~(2.6)]{CoxKemperman1983}
(see also \citet[Theorem~E and Sections~2.1--2.2]{Ruzankin2014}), third
absolute moments are strictly superadditive under convolution of nonconstant
independent centered random variables:

\begin{lemma}[Strict third-moment superadditivity]\label{lem:strict-cubic-splitting}
For any independent centered real random variables $U,W$ with finite third
absolute moments,
\[
                 \E|U+W|^3\ge\E|U|^3+\E|W|^3,
\]
with equality if and only if at least one of $U,W$ is identically zero.
\end{lemma}
\begin{proof}
For centered two-point variables on $\{-a,b\}$ and $\{-c,d\}$ ($a,b,c,d>0$),
multiplying $\E|U+W|^3-\E|U|^3-\E|W|^3$ by $(a+b)(c+d)$ gives a function
$N(a,b,c,d)$ that vanishes when any argument is $0$ and satisfies
$\partial_a\partial_b\partial_c\partial_d N=12\{\min(a,d)+\min(b,c)\}>0$ on
$(0,\infty)^4$, so $N>0$. Decomposing any nonconstant centered distribution
into a mixture of $\delta_0$ and centered two-point laws via
$M(da,db)=(a+b)\mu_-(da)\mu_+(db)/\E U_+$ proves strict inequality in
general.
\end{proof}

\begin{corollary}[Convolution indecomposability]\label{cor:original-indecomposable}
Every nonconstant summand $X_i$ of a global maximizer over admissible
representations (or of a fixed-$n$ global maximizer having at least one zero
summand) is indecomposable as a convolution $U+W$ of two nonconstant
independent real random variables. In particular, no summand can be a
three-point distribution supported on an arithmetic progression $\{0,g,2g\}$
($g>0$) with atom probabilities $(a,b,c)$ satisfying $b^2\ge4ac$.
\end{corollary}
\begin{proof}
Replacing a decomposable summand $X_i\stackrel d=U+W$ by the two independent
centered summands $U,W$ leaves the sum $S$ and total variance unchanged while
strictly reducing the third-moment sum $R$ by Lemma~\ref{lem:strict-cubic-splitting}.
A three-point law on $\{0,g,2g\}$ with weights $(a,b,c)$ factors as the
convolution of two Bernoulli laws on $\{0,g\}$ if and only if
$t^2-(b+2c)t+c=0$ has real roots in $(0,1)$, which is equivalent to
$b^2\ge4ac$.
\end{proof}

\subsection{Point masses of infinite sums via Sperner's theorem}

\begin{lemma}\label{lem:atoms-modal}
If $S=\sum_{i\ge1}X_i+Y$ has an atom of mass $m>0$ and each $X_i$ is
supported on at most three points, then with $q_i=\max_{u\in\R}\Pp(X_i=u)$,
\[
                         \sum_{i\ge1}(1-q_i)<\infty.
\]
\end{lemma}
\begin{proof}
Each three-point law $X_i$ can be written as a mixture in which, with
probability $\rho_i=\min\{1,2(1-q_i)\}\ge\frac32(1-q_i)$, a fair symmetric
two-point step is activated. Conditional on the number $N$ of activated fair
steps and their step sizes, every level set of the sum is an antichain in
$\{0,1\}^N$, so Sperner's theorem \citep{Sperner1928} bounds its conditional
probability by $2^{-N}\binom{N}{\lfloor N/2\rfloor}\le\sqrt{2/(N+1)}$;
averaging over $N$ gives $m^2\le2\E[(N+1)^{-1}]\le2(1-e^{-\mu})/\mu$ where
$\mu=\sum_i\rho_i$ (compare \citet{JessenWintner1935}).
\end{proof}

\section{Eliminating L\'evy jumps and nonshrinking supports}\label{sec:propagation}

\subsection{Pair contacts and finiteness of the L\'evy support}

\begin{lemma}\label{lem:levy-pairs}
For any two same-sign points $x,y\in\supp\nu\setminus\{0\}$, one has
$Q(x+y)=0$. Moreover, $\supp\nu\setminus\{0\}$ contains at most two positive
and two negative points.
\end{lemma}
\begin{proof}
Choose a probability law $H$ dominated by a finite multiple of $\nu$ and
supported on a compact interval on one side of $0$, with $m=\E H$,
$v=\E H^2$, $r=\E|H|^3$, and $k=m^2/v$. Replacing a compensated Poisson
component of intensity $p-kp^2$ and jump law $H$ by a single centered variable
$B_pH-pm$ at centered threshold $z-kp^2m$ preserves total variance to second
order in $p$ and expands the objective excess at $O(p^2)$ as
$(k+1)\E Q(H)-\frac12\E Q(H+H')\le0$; since $Q=0$ on $\supp\nu\setminus\{0\}$,
upper semicontinuity forces $Q(x+y)=0$ for all same-sign
$x,y\in\supp\nu\setminus\{0\}$.

Perturbing $\nu$ by a finite signed measure $\eta=f\nu$ with bounded $f$
supported away from $0$ on one side $\sigma\in\{-1,1\}$, and setting
$d_\eta=\int y\,d\eta$, $v_\eta=\int y^2\,d\eta$, $r_\eta=\int|y|^3\,d\eta$,
and $U_\eta g(w)=\int[g(w-y)-g(w)]\,\eta(dy)$, the centered threshold
$z-d_\eta$ gives
\begin{equation}\label{eq:intensity-objective}
 (1+v_\eta)^{3/2}
 \left\{[e^{U_\eta}F](z)
       -\Phi\!\left(\frac{z-d_\eta}{\sqrt{1+v_\eta}}\right)\right\}
        -C(\beta+r_\eta)\le0.
\end{equation}
At same-sign contacts $x=\sigma u,y=\sigma v$ whose sum $x+y$ is also a
contact of $Q$, the second-order intensity kernel is
\begin{equation}\label{eq:kernel}
 \mathcal K_\sigma(u,v)=uv\left\{-3D+
       \left(3C+\frac{\sigma\phi''(z)}2\right)(u+v)
       -Luv+\frac{3C}{2}uv(u+v)\right\},\qquad u,v>0,
\end{equation}
where $L=(15D+\phi'''(z))/4$. In the moment basis $(u,u^2,u^3)$, the
coefficient matrix of $\mathcal K_\sigma$ is
\begin{equation}\label{eq:moment-matrix}
 M_\sigma=
 \begin{pmatrix}
 -3D&3C+\sigma\phi''(z)/2&0\\
 3C+\sigma\phi''(z)/2&-L&3C/2\\
 0&3C/2&0
 \end{pmatrix},
\end{equation}
which is not negative semidefinite since $(0,t,1)M_\sigma(0,t,1)^{\mathsf T}=-Lt^2+3Ct>0$
for small $t>0$. By Vandermonde invertibility of $(u_j,u_j^2,u_j^3)_{j=1}^3$,
$\nu$ cannot have three distinct support points of the same sign.
\end{proof}

\subsection{Contact propagation and proof of Proposition~\ref{prop:shrink}}

\begin{lemma}\label{lem:limit-double}
If $i_n$ are distinct indices and $u_n\in\supp X_{i_n}$ satisfies
$u_n\to x\ne0$, then $Q(x)=Q(2x)=0$. Moreover, the mass transfer can be taken
from a state $v_n\to0$ to $u_n$ with gap $d_n=u_n-v_n\to x$, and if $F$ has
atoms and $d_n>x$ along a subsequence, then $F$ is continuous at $z-x$ and
$z-2x$.
\end{lemma}
\begin{proof}
Since $v_{i_n}\to0$, we may choose a state $v_n\in\supp X_{i_n}$ with
$v_n\to0$ (taking $v_n$ to be a modal state with probability $q_n\to1$ via
Lemma~\ref{lem:atoms-modal} if $F$ has atoms, so that the modal-branch CDF
$H_n(w)=\Pp(S-X_{i_n}+v_n\le w)$ satisfies $\|H_n-F\|_\infty\le1-q_n\to0$,
and similarly for double deletions). Passing to a subsequence on which
$d_n=u_n-v_n$ equals $x$, lies above $x$, or lies below $x$, each shifted CDF
has a definite side limit; let $Q_1^*,Q_2^*$ denote the corresponding versions
of $Q(x),Q(2x)$ formed with these side limits. First-order stationarity gives
$Q_1^*=0$, and the two-coordinate Hessian contrast $(1,-1)$ in
\eqref{eq:prob-hessian} gives $-2Q_2^*\le0$. Because closed CDF values
dominate their left limits and $Q\le0$, we have $0=Q_1^*\le Q(x)\le0$ and
$0\le Q_2^*\le Q(2x)\le0$, so $Q(x)=Q(2x)=0$. Moreover, when $d_n>x$ along a
subsequence, the arguments $z-d_n$ and $z-d_n-d_m$ approach $z-x$ and $z-2x$
from the left, so equality $Q_1^*=Q(x)$ and $Q_2^*=Q(2x)$ forces the left
limits of $F$ at $z-x$ and $z-2x$ to equal $F(z-x)$ and $F(z-2x)$.
\end{proof}

\begin{lemma}\label{lem:closed-copy}
Let $x=\sigma u\ne0$ be either a limit of support points $u_n\in\supp X_{i_n}$
or an atom of $\nu$, and let $y=\sigma v\ne0$ be any fixed same-sign contact
point with $Q(y)=0$. Then the limiting mixed Hessian entry between the old
two-sided direction at $x$ and an adjoined summand $(1-t)\delta_0+t\delta_y$
($t\ge0$) equals $\mathcal K_\sigma(u,v)+Q(x+y)$, while both the old diagonal
entry and its mixed entry with an adjoined summand at $x$ equal
$\mathcal K_\sigma(u,u)$.
\end{lemma}
\begin{proof}
Comparing the old direction at $x$, an adjoined summand at $x$, and an
adjoined summand at $y$ along the tangent direction $(-1,1,0)+t(0,0,1)$ ($t\ge0$),
where $(-1,1,0)$ has zero quadratic value, forces the old--adjoined-$y$ mixed
entry to be at least---and by upper semicontinuity equal to---the closed
adjoined-$x$--adjoined-$y$ mixed entry $\mathcal K_\sigma(u,v)+Q(x+y)$.
\end{proof}

\begin{proposition}\label{prop:shrink}
Every maximizing representation with three-point discrete summands satisfies
$\nu=0$ and $\max\{|u|:u\in\supp X_i\}\to0$ as $i\to\infty$.
\end{proposition}
\begin{proof}
Suppose $x=\sigma u\ne0$ is either an atom of $\nu$ or a limit of
$u_n\in\supp X_{i_n}$, and let $y=\sigma v$ be any same-sign contact with
$Q(y)=0$. For fixed $\eps>0$, adjoin a single three-point summand
$(1-(1+\eps)t)\delta_0+t\delta_x+\eps t\delta_y$ ($t\ge0$), which has first
derivative $Q(x)+\eps Q(y)=0$. Because both $x$ and $y$ lie in the same
adjoined summand, its raw CDF is affine in $t$ (with no double-shift term on
its diagonal), so its diagonal Hessian entry is
$\mathcal K_\sigma(u,u)+2\eps\mathcal K_\sigma(u,v)+\eps^2\mathcal K_\sigma(v,v)$,
while by Lemma~\ref{lem:closed-copy} its mixed entry with the old two-sided
direction at $x$ is $\mathcal K_\sigma(u,u)+\eps[\mathcal K_\sigma(u,v)+Q(x+y)]$.
Testing the $2\times2$ Hessian on $(-1,1)$ gives
$-2\eps Q(x+y)+\eps^2\mathcal K_\sigma(v,v)\le0$; sending $\eps\downarrow0$
forces $Q(x+y)\ge0$ and hence $Q(x+y)=0$. By induction on $n\ge1$ with
$y=nx$, we obtain $Q(nx)=0$ for all $n\ge1$, contradicting the boundedness of
the zero set of $Q$ in Lemma~\ref{lem:contacts}.
\end{proof}

\section{Eliminating the Gaussian component}\label{sec:gaussian}

By Proposition~\ref{prop:shrink}, $S=W+G_a$ where $W=\sum_i X_i$ has variance
$1-a$, $\beta=\sum_i\E|X_i|^3$, and $G_a\sim N(0,a)$ is independent. If
$a>0$, the smooth density $f$ of $S$ satisfies threshold and Gaussian-variance
stationarity:
\begin{equation}\label{eq:gauss-stationarity}
 f(z)=\phi(z),\qquad f'(z)=-3D-z\phi(z),\qquad |f''(z)|\le6C.
\end{equation}

\subsection{Zero-bias Ornstein--Uhlenbeck bound and lattice comparison}

For a centered random variable $T$ of variance $v>0$, its zero-biased
distribution $T^*$ is characterized by $\E[Tg(T)]=v\E g'(T^*)$, which together
with the independent-sum replacement construction \citep{GoldsteinReinert1997}
and the single-variable and additive transport bounds of
\citet[Lemma~1.1]{Goldstein2010} and \citet[Theorem~3 and Corollary~1]{Tyurin2009}
gives
\begin{equation}\label{eq:zero-bias}
                  W_1(W,W^*)\le\frac{\beta}{2(1-a)}.
\end{equation}
Solving the Stein equation $vg'(u)-ug(u)=h(u)-\E h(\sqrt v\,Z)$ for
$h(u)=\Phi((z-u)/\sqrt a)$ and $v=1-a$ gives:

\begin{lemma}\label{lem:gauss-large}
For every $0<a<1$,
\[
 \Delta(W+G_a)\le K(a)\beta,\qquad
 K(a)=\frac{\phz}{2}\int_0^1
            \frac{s^2}{[1-(1-a)s^2]^{3/2}}\,ds.
\]
Since $K(a)$ is decreasing and $K(1/10)<\ce$, any maximizing representation
satisfies $a<1/10$.
\end{lemma}

\begin{lemma}\label{lem:lattice-gauss}
At a maximizing representation with $a>0$,
\begin{equation}\label{eq:lattice-curvature}
             \phi(z)\le a\left(C-\frac{f''(z)^2}{36C}\right).
\end{equation}
\end{lemma}
\begin{proof}
Replacing the tail $\sum_{i>K}X_i$ by its accompanying Poisson law and
rounding the summands and jumps to a fine lattice $h_j\Z$ produces admissible
approximations $T_j+G_{a_j}$ ($a_j\to a$) with $T_j\in\theta_j+h_j\Z$ and
\begin{equation}\label{eq:lattice-approx}
 \beta_j=\beta+o(h_j),\qquad
 \|F_{T_j+G_{a_j}}-F\|_\infty=o(h_j),\qquad
 f_j^{(k)}\to f^{(k)}\text{ uniformly}\quad(k=0,1,2).
\end{equation}
Replacing $G_{a_j}$ by a centered binomial $B_j=h_j(\Bin(n_j,p_j)-n_jp_j)$ of
variance $a_j$ with $p_j\to p\in(0,1)$ ($u_j=1-2p_j$) and applying Esseen's
lattice Edgeworth expansion \citep[eq.~(3)]{DinevMattner2013}
\begin{equation}\label{eq:binomial-expansion}
 \Pp(B_j\le x)=\Phi_{a_j}(x)+\frac{h_j}{2}\phi_{a_j}(x)
              -\frac{a_jh_ju_j}{6}\phi_{a_j}''(x)+o(h_j)
\end{equation}
at a lattice point $z_j$ within $h_j/2$ of $z$ yields
$\phi(z)-\frac13auf''(z)-Ca(1+u^2)\le0$ for all $u=1-2p\in[-1,1]$. Optimizing
at $u=-f''(z)/(6C)$ proves \eqref{eq:lattice-curvature}.
\end{proof}

Combining \eqref{eq:lattice-curvature} with Gaussian convolution
semiconvexity $f'(z)^2\le f(z)f''(z)+f(z)^2/a$ gives
$\beta<\frac{17}{20}\sqrt a$, which together with Lemma~\ref{lem:gauss-large}
confines $(a,\beta)$ to the window
\begin{equation}\label{eq:gauss-window}
         0<\beta<\frac{269}{1000},\qquad
         \frac{400}{289}\beta^2<a<\frac1{10}.
\end{equation}

\subsection{Fourier exclusion of the remaining Gaussian window}

By \citet[Section~3.3, Theorem~2]{KorolevShevtsova2012}, any centered
infinitely divisible law $Y$ with variance $v>0$ and third absolute jump
integral $r$ satisfies
\begin{equation}\label{eq:cp-bound}
                       d_{\mathrm K}(Y,N(0,v))\le0.3051\,\frac{r}{v^{3/2}}.
\end{equation}

\begin{lemma}\label{lem:gauss-window-exclusion}
Every unit-variance representation $S=W+G_a$ satisfying \eqref{eq:gauss-window}
obeys $\Delta(S)<(2/5)\beta<\ce\beta$.
\end{lemma}
\begin{proof}
Using the global inequality
\begin{equation}\label{eq:cosine}
                   \cos x\le1-\frac{x^2}{2}+\frac{|x|^3}{10}\qquad(x\in\R)
\end{equation}
and $\E|X_i-X_i'|^3\le4\E|X_i|^3$, each summand characteristic function
$h_i(t)$ satisfies
\begin{equation}\label{eq:cf-modulus}
 |h_i(t)|\le\exp\left\{-\frac{v_it^2}{2}+\frac{b_i|t|^3}{5}\right\},
          \qquad b_i=\E|X_i|^3,
\end{equation}
and integrating the zero-bias differential inequality gives
\begin{equation}\label{eq:cf-difference}
 |h(t)-e^{-t^2/2}|
 \le\frac\beta6e^{25/54}t^3e^{-t^2/2+\beta t^3/5}\qquad(t\ge0).
\end{equation}
Splitting the exact Fourier inversion integral for $S=W+G_a$ at
$T=c/\beta>7$ ($c=19/10$) bounds the low-frequency part $[0,T]$ by
\begin{equation}\label{eq:low-integral}
 \frac1{\pi\beta}\int_0^T\frac{|h(t)-e^{-t^2/2}|}{t}\,dt
       \le\frac{e^{25/54}}{6\pi}(I+J),
\end{equation}
with $I=\int_0^6t^2e^{-t^2/2+(269/5000)t^3}\,dt<23/10$ and
$J=\int_6^\infty t^2e^{-3t^2/25}\,dt<3/8$, while on $[T,\infty)$ domination of
$|h_W(t)|$ by its symmetrized accompanying compound-Poisson characteristic
function and \eqref{eq:cp-bound} bound the tail by
$(11/19)e^{-361/289}+10^{-9}<1/6+10^{-9}$. The rational checks in
Appendix~\ref{app:rational} give $\Delta(S)/\beta<1182125003/3000000000<2/5$,
proving $a=0$.
\end{proof}

\section{Nonatomicity of infinitely active maximizing sums}\label{sec:nonatomic}

Suppose $S=\sum_{i\ge1}X_i$ has an atom, and let $c_i$ be a modal state of
$X_i$ with probability $q_i=1-p_i\ge1/3$, so $\sum_i p_i<\infty$ by
Lemma~\ref{lem:atoms-modal} and $\sum_i|c_i|<\infty$. Writing $X_i=c_i+Y_i$,
the set of sums using at most $M$ nonzero $Y_j$ is compact for every fixed
$M\ge1$, and if all selected nonzero displacements are positive ($Z_j\ge0$),
their sums avoid $[b-\eta,b)$ for each $b>0$ and some $\eta>0$ (since a
sequence approaching $b$ from below would have fixed terms summing to $b$ and
escaping positive terms, making it eventually at least $b$). Moreover, if the
maximizing threshold $z$ is itself an atom of $S$ of mass $m>0$, then all
sufficiently late gaps $d_i=u_i-c_i$ are negative: for $d_i>0$ with
$d_i\to0$, the modal-branch CDF $H_i(w)=\Pp(S-X_i+c_i\le w)$ satisfies
$\|H_i-F\|_\infty\le p_i\to0$ and hence $H_i(z-d_i)-H_i(z)\to-m<0$,
contradicting mass-transfer stationarity $g_i+\phi(z)d_i+Ab_i-Cr_i'=0$.

If infinitely many $X_i$ were nonconstant, choose $K$ so that $q_j\ge3/4$ for
$j\ge K$ (and $d_j<0$ for $j\ge K$ when $z$ is an atom). For a late
nonconstant coordinate $i\ge K$ with nonmodal state $u_i=c_i+d_i$ and
$S_i^0=S-X_i+c_i$, its mass-transfer derivative is
\begin{equation}\label{eq:pivotal}
 g_i=-\operatorname{sgn}(d_i)\Pp(E_i),\qquad
 E_i=\begin{cases}
       \{z-d_i<S_i^0\le z\},&d_i>0,\\
       \{z<S_i^0\le z-d_i\},&d_i<0.
      \end{cases}
\end{equation}
Letting $N_i$ count the nonmodal outcomes among $j\ge K$, $j\ne i$, we have
for every fixed $M\ge1$:
\begin{equation}\label{eq:large-count}
                  E_i\subset\{N_i>M\}
                  \qquad\text{for all sufficiently large }i.
\end{equation}
Indeed, if $z$ is not an atom, $z$ has positive distance from the compact set
of configurations with at most $M+K$ nonmodal outcomes; if $z$ is an atom,
conditioning on any prefix configuration before $K$ writes
$S_i^0=b-\sum_{j\ge K,j\ne i}Z_j$ with $Z_j\ge0$, so $E_i$ becomes
$\sum_{j\ge K,j\ne i}Z_j\in[b-z-|d_i|,b-z)$, which is empty for $b-z\le0$ and
avoids $\{N_i\le M\}$ for small $|d_i|$ by the one-sided compactness property.
Meanwhile, stationarity of the mass transfer gives
\begin{equation}\label{eq:pivotal-size}
          \frac{g_i}{d_i}\longrightarrow-\phi(z),\qquad
          \frac{\Pp(E_i)}{|d_i|}\longrightarrow\phi(z)>0.
\end{equation}
Pairing the mass transfer $s$ in coordinate $i$ with the collective
product-law variation $t$ that multiplies all nonmodal probabilities for
$j\ge K,j\ne i$ by $1+t$ (compensated at the mode), the product-law score is
\[
 L_i=\sum_{j\ge K,\,j\ne i}\ell_j,\qquad
 \ell_j=\begin{cases}1,&Y_j\ne0,\\-p_j/q_j,&Y_j=0,\end{cases}\qquad
 L_i\ge N_i-\sum_{j\ge K}\frac{p_j}{q_j},
\]
with mixed raw-CDF derivative $g_{i,t}=-\operatorname{sgn}(d_i)\E[\ind_{E_i}L_i]$,
and the $2\times2$ Hessian of \eqref{eq:objective} satisfies
\begin{equation}\label{eq:atomic-H}
       |J_{ss}|\le K_1d_i^2,\qquad
       |J_{st}-g_{i,t}|\le K_2|d_i|,\qquad
       |J_{tt}|\le K_3,
\end{equation}
whereas $|g_{i,t}|/|d_i|=\E[L_i\mid E_i]\Pp(E_i)/|d_i|\to\infty$ by
\eqref{eq:large-count}--\eqref{eq:pivotal-size}, contradicting
$|J_{st}|^2\le J_{ss}J_{tt}\le K_1K_3d_i^2$. This completes the proof of
Theorem~\ref{thm:structure}.

\section{Support motions and completion rank}\label{sec:rank}

To prove Theorem~\ref{thm:rank}, suppose that for arbitrarily late finite
sets $I$ there exist nonconstant centered functions $f_i(X_i)$ ($\E f_i(X_i)=0$,
$w_i=\E f_i(X_i)^2$, $W=\sum_{i\in I}w_i>0$, $V=\sum_{i\in I}v_i\to0$, and
$h=\max_{i\in I}\max|\supp X_i|\to0$) with $\sum_{i\in I}f_i(x_i)=c$ on
$\mathcal A_I$. Moving each support state $x_i$ to $x_i+tf_i(x_i)$ and the
threshold to $z+tc$ leaves the raw event probability $F_*(z+tc)$ identically
constant for small $|t|$. Choosing $j\in I$ to maximize $w_j/v_j$, and setting
\[
 \begin{gathered}
 X=X_j,\qquad f=f_j(X_j),\qquad v=v_j,\qquad w=w_j,\\
 m=\E[Xf],\qquad a_1=\E[X^2f],\qquad b_1=\E[Xf^2],\qquad
 M=\sum_{i\in I}\E[X_if_i(X_i)],
 \end{gathered}
\]
we have
\begin{equation}\label{eq:rank-selection}
                         \frac{v}{w}\le \frac VW.
\end{equation}
Tilting $\Law(X)$ by $(1+sf)$ while applying the support motion $t$, the
centered threshold $x$, variance $B$, and third-moment functional $R$ have
derivatives at $(s,t)=(0,0)$ given by
\begin{align}
 x_s=-m,\quad x_t=c,\quad x_{st}=-w,&\qquad
 B_s=a_1,\quad B_{ss}=-2m^2,\notag\\
 B_t=2M,\quad B_{tt}=2W,&\qquad B_{st}=2b_1,           \label{eq:rank-moments}
\end{align}
and
\begin{align}
 R_t&=3\sum_{i\in I}\E[X_i|X_i|f_i],&
 R_{tt}&=6\sum_{i\in I}\E[|X_i|f_i^2],\notag\\
 R_s&=\E[|X|^3f]-3m\E[X|X|],&
 R_{ss}&=6m^2\E|X|-6m\E[X|X|f],\notag\\
 R_{st}&=3\E[X|X|f^2]-3w\E[X|X|]
                         -6m\E[|X|f].              \label{eq:rank-cubic}
\end{align}
Using \eqref{eq:rank-selection}--\eqref{eq:rank-cubic}, the $2\times2$
Hessian of \eqref{eq:objective} satisfies $|J_{ss}|\le K_1vw$,
$|J_{tt}|\le K_2W$, and $J_{st}=\phi(z)w+O((h+V)w)$, which violates
$J_{st}^2\le J_{ss}J_{tt}\le K_1K_2Vw^2$ as $h+V\to0$. This proves
Theorem~\ref{thm:rank}. Corollary~\ref{cor:separated} follows immediately by
noting that if \eqref{eq:separation} held, at most $2N$ block-sum vectors
among $M>2N$ late blocks could belong to $\mathcal A_I$, yielding a nontrivial
linear relation $\sum_{k=1}^Ma_ku_k=c$ on $\mathcal A_I$.
