\begin{abstract}
We establish sharp and strict Berry--Esseen bounds for small independent
arrays and for centered two-point (Bernoulli) summands. First, we prove that
the Berry--Esseen constant $C_2$ for at most two arbitrary independent
centered summands satisfies $C_2<49/120=0.408333\ldots<\be$, where
$\be=(3+\sqrt{10})/(6\sqrt{2\pi})\approx0.409732$ is Esseen's constant.
Second, for independent centered two-point summands, we prove the sharp
inequality $\Delta\le\be L$ whenever either (i) there are at most three
summands with arbitrary support diameters, or (ii) there are arbitrarily
many summands whose nondegenerate members share a common support diameter,
extending Schulz's binomial theorem to unequal probabilities. For at most
two two-point summands with positive total variance, we prove the uniform
strict margin $\be L-\Delta>3/80$. Combining these finite-array bounds with
a Stein--Ornstein--Uhlenbeck Gaussian comparison yields $\Delta\le\be L$
for any independent array in which two two-point summands carry at least
$22/23$ of the total variance, or two arbitrary summands carry at least
$439/440$ of the total variance. The proof for two arbitrary summands uses
strong quadratic moment duality to reduce each summand at a fixed threshold
and variance to a distribution supported on at most three points, and then
rules out all possible threshold configurations by variational contact
identities and certified interval enclosures.
\end{abstract}

\section{Introduction and main results}\label{sec:main}

\subsection{Context and problem formulation}

Let $X_1,\ldots,X_n$ be independent centered real random variables with
positive total variance and finite third absolute moments. Define the total
variance $V$, the sum of third absolute moments $R$, the Lyapunov ratio $L$,
and the Kolmogorov normal-approximation error $\Delta$ by
\begin{equation}\label{eq:normalization}
\begin{gathered}
 V=B^2=\sum_{i=1}^n\E X_i^2>0,\qquad
 R=\sum_{i=1}^n\E|X_i|^3<\infty,\qquad
 L=\ell=\frac{R}{V^{3/2}},\\
 \Delta=\sup_{x\in\R}\left|\Pp\left\{\frac{\sum_{i=1}^nX_i}{\sqrt V}\le x\right\}-\Phi(x)\right|,
\end{gathered}
\end{equation}
where $\Phi$ and $\phi$ denote the standard normal distribution function
and density. (We write $V$ and $B^2$, as well as $L$ and $\ell$,
interchangeably when convenient.) Set
\begin{equation}\label{eq:esseen-constant}
 \be=\frac{3+\sqrt{10}}{6\sqrt{2\pi}}=0.4097321837\ldots,\qquad
 \ke=\sqrt{2\pi}\be=\frac{3+\sqrt{10}}6,
 \qquad c_2=\frac{49}{120},\qquad t_*=\sqrt{10}-3.
\end{equation}
By \citet{Esseen1956}, no universal constant smaller than $\be$ can hold in
the inequality $\Delta\le C L$ across all finite independent arrays. Indeed,
for $p_{\mathrm E}=(4-\sqrt{10})/2$ and $q_{\mathrm E}=1-p_{\mathrm E}$, the
centered unit-variance two-point distribution
\begin{equation}\label{eq:esseen-law}
 \mu_{\mathrm E}=p_{\mathrm E}\delta_{\sqrt{q_{\mathrm E}/p_{\mathrm E}}}
 +q_{\mathrm E}\delta_{-\sqrt{p_{\mathrm E}/q_{\mathrm E}}}
\end{equation}
achieves $\lim_{n\to\infty}\Delta/L=\be$ for sums of $n$ independent copies.

While the unrestricted conjecture $\Delta\le\be L$ remains open for general
independent arrays, restricting either the number of summands $n$ or the
structure of the summand distributions allows one to establish $\Delta\le\be L$
(and, for small $n$, strictly smaller constants). For a single summand
($n=1$), \citet[Theorem~1]{BentkusKirsa1989} proved that the optimal constant
is $C_1\approx0.370352<\be$, achieved by a two-point distribution. For $n$
identically distributed two-point summands ($N\sim\Bin(n,p)$ with $0<p<1$
and $q=1-p$), \citet[Theorem~1]{Schulz2016} proved the sharp binomial bound
\begin{equation}\label{eq:schulz}
 \sup_{x\in\R}\left|\Pp\left\{\frac{N-np}{\sqrt{npq}}\le x\right\}-\Phi(x)\right|
 <\be\frac{p^2+q^2}{\sqrt{npq}}.
\end{equation}

\subsection{Main results}

Our first theorem establishes that for at most two arbitrary independent
summands, the Berry--Esseen constant $C_2$ is strictly below $49/120<\be$.

\begin{theorem}[Two arbitrary summands]\label{thm:arbitrary-two}
Let $C_2$ be the supremum of $\Delta/L$ over all independent centered arrays
with at most two summands. Then
\[
 C_2<\frac{49}{120}=0.408333\ldots<\be.
\]
\end{theorem}

Throughout this paper, a \emph{two-point} (or \emph{centered Bernoulli})
\emph{summand} means a centered random variable supported on at most two
points; its support diameter is the distance between its two support points
when nondegenerate. Deterministic zero summands are permitted. Our second
theorem establishes the sharp Esseen bound $\Delta\le\be L$ for two fundamental
classes of non-identically distributed two-point arrays.

\begin{theorem}[Two-point arrays]\label{thm:bernoulli-classes}
The sharp inequality $\Delta\le\be L$ holds whenever either
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item the array consists of at most three two-point summands with arbitrary
support diameters, or
\item every summand is supported on at most two points and all nondegenerate
summands share a common support diameter.
\end{enumerate}
In class \textup{(ii)}, the constant $\be$ is optimal.
\end{theorem}

For at most two two-point summands, we establish a uniform positive gap
between $\be L$ and $\Delta$ across all probabilities and support diameters.

\begin{theorem}[Uniform deficit for two two-point summands]\label{thm:two-summands}
For every independent array consisting of at most two two-point summands
with positive total variance $V>0$,
\[
 \be L-\Delta>\frac3{80}.
\]
\end{theorem}

Combining Theorems~\ref{thm:arbitrary-two} and~\ref{thm:two-summands} with
a Stein--Ornstein--Uhlenbeck normal comparison for the remaining summands
yields the sharp inequality $\Delta\le\be L$ for arbitrary arrays whose
variance is concentrated in two summands.

\begin{corollary}[Arrays with variance concentrated in two summands]\label{cor:dominance-main}
An arbitrary finite independent array satisfies $\Delta\le\be L$ whenever it
contains either two two-point summands with combined variance at least
$\frac{22}{23}V$, or two arbitrary summands with combined variance at least
$\frac{439}{440}V$.
\end{corollary}

\subsection{Open questions and overview of the proofs}

Two natural questions directly extend the results of this paper:
\begin{enumerate}[label=(\arabic*),leftmargin=2em]
\item \emph{Exact value of $C_2$:} For two two-point summands with arbitrary
diameters, a companion paper proves that the maximal ratio is $C_1\approx0.370352$,
attained when one summand is zero. Whether $C_2=C_1$ holds for two
\emph{arbitrary} independent summands remains open; Theorem~\ref{thm:arbitrary-two}
suffices to show $C_2<\be$.
\item \emph{Arbitrary diameters for $n\ge4$ two-point summands and $n\ge3$
general summands:} Extending Theorem~\ref{thm:bernoulli-classes}(i) beyond
three two-point summands with unequal diameters, and extending
Theorem~\ref{thm:arbitrary-two} to $n\ge3$ general summands, are key steps
toward the general conjecture $\Cind=\be$. Meanwhile,
Theorem~\ref{thm:bernoulli-classes}(ii) supplies the finite-array input used
in the companion small-Lyapunov manuscript to prove $\Delta\le\be L$ for all
independent arrays with $L\le\exp(-\exp(97))$.
\end{enumerate}

Let us summarize the methods used in each part of the paper. For any centered
sum $S$ of variance $V=B^2$ and $\varepsilon\in\{-1,1\}$, define the closed
and left-continuous distribution functions and the signed deficits by
\[
 F_S^+(y)=\Pp(S\le y),\qquad F_S^-(y)=\Pp(S<y),\qquad
 H_S^\varepsilon(y)=\be L+\varepsilon\Phi(y/B)-\varepsilon F_S^\varepsilon(y).
\]
By continuity of $\Phi$,
\begin{equation}\label{eq:one-sided}
 \Delta=\sup_{y\in\R}\max\{F_S^+(y)-\Phi(y/B),\ \Phi(y/B)-F_S^-(y)\},
\end{equation}
and reflection $S\mapsto-S$ gives $H_S^-(y)=H_{-S}^+(-y)$.

\begin{itemize}[leftmargin=2em]
\item \textbf{Two and three two-point summands with arbitrary diameters
(Sections~\ref{sec:two}--\ref{sec:three} and Appendix~\ref{app:support}):}
At the two extreme support endpoints of any $n$-summand two-point array,
Proposition~\ref{cor:all-support-owners} proves $H_S^\pm>0$ analytically by
optimizing over the support diameters via H\"older's inequality and bounding
the resulting function of the Bernoulli odds $o_i=p_i/q_i$ via two exact
Bernstein polynomial certificates over $\mathbb Q(\sqrt5)$. For $n=2$ and
$n=3$, the remaining interior atoms are parameterized on compact cubes and
certified by outward-rounded Arb interval partitions.
\item \textbf{Common-diameter two-point arrays (Section~\ref{sec:equal-gap-two-point}):}
To prove Theorem~\ref{thm:bernoulli-classes}(ii), we minimize the deficit
$\mathcal H_+(p;k)$ over the probability cube $[0,1]^n$ at a fixed mean $\mu$
and integer threshold $k$. Adapting \citet{Mattner2022}, we vary three
probabilities along a curve preserving their elementary symmetric polynomials
$e_1,e_2$; because both $F_k(p)$ and the third-moment sum $\mathcal B(p)$ are
affine in $e_3$, any extremal point has at most two distinct interior
probabilities $a<b$. A uniform bound on third differences of Poisson-binomial
point probabilities (Lemma~\ref{lem:poisson-binomial-third-difference}) rules
out the case where both $a$ and $b$ have multiplicity at least two, reducing
the problem to $\Bin(M,p)+\Bern(x)$, which is resolved by Schulz's theorem
(when $p=x$), a certified second-derivative curvature bound (when $M\ge2$ and
$p\ne x$), and Theorem~\ref{thm:two-summands} (when $M=1$).
\item \textbf{Two arbitrary summands via moment duality and contact
exclusions (Sections~\ref{sec:moments}--\ref{sec:arbitrary-two} and
Appendix~\ref{app:contacts}):}
At a fixed threshold $z$ and fixed individual variances $v_1,v_2$, maximizing
$\Pp(X_1+X_2\le z)-C\sum_{i=1}^2\E|X_i|^3$ in each marginal law is a linear
program over probability measures with three moment constraints ($1,u,u^2$).
By strong quadratic moment duality (\citet{Winkler1988,BertsimasPopescu2005};
Theorem~\ref{thm:strongdual}), if $C_2\ge49/120$, the supremum is attained by
two distributions $X_1,X_2$ each supported on at most three points, and each
support lies in the contact set where a quadratic majorant $q_i(u)$ touches
$F_{X_{3-i}}^+(z-u)-C|u|^3$. Classifying the possible downsets of
$\{(j,k):x_j+y_k\le z\}$ on the $3\times2$ and $3\times3$ support grids and
combining the contact equations with second-order support and mass variations
rules out every configuration analytically or by interval arithmetic.
\item \textbf{Variance-concentration transfer (Section~\ref{sec:transfer}):}
When two summands carry variance $v\in(0,1)$ and the remaining summands
carry variance $1-v$, convolving the two-summand Kolmogorov or $L^1$ bound
with the Stein--Ornstein--Uhlenbeck zero-bias bound for the remainder yields
Corollary~\ref{cor:dominance-main}.
\end{itemize}

\section{Two Bernoulli summands with arbitrary diameters}\label{sec:two}

For independent Bernoulli variables $B_i\sim\Bern(p_i)$ with $p_i\in(0,1)$
and $q_i=1-p_i$, write
\[
 X_i=h_i(B_i-p_i),\qquad a_i=p_iq_i,\qquad r_i=p_i^2+q_i^2,\qquad
 V=\sum_{i=1}^n a_i h_i^2=B^2,\qquad R=\sum_{i=1}^n a_ir_ih_i^3=B^3\ell.
\]
We begin by recording the lower bound on $L=R/V^{3/2}$ obtained by
optimizing over the support diameters $h_i$.

\begin{samepage}
\begin{lemma}[Diameter lower bound via H\"older's inequality]\label{lem:reciprocal-gap-resource}
Let $0<p_i<1$ and $h_i\geq0$ with $\sum_{i=1}^n a_i h_i^2>0$. Then
\begin{equation}\label{eq:reciprocal-gap-resource}
 \frac{\sum_{i=1}^n a_ir_ih_i^3}{(\sum_{i=1}^n a_ih_i^2)^{3/2}}
 \geq\left(\sum_{i=1}^n\frac{a_i}{r_i^2}\right)^{-1/2},
\end{equation}
with equality if and only if $r_i h_i$ is constant across all $i$.
\end{lemma}
\end{samepage}
\begin{proof}
By H\"older's inequality with exponents $3/2$ and $3$,
\[
 \sum_{i=1}^n a_i h_i^2
 =\sum_{i=1}^n(a_ir_ih_i^3)^{2/3}(a_i/r_i^2)^{1/3}
 \leq\left(\sum_{i=1}^n a_ir_ih_i^3\right)^{2/3}
      \left(\sum_{i=1}^n \frac{a_i}{r_i^2}\right)^{1/3},
\]
with equality if and only if $a_ir_ih_i^3$ is proportional to $a_i/r_i^2$.
\end{proof}

Our next result shows that at the two extreme support endpoints of
$S=\sum_{i=1}^n h_i(B_i-p_i)$, the sharp Esseen bound holds strictly for
every $n\ge1$ and all choices of $p_i\in(0,1)$ and $h_i>0$.

\begin{samepage}
\begin{proposition}[Strict positivity at support endpoints]\label{cor:all-support-owners}
For every finite two-point array with $0<p_i<1$ and $h_i>0$, both
$H_S^+(y)$ and $H_S^-(y)$ are strictly positive at the minimum and maximum
of the support of $S=\sum_{i=1}^n h_i(B_i-p_i)$.
\end{proposition}
\end{samepage}
The proof of Proposition~\ref{cor:all-support-owners} is given in
Appendix~\ref{app:support}: the two branches where the endpoint atom is
excluded follow from an elementary normal-tail bound, while the two branches
where the endpoint atom is included reduce via Lemma~\ref{lem:reciprocal-gap-resource}
to a sharp inequality for the Bernoulli odds $o_i=p_i/q_i$ established using
exact Bernstein polynomials over $\mathbb Q(\sqrt5)$.

For $n=2$ summands, the two interior atoms of $S$ are covered by the
following two compact inequalities.

\begin{samepage}
\begin{lemma}[Interior-atom inequalities for two two-point summands]\label{lem:two-compact}
Let $0<p_1,p_2<1$, $q_i=1-p_i$, $a_i=p_iq_i$, and $r_i=p_i^2+q_i^2$.
For $0\leq t\leq1$, put $V=a_1t^2+a_2$ and $R=a_1r_1t^3+a_2r_2$. Then
\begin{equation}\label{eq:two-upper-middle}
 \be\frac{R}{V^{3/2}}+
 \Phi\left(\frac{-p_1t+q_2}{\sqrt V}\right)-(1-p_1p_2)>\frac3{80}.
\end{equation}
Similarly, for $0\leq\theta\leq\pi/2$, set $\alpha=\cos\theta$ and
$\beta=\sin\theta$. Then
\begin{equation}\label{eq:two-lower-middle}
 \be\left(\alpha^3\frac{r_1}{\sqrt{a_1}}+
                  \beta^3\frac{r_2}{\sqrt{a_2}}\right)
 +\Phi\left(\alpha\sqrt{\frac{q_1}{p_1}}
            -\beta\sqrt{\frac{p_2}{q_2}}\right)-q_2>\frac3{80}.
\end{equation}
\end{lemma}
\end{samepage}
\begin{proof}[Method of verification]
Inequalities \eqref{eq:two-upper-middle} and \eqref{eq:two-lower-middle}
(with $\theta=\pi t/2$), together with the endpoint square $[0,99/100]^2$ in
Lemma~\ref{lem:pair-endpoint-margin} below, are verified on their closed unit
domains by a 63,792-leaf partition evaluated in 320-bit Arb ball arithmetic
(with minimum certified excess $>2.5361\times10^{-7}$ over $3/80$); the
monotonicity bounds used near the singular boundary faces $p_i\in\{0,1\}$ are
recorded in Section~\ref{sec:verification}.
\end{proof}

\begin{lemma}[Uniform endpoint deficit for at most two summands]\label{lem:pair-endpoint-margin}
At either support endpoint of a one- or two-summand two-point array with
$V>0$, both $H_S^+(y)$ and $H_S^-(y)$ exceed $3/80$.
\end{lemma}
\begin{proof}
At the lower support endpoint, set $A=\sum_i a_i/r_i^2$, $s=\sum_i p_i/q_i$,
and $\Pi=\prod_iq_i$. Normalizing $V=1$, Lemma~\ref{lem:reciprocal-gap-resource}
and Cauchy--Schwarz give $R\ge A^{-1/2}$ and $m=\sum_i p_i h_i\le\sqrt s$, so
the closed deficit is bounded below by
\[
 \frac{\be}{\sqrt A}+\Phi(-\sqrt s)-\Pi.
\]
On $[0,99/100]^2$, the interval partition in the supplement certifies that
this lower bound exceeds $3/80$. If some $p_i\ge99/100$, then $\Pi\le1/100$
while $R\ge1/\sqrt2>2/3$, giving a deficit exceeding $49/180-1/100>3/80$.
For the strict branch at the lower endpoint, where $F_S^-=0$, the proof of
Lemma~\ref{lem:support-owners-without-atom-term} gives $R\ge c/x$ with
$c=2(\sqrt2-1)>49/60$ and $x=m$. Since $x\Phi(-x)<1/4$ for all $x>0$, the
strict deficit exceeds
\[
 \left(\be-\frac1{4c}\right)R>\left(\frac{49}{120}-\frac{15}{49}\right)\frac23
 =\frac{601}{8820}>\frac3{80}.
\]
Reflection $S\mapsto-S$ establishes the same bounds at the upper endpoint.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:two-summands}]
Omit any deterministic summand. If only one summand remains, both its atoms
are support endpoints, which are covered by Lemma~\ref{lem:pair-endpoint-margin}.
If two nondegenerate summands remain, relabel and scale so that $h_2=1$ and
$0<h_1=t\leq1$. For $t<1$, the four atoms $x_A=\sum_{i=1}^2(A_i-p_i)h_i$
($A\in\{0,1\}^2$) are ordered as $00<10<01<11$, with distribution values
\[
\begin{array}{c|cc}
 A&F_S^-(x_A)&F_S^+(x_A)\\\hline
 00&0&q_1q_2\\
 10&q_1q_2&q_2\\
 01&q_2&1-p_1p_2\\
 11&1-p_1p_2&1.
\end{array}
\]
Lemma~\ref{lem:pair-endpoint-margin} covers $00$ and $11$. For the closed
branch at $01$, the deficit $H_S^+(x_{01})$ is exactly the left-hand side of
\eqref{eq:two-upper-middle}. For the closed branch at $10$, setting
$\alpha=h_1\sqrt{a_1/V}$ and $\beta=h_2\sqrt{a_2/V}$ (so that
$\alpha^2+\beta^2=1$) transforms $H_S^+(x_{10})$ into the left-hand side of
\eqref{eq:two-lower-middle}. When $t=1$, the interior atoms $10$ and $01$
coalesce, with $F_S^-=q_1q_2$ and $F_S^+=1-p_1p_2$, whose closed deficit is
again \eqref{eq:two-upper-middle}. Finally, replacing $(B_1,B_2)$ by
$(1-B_1,1-B_2)$ negates $S$ while preserving $V$, $R$, and $0<h_1\le h_2$,
so $F_S^-(y)=1-F_{-S}^+(-y)$ reduces all strict deficits $H_S^-(x_A)$ to the
closed deficits already proved.
\end{proof}

As an immediate consequence of Theorem~\ref{thm:bernoulli-classes}(ii), the
sharp Esseen bound also holds for arbitrary mixtures of positively and
negatively oriented Esseen two-point variables of equal variance.

\begin{samepage}
\begin{corollary}[Mixtures of the two Esseen orientations]\label{cor:mixtures}
Fix $c>0$ and $0\leq\eta_i\leq1$ for $1\le i\le n$. If $X_1,\ldots,X_n$ are
independent with marginal distributions
\[
 \mathcal L(X_i)=(1-\eta_i)\mathcal L(cZ)+\eta_i\mathcal L(-cZ),
 \qquad Z\sim\mu_{\mathrm E},
\]
then $\Delta\leq\be L$.
\end{corollary}
\end{samepage}
\begin{proof}
Realize each $X_i$ by first drawing an independent sign choice and then
drawing from $\mathcal L(\pm cZ)$. Conditional on the vector of sign choices,
the summands are independent two-point variables with identical variances
$c^2$, identical third absolute moments $c^3(p_{\mathrm E}^2+q_{\mathrm E}^2)/\sqrt{p_{\mathrm E}q_{\mathrm E}}$,
and common support diameter $c/\sqrt{p_{\mathrm E}q_{\mathrm E}}$.
Theorem~\ref{thm:bernoulli-classes}(ii) gives $|\Pp(S\le x\mid\text{signs})-\Phi(x/\sqrt V)|\le\be L$
for every sign configuration, and averaging over the signs preserves the
inequality.
\end{proof}

\section{Three Bernoulli summands with arbitrary diameters}\label{sec:three}

We now prove Theorem~\ref{thm:bernoulli-classes}(i). Write
$X_i=h_i(B_i-p_i)$ with $q_i=1-p_i$, $a_i=p_iq_i$, and $r_i=p_i^2+q_i^2$ for
$i=1,2,3$; arrays with at most two active summands are already covered by
Theorem~\ref{thm:two-summands}.

Normalize total variance to $B^2=1$. By Cantelli's inequality (see
\citet{BellIndependent2026}), every
centered unit-variance law satisfies
\begin{equation}\label{eq:kappa}
 \Delta<\kappa:=\frac{271}{500}=0.542,
\end{equation}
so any counterexample to $\Delta\le\be R$ must satisfy $R<\kappa/\be<4/3$.

Furthermore, no summand in a counterexample can have variance $w\le1/25$.
Indeed, suppose a summand $X$ has variance $w\le1/25$, leaving a two-summand
two-point sum $T$ of variance $v=1-w\ge24/25$. By Theorem~\ref{thm:two-summands},
$T/\sqrt v$ has deficit exceeding $3/80>1/30$. Replacing $T$ by a normal
variable $N(0,v)$ and comparing $N(0,v)+X$ with $N(0,1)$ by a second-order
Taylor expansion of $\Phi(\cdot/\sqrt v)$ incurs an error bounded by
\[
 L(v)R_X,\qquad
 L(v)=\frac{\phi(0)(1+4\phi(0))}{6v^{3/2}}<\frac15
 \qquad(v\ge24/25),
\]
where $R_X=\E|X|^3\ge w^{3/2}$ and $R_T=R-R_X$. The full deficit therefore
exceeds
\[
 \frac1{30}-\be R_T(v^{-3/2}-1)+(\be-L(v))R_X
 > b(w):=\frac1{30}-\kappa\bigl((1-w)^{-3/2}-1\bigr)+\frac5{24}w^{3/2}.
\]
On $[0,1/25]$, $b'(w)\le-3\kappa/2+1/16<0$ and $b(1/25)>1/2000>0$. Thus any
counterexample must have $w_i=a_ih_i^2>1/25$ for each $i=1,2,3$. Since
$R\ge w_i^{3/2}/(2\sqrt{a_i})$, $R\ge w_ih_i/2$, and $h_i\ge2\sqrt{w_i}$,
the bound $R<4/3$ implies the a priori bounds
\begin{equation}\label{eq:n3-collars}
 a_i>9\times10^{-6},\qquad \frac25<h_i<\frac{200}{3},
 \qquad p_i\in\left[\frac1{125000},\,1-\frac1{125000}\right].
\end{equation}

Relabel and rescale the raw diameters so that $0<h_1\le h_2\le h_3=1$. Depending
on whether $h_1+h_2\le1$ or $h_1+h_2\ge1$, the eight atoms $A\in\{0,1\}^3$
are ordered either as
\begin{gather*}
 000\le100\le010\le001\le110\le101\le011\le111\\
 \text{or}\quad
 000\le100\le010\le110\le001\le101\le011\le111.
\end{gather*}
Proposition~\ref{cor:all-support-owners} covers the extreme endpoints $000$
and $111$. With $V=\sum_{i=1}^3 a_ih_i^2$, $R_h=\sum_{i=1}^3 a_ir_ih_i^3$,
and $z_A=(A-p)\cdot h/\sqrt V$, the six interior atoms in both orderings are
covered by the eight parameterizations $O=(st,s,1)$,

\section{Common-diameter Bernoulli arrays}\label{sec:equal-gap-two-point}

\begin{theorem}[Common support diameter]\label{thm:equal-gap-two-point}
Part \textup{(ii)} of Theorem~\ref{thm:bernoulli-classes} holds: every finite
independent array of two-point summands with a common support diameter
satisfies $\Delta\le\be L$, and the constant $\be$ is optimal.
\end{theorem}

After omitting zero summands and dividing by the common support diameter,
we have $X_i=B_i-p_i$ where $B_1,\ldots,B_n$ are independent with
$B_i\sim\Bern(p_i)$, $p_i\in(0,1)$, and $q_i=1-p_i$. Put
\begin{equation}
 N_p:=\sum_{i=1}^nB_i,\qquad
 \mu:=\sum_{i=1}^np_i,\qquad
 V(p):=\sum_{i=1}^np_iq_i,
 \qquad
 \mathcal B(p):=\sum_{i=1}^np_iq_i(p_i^2+q_i^2),
 \label{eq:equal-gap-bernoulli-data}
\end{equation}
and write $F_j(p)=\Pp(N_p\leq j)$. For an integer $k$ and $V(p)>0$, the closed
and strict deficits are
\begin{align}
 \mathcal H_+(p;k)
 &:=\be\frac{\mathcal B(p)}{V(p)^{3/2}}
   +\Phi\!\left(\frac{k-\mu}{\sqrt{V(p)}}\right)-F_k(p),
 \label{eq:equal-gap-closed-deficit}\\
 \mathcal H_-(p;k)
 &:=\be\frac{\mathcal B(p)}{V(p)^{3/2}}
   -\Phi\!\left(\frac{k-\mu}{\sqrt{V(p)}}\right)+F_{k-1}(p).
 \label{eq:equal-gap-strict-deficit}
\end{align}
By \eqref{eq:one-sided}, $\Delta\le\be L$ is equivalent to
$\mathcal H_\pm(p;k)\ge0$ for all $k\in\Z$, and complementation $B_i\mapsto1-B_i$
gives the exact reflection identity
\begin{equation}
 \mathcal H_-(p_1,\ldots,p_n;k)
 =\mathcal H_+(q_1,\ldots,q_n;n-k).
 \label{eq:equal-gap-owner-reflection}
\end{equation}

\subsection{Third differences of Poisson-binomial point probabilities}

\begin{samepage}
\begin{lemma}[Poisson-binomial third-difference bound]\label{lem:poisson-binomial-third-difference}
Let $c_j=\Pp(C=j)$ ($j\in\Z$) be the point mass function of a finite
Poisson-binomial sum $C=\sum_i B_i$ of variance $W=\Var C$, and let
$\Delta^3c_j:=c_j-3c_{j-1}+3c_{j-2}-c_{j-3}$. Then
\begin{equation}
 \sup_{j\in\Z}|\Delta^3c_j|
 < \frac{8\be}{(W+1)^{3/2}}.
 \label{eq:poisson-binomial-third-difference}
\end{equation}
\end{lemma}
\end{samepage}

\begin{proof}
Set $A=8\be$; the rational bounds $\sqrt{10}>79/25$ and $\pi<22/7$ give
$A>16/5$.

First suppose $0\leq W\leq1/4$. Taking the more probable outcome of each $B_i$
as a reference configuration, let $\varepsilon_i=\min(p_i,q_i)$, $E=\sum_i\varepsilon_i\le2W\le1/2$,
$R=\sum_i\varepsilon_i/(1-\varepsilon_i)\in[W,1]$, and let $J$ count the number
of coordinates that depart from the reference configuration. Then
$P_0=\Pp(J=0)\ge\max(1-E,e^{-R})\ge1/2$ and $P_1=\Pp(J=1)=P_0R\ge We^{-W}$.
If $m$ is the sum of the reference configuration, then at $j\in\{m+1,m+2\}$
the event $J=0$ contributes $\pm3$ to $\Delta^3c_j$ while every $J=1$ outcome
contributes with the opposite sign and magnitude at least $1$, so that
\begin{equation}
 |\Delta^3c_j|
 \leq3P_0-P_1+3\Pp(J\geq2)
 =3-4P_1
 \leq3-4We^{-W}
 \label{eq:third-difference-small-central}
\end{equation}
for $j\in\{m+1,m+2\}$, whereas at all other $j$ the $J=0$ contribution has
magnitude at most $1$ and
\begin{equation}
 |\Delta^3c_j|\leq P_0+3(1-P_0)\leq1+4W\le2<A(5/4)^{-3/2}.
 \label{eq:third-difference-small-outer}
\end{equation}
In \eqref{eq:third-difference-small-central}, using $e^{-W}\ge1-W$ and
$(1+W)^{-3/2}\ge h(W):=1-\frac32W+\frac{15}{8}W^2-\frac{35}{16}W^3$, the
difference $P(W)=Ah(W)-(3-4W+4W^2)$ satisfies
$P'(W)<3h'(W)+4-8W=-\frac12+\frac{13}{4}W-\frac{315}{16}W^2<0$ on $[0,1/4]$
and $P(1/4)=\frac{725}{1024}A-\frac94>\frac1{64}>0$.

Now suppose $W\geq1/4$. Fourier inversion and $|1-p+pe^{\ii t}|\leq\exp\{-2p(1-p)\sin^2(t/2)\}$
give
\begin{equation}
 |\Delta^3c_j|
 \leq I(W):=\frac{16}{\pi}\int_0^{\pi/2}
 \sin^3x\,e^{-2W\sin^2x}\,dx.
 \label{eq:third-difference-fourier-majorant}
\end{equation}
Writing $J_r(W)=\int_0^{\pi/2}\sin^rx\,e^{-2W\sin^2x}\,dx$, integration by
parts of $(\sin^4x\cos x\,e^{-2W\sin^2x})'$ gives $(5+4W)J_5=4J_3+4WJ_7$, so
$J_5/J_3\ge4/(5+4W)$ and
\[
 \frac{d}{dW}\log\bigl\{I(W)(1+W)^{3/2}\bigr\}
 =-2\frac{J_5}{J_3}+\frac{3}{2(1+W)}
 \leq-\frac8{5+4W}+\frac3{2(1+W)}<0.
\]
At $W=1/4$, $e^{-u}\leq1-u+u^2/2$ gives
$I(1/4)\leq\frac{16}{\pi}(J_3(0)-\frac12J_5(0)+\frac18J_7(0))=\frac{256}{35\pi}<A(5/4)^{-3/2}$.
\end{proof}

\subsection{Reduction to a binomial sum plus one Bernoulli summand}

\begin{samepage}
\begin{proposition}[Reduction along elementary-symmetric probability fibers]
\label{prop:equal-gap-one-exception-reduction}
Fix $n\ge1$, $\mu\in(0,n)$, $k\in\Z$, and either deficit in
\eqref{eq:equal-gap-closed-deficit}--\eqref{eq:equal-gap-strict-deficit}.
If that deficit is negative somewhere on $\mathcal P_{n,\mu}:=\{p\in[0,1]^n:\sum_{i=1}^np_i=\mu\}$,
then after deleting deterministic coordinates in $\{0,1\}$ and possibly
applying \eqref{eq:equal-gap-owner-reflection}, a negative global minimum on
a face of minimal dimension has the form
\begin{equation}
 N_p\sim\Bin(M,p)+\Bern(x),
 \qquad M\geq1,\quad 0<p,x<1,\quad p\ne x.
 \label{eq:one-exceptional-family}
\end{equation}
\end{proposition}
\end{samepage}

\begin{proof}
Since $\mathcal B(p)\geq V(p)/2$, the deficit tends to $+\infty$ as
$V(p)\to0$, so it attains its minimum on every closed face of
$\mathcal P_{n,\mu}$. Choose a face of smallest dimension on which the minimum
is negative, and among the minimizers on that face choose one maximizing
$\sum_i p_i^3$. Deleting coordinates equal to $0$ or $1$ (replacing
$(\mu,k)$ by $(\mu-1,k-1)$ when $p_i=1$) preserves the deficit, so minimal
dimension implies $p_i\in(0,1)$ for all remaining coordinates.

Following \citet[pp.~5--6]{Mattner2022}, for any three coordinates $p_1,p_2,p_3$
let $e_1,e_2,e_3$ be their elementary symmetric polynomials:
\begin{equation}
 \prod_{i=1}^3\{1+p_i(w-1)\}
 =1+e_1(w-1)+e_2(w-1)^2+e_3(w-1)^3.
 \label{eq:equal-gap-triple-generating}
\end{equation}
Holding $e_1,e_2$ fixed preserves both their mean $e_1$ and their variance
$V_3=e_1-e_1^2+2e_2$, while making both $F_k(p)$ and their third-moment sum
\begin{equation}
 \mathcal B_3=e_1-3e_1^2+4e_1^3-2e_1^4
 +(6-12e_1+8e_1^2)e_2-4e_2^2+4(3-2e_1)e_3
\label{eq:equal-gap-triple-budget}
\end{equation}
affine in $e_3$. If $p_1,p_2,p_3\in(0,1)$ were distinct, $e_3$ could be varied
in both directions, contradicting either minimality of the deficit (if the
slope is nonzero) or maximality of $p_1^3+p_2^3+p_3^3=e_1^3-3e_1e_2+3e_3$ (if
the slope is zero). Thus at most two distinct values $0<a<b<1$ occur, with
multiplicities $r,s\ge1$ (since $r=0$ or $s=0$ is ruled out by Schulz's
theorem \eqref{eq:schulz}).

Suppose $r,s\ge2$. After applying \eqref{eq:equal-gap-owner-reflection} if
necessary so the deficit is closed, let $d_j$ be the point mass function of
the $n-3$ coordinates outside a chosen triple. The slope of $\mathcal H_+$
with respect to $e_3$ is
\begin{equation}
 L(e_1;d)=4\gamma(3-2e_1)+\Delta^2d_k,
 \qquad \gamma:=\frac{\be}{V^{3/2}},
 \label{eq:equal-gap-fiber-slope}
\end{equation}
where $\Delta^2d_k=d_k-2d_{k-1}+d_{k-2}$. Because $(a,a,b)$ maximizes $e_3$
and $(a,b,b)$ minimizes $e_3$ on their respective $(e_1,e_2)$ fibers,
minimality of $\mathcal H_+$ requires
\begin{equation}
 L_{aab}\leq0\leq L_{abb}.
 \label{eq:equal-gap-fiber-endpoint-signs}
\end{equation}
Letting $c_j$ denote the point mass function of the $n-4$ coordinates obtained
by deleting two copies of $a$ and two copies of $b$, we have
$d^{(aab)}=c*\Bern(b)$ and $d^{(abb)}=c*\Bern(a)$, so subtraction gives
\begin{equation}
 0\le L_{abb}-L_{aab}
 =(b-a)\{-8\gamma+\Delta^3c_k\},
 \label{eq:equal-gap-two-block-slope-difference}
\end{equation}
which contradicts Lemma~\ref{lem:poisson-binomial-third-difference} since
$V\le\Var(C)+1$. Hence $\min(r,s)=1$.
\end{proof}

\subsection{Excluding the one-exception family}

\begin{samepage}
\begin{lemma}[Curvature inequality on the bounded variance domain]
\label{lem:one-exceptional-compact}
Let $0<p,x<1$, $q=1-p$, $pq\leq W\leq3$, $0\leq f\leq1$, and
$A_0:=W-(1-f)q^2\geq0$. Set
\[
 B_0:=p^2+(1-f)(W+2pq),\qquad
 V:=x(1-x)+W+pq,
\]
\[
 t_x:=1-2x,\qquad t_p:=1-2p,\qquad
 y:=\frac{f(W+pq)}{1-qf},\qquad z:=\frac{y-x}{\sqrt V}.
\]
Then
\begin{equation}
 \frac{2\be}{\sqrt V}\{4W+2+t_xt_p\}
 +z\ph(z)(2+z^2)
 -5V\frac{fB_0}{(2-f)B_0+A_0}>0.
 \label{eq:one-exceptional-compact-gap}
\end{equation}
\end{lemma}
\end{samepage}

\begin{proof}
For $V\leq1/5$, set $S_2=t_x^2+t_xt_p+t_p^2\ge(t_x^2+t_p^2)/2\ge1-2V\ge1/3$.
Since $4V+S_2=4W+2+t_xt_p$, the arithmetic--geometric mean inequality gives
$\be(8\sqrt V+2S_2/\sqrt V)\ge8\be/\sqrt3$. Because $z\ph(z)(2+z^2)\ge-4/(e\sqrt\pi)$
and $fB_0/[(2-f)B_0+A_0]\le1$, we have for $V\le1/5$
\begin{equation}
 \frac{8\be}{\sqrt3}-\frac4{e\sqrt\pi}-5V
 \ge\frac{8\be}{\sqrt3}-\frac4{e\sqrt\pi}-1
 >\frac{24}{13}-\frac{16}{19}-1=\frac1{247}>0.
 \label{eq:one-exceptional-small-variance-margin}
\end{equation}
For $V\geq1/5$, using the lower bound
\begin{equation}
 z\ph(z)(2+z^2)\geq\sqrt{\frac2\pi}\min\{z,0\},
 \label{eq:one-exceptional-normal-lower}
\end{equation}
the domain is covered by three explicit maps from $[0,1]^4$ (with fourth
coordinate $x$):
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item $p=(1+s_p)/2$, $W=pq+s_W(3-pq)$, $f=s_f$ (covering $p\ge1/2$);
\item $p=s_p/2$, $W=q^2+s_W(3-q^2)$, $f=s_f$ (covering $p\le1/2$, $W\ge q^2$);
\item $p=s_p/2$, $W=pq+s_W(q^2-pq)$, $f=1-(1-s_f)W/q^2$ (covering $p\le1/2$,
$W\le q^2$, $A_0\ge0$).
\end{enumerate}
The verifier \path{verify/structure/check.py} (\verb|--arithmetic --case common-gap|)
certifies positivity of \eqref{eq:one-exceptional-compact-gap} at the smaller
coefficient $199/500<\be$ across 275,283 accepted boxes (from 550,563
processed boxes) at 192-bit Arb precision.
\end{proof}

\begin{samepage}
\begin{proposition}[Nonnegativity for $\Bin(M,p)+\Bern(x)$]
\label{prop:one-exceptional-bernoulli}
For every $M\geq1$, $0<p,x<1$ with $p\ne x$, and $k\in\Z$, both deficits of
$\Bin(M,p)+\Bern(x)$ are nonnegative.
\end{proposition}
\end{samepage}

\begin{proof}
By \eqref{eq:equal-gap-owner-reflection} and Proposition~\ref{prop:equal-gap-one-exception-reduction},
if a negative deficit occurred, $\mathcal H_+$ would have a negative global
minimum at some $\Bin(M,p)+\Bern(x)$ with $p\ne x$. When $M=1$, this is
excluded by Theorem~\ref{thm:two-summands}, so assume $M\geq2$, put $m=M-1$,
and let $D\sim\Bin(m,p)$, $d_j=\Pp(D=j)$, and $W=mpq$.

Varying the exceptional coordinate $x$ and one copy of $p$ as
$\bar p+d,\bar p-d$ with $u=d^2$ makes the closed CDF affine in $u$:
\begin{equation}
 \mathcal F_k(u)=\mathcal F_k(0)+(d_{k-1}-d_k)u.
 \label{eq:one-exceptional-pair-cdf-affinity}
\end{equation}
With $t_x=1-2x$, $t_p=1-2p$, $\xi=d_{k-1}-d_k$, $S_2=t_x^2+t_xt_p+t_p^2$,
$V=x(1-x)+Mpq$, and $z=(k-x-Mp)/\sqrt V$, we have $V'=-2$,
$\mathcal B'=-2S_2$, and $\mathcal B''=-8$ with respect to $u$. Eliminating
$\mathcal B$ at the stationary point $\mathcal H_+'(u)=0$ yields the exact
second derivative
\begin{equation}
 V^2\mathcal H_+''
 =5V\xi
 -\be\left(8\sqrt V+\frac{2S_2}{\sqrt V}\right)
 -z\ph(z)(2+z^2).
 \label{eq:one-exceptional-stationary-curvature}
\end{equation}
By the binomial recurrence,
\begin{equation}
 \xi=
 \begin{cases}
 0, & k<0\text{ or }k\geq m+2,\\
 -d_0, & k=0,\\
 d_{k-1}\dfrac{k-(m+1)p}{kq}, & 1\leq k\leq m+1.
 \end{cases}
 \label{eq:one-exceptional-rank-split}
\end{equation}
If $\xi\leq0$, then $V^2\mathcal H_+''\le-(8\be/\sqrt3-4/(e\sqrt\pi))<0$. If
$\xi>0$, put $y=k-(m+1)p>0$, $f=y/(kq)\in(0,1]$, and $j=k-1$, so that
$d_{j+1}=(1-f)d_j$, $d_{j-1}=r_0d_j$, and $d_{j-1}+d_j+d_{j+1}\le1$ give
\begin{equation}
 \xi=fd_j\leq\frac{f}{2-f+r_0},
 \label{eq:one-exceptional-three-neighbor}
\end{equation}
with
\begin{equation}
 r_0=\frac{A_0}{B_0},\qquad
 A_0=W-(1-f)q^2\ge0,\qquad
 B_0=p^2+(1-f)(W+2pq),
 \label{eq:one-exceptional-neighbor-elimination}
\end{equation}
and
\begin{equation}
 y=\frac{f(W+pq)}{1-qf},\qquad
 \be\left(8\sqrt V+\frac{2S_2}{\sqrt V}\right)
 =\frac{2\be}{\sqrt V}\{4W+2+t_xt_p\}.
 \label{eq:one-exceptional-phase-compression}
\end{equation}
For $W\leq3$, Lemma~\ref{lem:one-exceptional-compact} implies
$V^2\mathcal H_+''<0$. For $W\geq3$, Fourier inversion gives
\begin{equation}
 |\xi|=|d_{k-1}-d_k|
 \leq\frac4\pi\int_0^{\pi/2}\sin u\,e^{-2W\sin^2u}\,du
 <\frac2{\pi W},
 \label{eq:one-exceptional-fourier-tail}
\end{equation}
so $-V^2\mathcal H_+''\ge8\be\sqrt V-4/(e\sqrt\pi)-10V/(\pi W)\ge44/45>0$ on
$V\in[W,W+1/2]$. In every case $\mathcal H_+''<0$, contradicting local
minimality.
\end{proof}

\section{Moment duality and the three-point reduction}\label{sec:moments}

When all summands except $X_i$ are held fixed, the distribution function of
$S$ at a fixed threshold $y$ is an expectation $\E h(X_i)$. Maximizing the
Berry--Esseen deficit at fixed summand variance $s=\E X_i^2$ is therefore a
linear optimization problem over probability measures with prescribed first
and second moments, governed by the classical extreme-point and duality
theorems of \citet[Theorem~2.1]{Winkler1988} and
\citet[Theorem~2.2]{BertsimasPopescu2005}.

\begin{samepage}
\begin{theorem}[Strong quadratic moment duality]\label{thm:strongdual}
Let $s>0$ and let $h:\R\to\R$ be upper semicontinuous and satisfy
\[
 -K_-(1+|u|^3)\leq h(u)\leq K_+-\lambda|u|^3
\]
for some constants $K_-,K_+<\infty$ and $\lambda>0$. Then
\begin{equation}
 \max_{\substack{P:\ \int dP=1,\ \int u\,dP=0,\ \int u^2\,dP=s\\
                         \int|u|^3dP<\infty}}
       \int h\,dP
 =\min_{\substack{q(u)=A+Bu+Cu^2\\q\geq h\ \text{on }\R}}(A+Cs).
 \label{eq:coordinate-duality}
\end{equation}
Both extrema are attained. Every extreme point of the set of primal maximizers
is supported on at most three points, and every primal maximizer is supported
within the contact set $\{u\in\R:q(u)=h(u)\}$ of every optimal quadratic
majorant $q$.
\end{theorem}
\end{samepage}

\begin{proof}
The upper bound $h(u)\le K_+-\lambda|u|^3$ forces any maximizing sequence to
have uniformly bounded third absolute moments, which ensures tightness and
uniform integrability of $u$ and $u^2$. By upper semicontinuity of $h$ and
Fatou's lemma, the set of primal maximizers is nonempty, convex, and weakly
compact, and forms an extremal face of the feasible moment set. If an extreme
feasible measure $P$ charged four disjoint Borel sets $A_1,\ldots,A_4$, their
moment vectors $(P(A_j),\int_{A_j}u\,dP,\int_{A_j}u^2\,dP)\in\R^3$ would be
linearly dependent, yielding a nonzero bounded density perturbation
$(1\pm\eps\sum_{j=1}^4c_j\ind_{A_j})P$ preserving the three moments. Thus
every extreme maximizer has at most three atoms.

For strong duality, the moment-hypograph cone
\[
 \mathcal D=\left\{\left(\mu(\R),\int u\,d\mu,\int u^2\,d\mu,z\right):
 \mu\ge0,\ \int|u|^3\,d\mu<\infty,\ z\le\int h\,d\mu\right\}\subset\R^4
\]
is convex and closed (by the same cubic-coercivity compactness argument). At
its boundary point $p=(1,0,s,M)$, where $M$ is the primal maximum, a nonzero
supporting hyperplane $am_0+bm_1+cm_2+dz\le0$ has $d>0$ because $(1,0,s)$ lies
in the interior of the moment cone $\{(m_0,m_1,m_2):m_0>0,m_2\ge m_1^2/m_0\}$.
Normalizing $d=1$ and testing against point masses $\delta_u$ shows that
$q(u)=-a-bu-cu^2\ge h(u)$ on $\R$ with $q(0)+q''(0)s/2=-a-cs=M$, establishing
\eqref{eq:coordinate-duality} and complementary slackness $\int(q-h)\,dP=0$.
\end{proof}

\begin{samepage}
\begin{theorem}[Three-point reduction]\label{thm:threepoint}
Any independent centered array satisfying $\Delta>c L$ for some $c>0$ can be
replaced by an independent centered array with the same number of summands
and the same individual variances, each supported on at most three points,
satisfying $\Delta>c L$.
\end{theorem}
\end{samepage}
\begin{proof}
After reflection if needed, fix a threshold $y$ with $F_S^+(y)-\Phi(y/B)>cL$.
Holding all summands except $X_i$ fixed with $T_i=S-X_i$, apply
Theorem~\ref{thm:strongdual} to the upper semicontinuous function
$h_i(u)=F_{T_i}^+(y-u)-cB^{-3}|u|^3$ at variance $s_i=\E X_i^2$ to replace
$X_i$ by a three-point extreme maximizer without decreasing
$F_S^+(y)-\Phi(y/B)-cB^{-3}\sum_j\E|X_j|^3$.
\end{proof}

\section{Proof of Theorem~\ref{thm:arbitrary-two}: two arbitrary summands}\label{sec:arbitrary-two}

Normalize total variance to $B^2=1$. By reflection, $C_2$ is the supremum of
$D/R$, where $D=\Pp(X_1+X_2\le z)-\Phi(z)>0$ and $R=\E|X_1|^3+\E|X_2|^3$.

\begin{lemma}[Attainment by three-point marginals]\label{lem:c2-attainment}
If $C_2\ge49/120$, then $C_2$ is attained at some threshold $z\in\R$ by two
nondegenerate centered distributions $X_1,X_2$, each supported on at most
three points.
\end{lemma}
\begin{proof}
Along a maximizing sequence with $D_j/R_j\to C_2\ge49/120$, we have
$1/\sqrt2\le v_{1,j}^{3/2}+v_{2,j}^{3/2}\le R_j\le D_j/c\le1/c$ (for any fixed
$c\in(463/1250,49/120)$), while Chebyshev's inequality and $1-\Phi(z_j)$ keep
the thresholds $z_j$ bounded. Passing to a weak limit preserves total
variance one (by uniform integrability of squares) and yields $D\ge C_2r\ge C_2R$,
so equality $D/R=C_2$ holds. Because $C_1<463/1250<49/120$ by
\citet[Theorem~1]{BentkusKirsa1989}, both limiting marginals have positive
variance. Replacing each marginal by a three-point extreme maximizer from
Theorem~\ref{thm:strongdual} preserves $D-C_2R=0$ and shifts $z$ to a sum
atom $x_j+y_k=z$.
\end{proof}

At an attained maximum with $C=C_2\in[49/120,227/500)$ (where the upper bound
follows from \citet{BellIndependent2026}), set $s_i=\E X_i^2$,
$\beta_i=\E|X_i|^3$, $M_i=\E[X_i|X_i|]$, and
\[
 A=\frac{z\phi(z)+3D}{2},\qquad
 K=\frac{2A}{3C}=R+\frac{z\phi(z)}{3C}.
\]

\begin{lemma}[Variational contact majorants]\label{lem:full-law-contacts}
For $i=1,2$, the quadratic polynomial
\[
 q_i(u)=F+A s_i-C\beta_i-\bigl(\phi(z)+3CM_i\bigr)u-Au^2
\]
majorizes $F_{X_{3-i}}^+(z-u)-C|u|^3$ on $\R$, with equality at every support
point of $X_i$. Moreover, $f_i(u)=q_i(u)+C|u|^3$ satisfies $f_i'(u)\le0$ at
every support point $u\in\supp X_i$, with $f_i'(u)=0$ whenever $z-u\notin\supp X_{3-i}$.
\end{lemma}
\begin{proof}
Perturbing $\mathcal L(X_i)$ to $(1-t)\mathcal L(X_i)+t\delta_u$, recentering
by $tu$, and shifting the threshold to $z-tu$ gives right derivatives
$F_{X_{3-i}}^+(z-u)-F$, $u^2-s_i$, and $|u|^3-\beta_i-3uM_i$ for the event
probability, variance, and third-moment sum at $t=0$. Nonpositivity of the
right derivative of $D(t)V(t)^{3/2}/R(t)$ yields $F_{X_{3-i}}^+(z-u)-C|u|^3\le q_i(u)$,
and integrating against $\mathcal L(X_i)$ gives equality on $\supp X_i$.
Because $u\mapsto F_{X_{3-i}}^+(z-u)$ is nonincreasing and is constant near
$u$ when $z-u\notin\supp X_{3-i}$, the smooth majorant $f_i$ satisfies
$f_i'(u)\le0$ at every contact point and $f_i'(u)=0$ away from threshold
representations.
\end{proof}

From Cantelli's inequality ($D<11/20$), $|z\phi(z)|<1/\sqrt{2\pi e}<49/200$,
and $R\ge1/\sqrt2$, we obtain the a priori bounds
\begin{equation}\label{eq:contact-basic-collars}
 \frac12<K<\frac85,
\end{equation}
and comparing a dominant summand of variance $v\ge187/200$ against a Gaussian
via the Bentkus--Kir\v{s}a bound $C_1<463/1250$ and $L(v)<1/5$ yields
\begin{equation}\label{eq:contact-variance-collar}
 \frac{13}{200}<s_i<\frac{187}{200}\qquad(i=1,2).
\end{equation}

Furthermore, any centered three-point law on $\{-a,c,b\}$ ($a,b>0$,
$-a<c<b$) is a mixture of two centered two-point laws sharing an extreme
atom, whose conditional variances differ by $d=\min\{b(a+c),a(b-c)\}$ (we set
$d=0$ for a two-point law). When $d_1+d_2\le B^2/3$, conditioning on the
mixture components and applying Theorem~\ref{thm:two-summands} bounds the
ratio strictly below $49/120$:

\begin{lemma}[Exclusion of small mixture-variance widths]\label{lem:mixture-width}
If two centered distributions supported on at most three points have
mixture-variance widths $d_1+d_2\le B^2/3$, then
\[
 \be L-\Delta>\frac{4877}{1056500},\qquad
 \frac{\Delta}{L}<\frac{572623}{577500}\be<\frac{49}{120}.
\]
\end{lemma}
\begin{proof}
Normalize $B^2=1$ and draw the two mixture labels independently. Each
conditional array consists of two two-point summands with total variance
$W\in[2/3,4/3]$ of mean $1$, and its conditional third absolute moment is
affine in its variance with nonnegative slope. Because $u\mapsto u^{-3/2}$ is
decreasing, $\E[(U_i-\E U_i)W^{-3/2}]\le0$, so the average of $\be$ times the
conditional Lyapunov ratio exceeds $\be R$ by at most $\be R(\E W^{-3/2}-1)$,
while the normal CDF convexity error is at most $(7/50)\E[-\log W]$ (since
$|\partial_v^2\Phi(y/\sqrt v)|\le\phi'''(y/\sqrt v)/(4v^2)<7/(50v^2)$). Over
all centered random variables $W$ of mean $1$ and support width at most $1/3$,
a 256-cell rational midpoint-derivative check verifies $\E W^{-3/2}-1<113/2000$
and $\E[-\log W]<1/70$. Combining the conditional margin $3/80$ from
Theorem~\ref{thm:two-summands} with $\Delta<\kappa=271/500$ gives the stated
bounds.
\end{proof}

\begin{proposition}[Exclusion of all three-point contact configurations]\label{prop:finite-contact-exclusion}
No pair of nondegenerate centered distributions supported on at most three
points can attain a global maximum of $\Delta/L$ with $49/120\le C<227/500$.
\end{proposition}
\begin{proof}
By Theorem~\ref{thm:two-summands}, at least one summand must be a
nondegenerate three-point distribution, and Lemma~\ref{lem:mixture-width}
gives $d_1+d_2>1/3$. Ordering the support points of $X_1$ and $X_2$, the event
$\{(j,k):x_j+y_k\le z\}$ is a downset encoded by the number of included pairs
in each row of $X_1$.
\begin{itemize}[leftmargin=2em]
\item \emph{One three-point and one two-point summand ($3\times2$ grid):}
Among the ten downsets $000,100,110,111,200,210,211,220,221,222$, the contact
conditions of Lemma~\ref{lem:full-law-contacts} immediately rule out $000,100,200,111,222$.
Appendix~\ref{app:contacts} rules out $211,220,221$ (unique representation),
the middle/low representation of $210$, and the $211$ collision analytically,
and rules out the remaining four configurations (\texttt{lower110}, \texttt{low210},
\texttt{collision210}, \texttt{collision221}) by certified interval covers.
\item \emph{Two three-point summands ($3\times3$ grid):}
Letting $\eta_i=(\alpha_i\delta_{x_{i,0}}-\delta_{x_{i,1}}+\beta_i\delta_{x_{i,2}})/(u_iv_i)$
be the mean-zero, variance-zero signed measure on $\supp X_i$, the joint
perturbation $(P_1+t\eta_1)\otimes(P_2-t\eta_2)$ preserves total variance and
makes $R$ affine in $t$, so maximality requires
$g=\iint\ind_{\{x+y\le z\}}\,d\eta_1d\eta_2\ge0$. This sign condition
immediately rules out $110,200,210,211,310,311,332$, while the $g=0$ shapes
$000,111,222,300,330,333$ and $100$ are ruled out by Lemma~\ref{lem:full-law-contacts}.
Appendix~\ref{app:contacts} rules out all remaining unique-representation
configurations and $322,331$ collisions analytically, and rules out the
remaining collision configurations (\texttt{three\_owner321},
\texttt{adjacent\_owner321}, \texttt{endpoint\_owner321}, \texttt{tt221\_negative},
\texttt{tt221\_positive}) by certified interval covers, including strict
exclusion at the boundary $C=49/120$.
\end{itemize}
Together with Lemma~\ref{lem:c2-attainment}, this completes the proof of
Theorem~\ref{thm:arbitrary-two}.
\end{proof}

\section{Transfer to larger arrays with two dominant summands}\label{sec:transfer}

Normalize the total variance of $(X_1,\ldots,X_n)$ to $B^2=1$. For $0<v\le1$,
set
\[
 K(v)=\frac{\phi(0)}2\int_0^1
       \frac{s^2\,ds}{[1-(1-v)s^2]^{3/2}}.
\]

\begin{lemma}[Gaussian remainder comparison]\label{lem:gaussian-comparison}
If $T$ is a sum of independent centered random variables with total variance
$1-v$ and third absolute moment sum $r$, and $G_v\sim N(0,v)$ is independent
of $T$, then $d_{\mathrm K}(T+G_v,N(0,1))\le K(v)r$.
\end{lemma}
\begin{proof}
Put $a=1-v>0$. By \citet[Lemma~1.1]{Goldstein2010} and
\citet[Theorem~3 and Corollary~1]{Tyurin2009}, the zero-biased distribution
$T^*$ satisfies $W_1(T,T^*)\le r/(2a)$. For $h(u)=\Phi((x-u)/\sqrt v)$, the
Stein equation $af'(u)-uf(u)=h(u)-\E h(\sqrt a Z)$ has solution
$f(u)=-\int_0^1\E h'(su+\sqrt{a(1-s^2)}Z)\,ds$, with
$\|f''\|_\infty\le\phi(0)\int_0^1s^2(1-as^2)^{-3/2}\,ds$, so
$|\E h(T)-\Phi(x)|=a|\E f'(T)-\E f'(T^*)|\le aW_1(T,T^*)\|f''\|_\infty\le K(v)r$.
\end{proof}

Partitioning the array into two disjoint blocks $H_1,H_2$ of combined variance
$v=\Var(H_1+H_2)>0$ and original third-moment sum $b$, plus an independent
remainder $T$ of third-moment sum $r=R-b$, let
$e=\E|H_1|^3+\E|H_2|^3-b\ge0$ be the increase in third absolute moments under
grouping (nonnegativity follows from the identity
\begin{equation}\label{eq:cubic-grouping}
 \E|X+Y|^3-\E|X|^3-\E|Y|^3
 =2\E\!\left[\ind_{\{XY<0\}}\min(|X|,|Y|)^2
       \{3\max(|X|,|Y|)-\min(|X|,|Y|)\}\right]\ge0
\end{equation}
for independent centered $X,Y$). Conditioning on $T$, applying
Theorem~\ref{thm:arbitrary-two} to $H_1+H_2$, and then applying
Lemma~\ref{lem:gaussian-comparison} gives
\begin{equation}\label{eq:cost-aware}
 \Delta\le\frac{c_2(b+e)}{v^{3/2}}+K(v)r,
\end{equation}
so that $\Delta\le\be R$ holds whenever
\begin{equation}\label{eq:cost-criterion}
 \frac{c_2}{v^{3/2}}\left(\frac bR+\frac eR\right)
 +K(v)\left(1-\frac bR\right)\le\be.
\end{equation}
In particular, since $K(v)<\frac2{75}+\frac1{25v^{3/2}}$ by convexity,
\begin{equation}\label{eq:half-resource}
 v\ge\frac{11}{16},\qquad b\le r,\qquad e\le\frac R{500}
 \quad\Longrightarrow\quad\Delta<\be R.
\end{equation}

\begin{proof}[Proof of Corollary~\ref{cor:dominance-main}]
For two arbitrary summands ($e=0$), \eqref{eq:cost-aware} implies
$\Delta\le\be R$ whenever $c_2/v^{3/2}\le\be$ and $K(v)\le\be$, both of which
hold for $v\ge439/440$ since $(49/120)^2<(40973/100000)^2(439/440)^3$. For
two two-point summands, Theorem~\ref{thm:two-summands} gives
$\Delta<\be b/v^{3/2}-3/80+K(v)r$; if $\Delta>\be R$, then $\be R<\kappa=271/500$
and $K(v)<\be$ give $\Delta-\be R<\kappa(v^{-3/2}-1)-3/80\le0$ for
$v\ge22/23$ since $(22/23)^3>(1084/1159)^2$.
\end{proof}

\subsection{Signed convolution refinements}

Retaining the signed error $f_H(x)=F_H(x)-\Phi(x/\sqrt v)$ of a two-summand
partial sum $H=X_1+X_2$ under convolution with $T$ gives the sharper bound
\begin{equation}\label{eq:signed-convolution}
 \Delta(H+T)\le\mathcal N_T(f_H)+K(v)r,\qquad
 \mathcal N_T(f_H)=\sup_{x\in\R}|\E f_H(x-T)|.
\end{equation}
For example, for three independent Rademacher summands of size $1/\sqrt3$
($v=2/3$), $\|f_H\|_\infty=1/4$ exceeds $\be/\sqrt3$, whereas
$\mathcal N_T(f_H)=\frac14-\frac12\Phi(-\sqrt2)$ and
$\mathcal N_T(f_H)+K(2/3)/(3\sqrt3)<\be/\sqrt3$. Similarly, convolving $f_H$
with an explicit independent normal $G_a\sim N(0,a)$ and using Both
$\|f_H\|_\infty\le c_2b/v^{3/2}$ and $\|f_H\|_1\le b/v$
\citep{Goldstein2010,Tyurin2009} yields
\begin{equation}\label{eq:signed-gaussian}
 d_{\mathrm K}(H+G_a,G_{v+a})\le\frac{c_2b}{v^{3/2}}
 \left\{2\Phi\!\left(\frac{\sqrt v}{4c_2\sqrt a}\right)-1\right\}.
\end{equation}

\section{Proof of the support-endpoint inequalities}\label{app:support}

In this appendix we prove Proposition~\ref{cor:all-support-owners}. Set
$V=\sum_{i=1}^n p_iq_ih_i^2$ and $\mathcal B_h=\sum_{i=1}^n p_iq_i(p_i^2+q_i^2)h_i^3$.

\begin{samepage}
\begin{lemma}[Atom-free branches at the support endpoints]
\label{lem:support-owners-without-atom-term}
For any $n\geq1$, $p_i\in(0,1)$, $q_i=1-p_i$, $h_i>0$, $u=\sum_{i=1}^nq_ih_i$,
and $m=\sum_{i=1}^np_ih_i$,
\begin{equation}
 \be\frac{\mathcal B_h}{V^{3/2}}
 >\max\left\{
  1-\Phi\!\left(\frac{u}{\sqrt V}\right),\,
  \Phi\!\left(-\frac{m}{\sqrt V}\right)
 \right\}.
\label{eq:support-owner-tail-bound}
\end{equation}
Consequently, the closed deficit at the upper support endpoint and the strict
deficit at the lower support endpoint are strictly positive.
\end{lemma}
\end{samepage}

\begin{proof}
By complementation $B_i\mapsto1-B_i$, it suffices to prove the upper-endpoint
bound. Normalize $V=1$ and put $x_i=q_ih_i$, $v_i=p_iq_ih_i^2$,
$b_i=p_iq_i(p_i^2+q_i^2)h_i^3$, and $x=\sum_{i=1}^nx_i$. With
$c=2(\sqrt2-1)>49/60$, the identity $(p_i^2+q_i^2)/p_i-c=(\sqrt2p_i-1)^2/p_i\ge0$
and Cauchy--Schwarz give
\begin{equation}
 \mathcal B_h=\sum_{i=1}^n\frac{p_i^2+q_i^2}{p_i}\frac{v_i^2}{x_i}
 \geq c\sum_{i=1}^n\frac{v_i^2}{x_i}
 \geq\frac{c}{x}.
\label{eq:support-owner-resource-bound}
\end{equation}
On the other hand, for every $x>0$,
\begin{equation}
 x\{1-\Phi(x)\}<\be c.
\label{eq:support-owner-normal-tail}
\end{equation}
Indeed, for $0<x\le2/3$, $x\{1-\Phi(x)\}\le1/3<2401/7200<\be c$, while for
$x\ge2/3$, Mills' ratio gives $x\{1-\Phi(x)\}<\ph(x)\le\ph(2/3)=\ph(0)e^{-2/9}<\ph(0)c<\be c$.
Combining \eqref{eq:support-owner-resource-bound} and
\eqref{eq:support-owner-normal-tail} proves \eqref{eq:support-owner-tail-bound}.
\end{proof}

\begin{samepage}
\begin{proposition}[Sharp envelope for Bernoulli odds]
\label{prop:product-normalized-odds-envelope}
Let $o_1,\ldots,o_n\geq0$, not all zero, and set
\begin{equation}
 s=\sum_{i=1}^n o_i,\qquad
 \Pi=\prod_{i=1}^n(1+o_i)^{-1},\qquad
 g(o)=\frac{o(1+o)^2}{(1+o^2)^2},\qquad
 \mathcal A=\sum_{i=1}^n g(o_i).
\label{eq:support-odds-data}
\end{equation}
Then
\begin{align}
 \Pi^2\mathcal A&\leq\frac13&&\hspace{-4em}(0<s\leq1),
\label{eq:support-odds-small-envelope}\\
 \Pi^2\mathcal A&\leq
 \frac{s}{(1+s/2)^2(1+s^2/4)^2}&&\hspace{-4em}\left(1\leq s\leq\frac{16}{9}\right),
\label{eq:support-odds-live-envelope}\\
 \Pi^2\mathcal A&\leq\frac{944784}{6076225}&&\hspace{-4em}\left(s\geq\frac{16}{9}\right).
\label{eq:support-odds-large-envelope}
\end{align}
Moreover, along any nonzero ray $v\in[0,\infty)^n$, $\mathcal F_v(t)=\Pi(tv)^2\mathcal A(tv)$
satisfies $\frac{d}{d\log t}\log\mathcal F_v(t)<-3/100$ for $t\sum_iv_i\ge16/9$,
and equality in \eqref{eq:support-odds-live-envelope} holds if and only if
two odds equal $s/2$ and all others vanish.
\end{proposition}
\end{samepage}

\begin{proof}
To analyze the intermediate range $1\le s\le16/9$, isolate two coordinates
$x=o_1,y=o_2$ with sum $a=x+y$ and product $k=xy\in(0,a^2/4]$. Writing
$H=\sum_{j=3}^ng(o_j)$, $\Pi_0=\prod_{j=3}^n(1+o_j)^{-1}$,
$z_a(k)=(1+a+k)^{-2}$, and $R_a(k)=z_a(k)\{g(x)+g(y)\}$, we have
\begin{equation}
 \Pi^2\mathcal A=\Pi_0^2\{R_a(k)+Hz_a(k)\}.
\label{eq:support-odds-affine-environment}
\end{equation}
With $D=1+a^2-2k+k^2$, we have $R_a(k)=N(a,k)/[D^2(1+a+k)^2]$, where
$N(a,k)=a+2a^2+a^3+(a^3-a-4)k+(2a^2-a+8)k^2+(a-4)k^3$. At the boundary
endpoints $k=0$ (merging $(x,y)\mapsto(a,0)$) and $k=a^2/4$ (equal split
$x=y=a/2$), the values $\mathcal C_1(a)=a/(1+a^2)^2$ and
$\mathcal C_2(a)=a/[(1+a/2)^2(1+a^2/4)^2]$ cross at $a_*=3-\sqrt5$.
\begin{itemize}[leftmargin=2em]
\item For $0\le a\le a_*$, converting the polynomial quotient
$\mathcal S(a,k)=[aD^2(1+a+k)^2-(1+a^2)^2N(a,k)]/[k(1+a)^2]$ to the tensor
Bernstein basis of degree $(60,20)$ on $(a/a_*,4k/a^2)\in[0,1]^2$ yields
$1280$ strictly positive coefficients and a single zero at the corner
$(a,k)=(a_*,a_*^2/4)$, proving $R_a(k)\le R_a(0)$ (and hence
$R_a(k)+Hz_a(k)\le R_a(0)+Hz_a(0)$ since $z_a(k)$ decreases in $k$).
\item For $a_*\le a\le16/9$, parametrizing $z_a(k)=\theta z_a(0)+(1-\theta)z_a(a^2/4)$
by $\theta\in[0,1]$, all $70$ tensor Bernstein coefficients of the polynomial
quotient $\mathcal L(a,k)$ on $[a_*,16/9]\times[0,a^2/4]$ are strictly
positive, proving the strict chord inequality
\begin{equation}
 R_a(k)<\theta R_a(0)+(1-\theta)R_a(a^2/4)
 \qquad\left(0<k<\frac{a^2}{4}\right).
\label{eq:support-odds-pair-chord}
\end{equation}
\end{itemize}
Consequently, at any maximizer of $\Pi^2\mathcal A$ for $s\in[1,16/9]$, no
pair of positive odds can have sum $\le a_*$, and every pair in the strict
chord range must be equal, so all $d\in\{1,2,3,4\}$ positive odds are equal to
$s/d$. Comparing the four functions $\mathcal C_d(s)=s/[(1+(s/d)^2)^2(1+s/d)^{2(d-1)}]$
on $[1,16/9]$ shows $\mathcal C_2(s)>\mathcal C_d(s)$ for $d\in\{1,3,4\}$,
proving \eqref{eq:support-odds-live-envelope}.

For $0<s\le1$, expanding $\Pi^2\mathcal A=\sum_i o_i/[(1+o_i^2)^2\prod_{j\ne i}(1+o_j)^2]$
and applying the elementary inequality
\begin{equation}
 (1+x^2)^2(1+y)^2\geq3(x+y)
 \qquad(x>0,\ y\geq0)
\label{eq:support-odds-termwise-denominator}
\end{equation}
with $x=o_i$ and $y=s-o_i\le\prod_{j\ne i}(1+o_j)-1$ bounds each term by
$o_i/(3s)$, proving \eqref{eq:support-odds-small-envelope}. Finally, for
$s\ge16/9$, using $e(o):=og'(o)/g(o)<5/4$ and subadditivity of $o\mapsto o/(1+o)$
gives
\[
 \frac{d}{d\log t}\log\mathcal F_v(t)
 =\frac{\sum_i g(o_i)e(o_i)}{\mathcal A}-2\sum_i\frac{o_i}{1+o_i}
 <\frac54-\frac{2s}{1+s}\leq-\frac3{100},
\]
so scaling back to $s=16/9$ yields \eqref{eq:support-odds-large-envelope}. The
Bernstein conversions use
\begin{equation}\label{eq:bernstein-conversion}
 b_\beta=\sum_{\alpha\le\beta}a_\alpha\prod_j
 \frac{\binom{\beta_j}{\alpha_j}}{\binom{d_j}{\alpha_j}}
\end{equation}
and are checked in \path{verify/finite/support_odds/}.
\end{proof}

\begin{samepage}
\begin{lemma}[Atom-bearing branches at the support endpoints]
\label{lem:atom-bearing-support-owners}
Under the hypotheses of Lemma~\ref{lem:support-owners-without-atom-term}, the
closed deficit at the lower support endpoint and the strict deficit at the
upper support endpoint are strictly positive.
\end{lemma}
\end{samepage}

\begin{proof}
Normalizing $V=1$ and setting $o_i=p_i/q_i$, $w_i=p_iq_ih_i^2$, $x_0=\sum_i\sqrt{o_iw_i}$,
and $\Pi=\prod_i(1+o_i)^{-1}$, Lemma~\ref{lem:reciprocal-gap-resource} and
Cauchy--Schwarz give $\mathcal B_h\ge\mathcal A^{-1/2}$ and $x_0\le\sqrt s$.
With $Q(x)=1-\Phi(x)$, it therefore suffices to show
\begin{equation}
 \frac{\be}{\sqrt{\mathcal A}}+Q(\sqrt s)>\Pi.
\label{eq:atom-bearing-weights-free-target}
\end{equation}
First note that
\begin{equation}
 f(s):=(1+s)Q(\sqrt s),\qquad f'(s)<0\quad(s>0),
\label{eq:atom-bearing-tail-monotonicity}
\end{equation}
since $Q(x)/\ph(x)<(1+x^2)/(2x)$ for all $x>0$. On $0<s\le1$,
\eqref{eq:support-odds-small-envelope} gives $\be/(\Pi\sqrt{\mathcal A})>7/10$,
while $Q(\sqrt s)/\Pi\ge(1+s)Q(\sqrt s)\ge2Q(1)>47/150>3/10$, whose sum
exceeds $1$. On $1\le s\le16/9$, \eqref{eq:support-odds-live-envelope} with
$u=s/2\ge1/2$ gives
\begin{equation}
 \frac{\be}{\Pi\sqrt{\mathcal A}}
 \geq\be\frac{(1+u)(1+u^2)}{\sqrt{2u}}
 >\frac{49}{64},
\label{eq:atom-bearing-live-resource-payment}
\end{equation}
while \eqref{eq:atom-bearing-tail-monotonicity} gives
\begin{equation}
 \frac{Q(\sqrt s)}\Pi\geq(1+s)Q(\sqrt s)\ge\frac{25}{9}Q\!\left(\frac43\right)>\frac{15}{64},
\label{eq:atom-bearing-live-tail-payment}
\end{equation}
which again sum to more than $1$. Finally, for $s\ge16/9$,
\eqref{eq:support-odds-large-envelope} gives
$\Pi^2\mathcal A\le944784/6076225<(49/120)^2<\be^2$, so $\be/\sqrt{\mathcal A}>\Pi$
already holds. Together with Lemma~\ref{lem:support-owners-without-atom-term},
this completes the proof of Proposition~\ref{cor:all-support-owners}.
\end{proof}
