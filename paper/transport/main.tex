\begin{abstract}
For a finite sum $S=\sum_{i=1}^nX_i$ of independent centered real random
variables with total variance $V=\sum_{i=1}^n\E X_i^2>0$ and third absolute
moment sum $R=\sum_{i=1}^n\E|X_i|^3<\infty$, the $L^1$-Wasserstein distance
between $S$ and its zero-biased distribution $S^*$ satisfies
$2VW_1(S,S^*)\le R$. We prove that equality holds if and only if all
nonconstant summands $X_i$ share a single centered two-point distribution.
The single-summand characterization uses Choquet's theorem on the compact
convex set of centered laws with variance at most one to represent any unit-variance
law as a barycenter of three-point laws of variance equal to one, and shows
that three-point laws and mixtures of distinct two-point laws have strictly
smaller transport distance. Across multiple summands, unequal support
diameters incur strict convolution loss, while for common-diameter Bernoulli
arrays an exact conditional-variance identity expresses the transport deficit
$R/(2V)-W_1(S,S^*)$ as a weighted variance of the success probabilities.
We then establish a sharp inequality relating lattice span, skewness, and
zero-bias transport, where equality holds precisely for centered binomial
laws with Esseen parameter $p_{\mathrm E}=(4-\sqrt{10})/2$ or $1-p_{\mathrm E}$.
Combining these two rigidity theorems determines the exact asymptotic
Berry--Esseen ratio when a fixed finite block of independent summands is
repeated independently.
\end{abstract}

\section{Introduction and main results}\label{sec:results}

\subsection{The zero-bias transport inequality and its equality cases}

In Esseen's classical construction \citep{Esseen1956}, the extremal constant
$\ce=(3+\sqrt{10})/(6\sqrt{2\pi})$ in the Berry--Esseen inequality is
approached by sums of identically distributed two-point random variables.
When one passes from identically distributed summands to general independent
arrays $(X_1,\ldots,X_n)$, two structural questions arise: what mechanism
forces extremal summands to be supported on two points, and what forces
distinct summands to share the same two-point distribution?

In this paper we answer both questions completely for the zero-bias
Wasserstein transport inequality, and then combine that characterization with
a sharp lattice transport inequality to determine the extremal arrays for
repeated independent blocks.

For probability measures $\mu,\nu$ on $\R$ with finite first moments, let
\[
 W_1(\mu,\nu)=\inf\{\E|U-V|:\Law(U)=\mu,\ \Law(V)=\nu\}
             =\int_\R|F_\mu(x)-F_\nu(x)|\,dx
\]
denote their $L^1$-Wasserstein distance; we also write $W_1(U,V)$ for
$W_1(\Law(U),\Law(V))$. If $X$ is a centered real random variable with
variance $v=\E X^2>0$, its zero-biased distribution $X^*$, introduced by
\citet[Definition~1.1 and Lemma~2.1]{GoldsteinReinert1997}, is characterized
by the identity
\begin{equation}\label{eq:zero-bias}
             \E[Xf(X)]=v\E f'(X^*)
\end{equation}
for all absolutely continuous functions $f:\R\to\R$ with $\E|Xf(X)|<\infty$.
\citet[Lemma~1.1]{Goldstein2010} and \citet[Theorem~3]{Tyurin2009}
independently proved the sharp single-variable transport inequality
\begin{equation}\label{eq:single-bound}
 W_1(X,X^*)\le\frac{\E|X|^3}{2v}
\end{equation}
for every centered random variable with finite third absolute moment, and
observed that every nonconstant two-point distribution attains equality in
\eqref{eq:single-bound}.

Now let $X_1,\ldots,X_n$ be independent centered real random variables with
\begin{equation}\label{eq:array-data}
 S=\sum_{i=1}^nX_i,\qquad
 V=\sum_{i=1}^n\E X_i^2>0,\qquad
 R=\sum_{i=1}^n\E|X_i|^3<\infty.
\end{equation}
Combining \eqref{eq:single-bound} with the independent-sum replacement
characterization of $S^*$ yields the additive zero-bias transport inequality
$2VW_1(S,S^*)\le R$ \citep[Corollary~1]{Tyurin2009}. Our first theorem
characterizes all equality cases in both \eqref{eq:single-bound} and its
independent-sum extension.

\begin{theorem}[Rigidity of zero-bias transport]\label{thm:array-equality}
For every finite independent array satisfying \eqref{eq:array-data},
\begin{equation}\label{eq:additive-bound}
                      2V W_1(S,S^*)\le R.
\end{equation}
Equality holds in \eqref{eq:additive-bound} if and only if all nonconstant
summands $X_i$ have one identical centered two-point distribution (and any
remaining summands are identically zero). In particular, for $n=1$, equality
holds in \eqref{eq:single-bound} if and only if $X$ has a two-point
distribution.
\end{theorem}

The distinction between the sum of individual third absolute moments
$R=\sum_{i=1}^n\E|X_i|^3$ and the third absolute moment of the sum $\E|S|^3$
is essential in Theorem~\ref{thm:array-equality}. When $S$ is a sum of
$n\ge2$ independent, identically distributed nonconstant two-point variables,
$S$ is a centered binomial variable supported on $n+1\ge3$ points; by the
$n=1$ case of Theorem~\ref{thm:array-equality}, its single-variable bound
$2VW_1(S,S^*)<\E|S|^3$ is strict, whereas by the $n$-summand case of
Theorem~\ref{thm:array-equality}, its additive bound $2VW_1(S,S^*)=R$ holds
with exact equality.

\subsection{Lattice transport and repeated independent blocks}

Our second main result applies Theorem~\ref{thm:array-equality} to the
Berry--Esseen ratio of arrays formed by repeating a fixed finite block of
independent summands. Let $S_1,S_2,\ldots$ be independent copies of the block
sum $S$ in \eqref{eq:array-data}. For each integer $m\ge1$, the combined
array of $mn$ summands has total variance $mV$ and third absolute moment sum
$mR$, so its Berry--Esseen ratio is
\begin{equation}\label{eq:block-ratio}
 C_m=\frac{\sqrt m\,V^{3/2}}{R}
 \sup_{x\in\R}\left|\Pp\left\{\frac{S_1+\cdots+S_m}{\sqrt{mV}}\le x\right\}
                   -\Phi(x)\right|.
\end{equation}
Set
\[
 \ce=\frac{3+\sqrt{10}}{6\sqrt{2\pi}},\qquad
 p_{\mathrm E}=\frac{4-\sqrt{10}}2.
\]

\begin{theorem}[Asymptotic Berry--Esseen ratio for repeated blocks]\label{thm:block}
For any fixed independent block $(X_1,\ldots,X_n)$ satisfying \eqref{eq:array-data},
the limit of \eqref{eq:block-ratio} as $m\to\infty$ exists and satisfies
\begin{equation}\label{eq:block-bound}
 \lim_{m\to\infty}C_m
 \le \ce\frac{2V W_1(S,S^*)}{R}\le\ce.
\end{equation}
Equality $\lim_{m\to\infty}C_m=\ce$ holds if and only if, for some $h>0$,
every nonconstant summand $X_i$ has the distribution of $h(B-p_{\mathrm E})$
with $B\sim\Bern(p_{\mathrm E})$, or every nonconstant summand has the
reflected distribution $h(p_{\mathrm E}-B)$, and any remaining summands are
identically zero.
\end{theorem}

The bridge between Esseen's asymptotic expansion for $S_1+\cdots+S_m$ and the
zero-bias transport distance $W_1(S,S^*)$ in \eqref{eq:block-bound} is a
sharp inequality for centered lattice distributions proved in
Section~\ref{sec:lattice} (Theorem~\ref{thm:span}): for any centered random
variable $X$ of variance $v>0$ supported on a coset $a+h\Z$ ($h>0$) with
third moment $\kappa=\E X^3$,
\[
 |\kappa|+3hv\le(3+\sqrt{10})\,2vW_1(X,X^*),
\]
with equality if and only if $X\stackrel d=h(N-np)$, where $N\sim\Bin(n,p)$
($n\ge1$) and $p\in\{p_{\mathrm E},1-p_{\mathrm E}\}$.

Notice how the two inequalities in \eqref{eq:block-bound} play complementary
roles:
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item The first inequality in \eqref{eq:block-bound} depends only on the
distribution of the block sum $S$, and attains equality whenever $S$ is a
centered binomial random variable $h(\Bin(k,p_{\mathrm E})-kp_{\mathrm E})$
(or its reflection) for any $k\ge1$.
\item The second inequality in \eqref{eq:block-bound} is
Theorem~\ref{thm:array-equality}, which compares $2VW_1(S,S^*)$ against the
sum of the third absolute moments $R$ of the individual summands $X_1,\ldots,X_n$;
it attains equality if and only if the binomial sum $S$ is decomposed into
identically distributed two-point summands $h(B_i-p_{\mathrm E})$ and zeros.
\end{enumerate}

\subsection{Open questions and overview of the proofs}

Theorem~\ref{thm:block} treats the limit $m\to\infty$ in which a fixed
finite block $(X_1,\ldots,X_n)$ is repeated independently. For general
triangular arrays $(X_{n,1},\ldots,X_{n,k_n})$ whose laws may change with $n$,
a companion paper \citep[Theorem~1.1]{BellSmallStructure2026} proves that
whenever the Lyapunov ratio $L_n=R_n/V_n^{3/2}\to0$ and the Berry--Esseen
ratio $C_n\to\ce$, the normalized transport ratio $2V_nW_1(S_n,S_n^*)/R_n$
converges to $1$, and that for actual finite-$n$ global maximizers with
sufficiently small $L_n$, Theorem~\ref{thm:array-equality}'s exact equality
class is recovered. Determining whether $C_n$ is attained by identically
distributed two-point summands and zeros at *every* finite $n$ (without a
small-Lyapunov or common-diameter restriction) remains open.

The remainder of the paper is organized as follows. Section~\ref{sec:basic}
reviews the density and moment formulas for $X^*$ and identifies the three
inequalities in the independent-sum transport chain \eqref{eq:bound-chain}.
Section~\ref{sec:single-equality} proves the single-summand characterization
in Theorem~\ref{thm:array-equality} using a fixed-variance Choquet
representation over three-point extreme laws. Section~\ref{sec:array}
completes the proof of Theorem~\ref{thm:array-equality} for $n$ summands by
proving strict convolution contraction for unequal diameters and establishing
an exact conditional-variance identity (Proposition~\ref{prop:posterior}) for
common-diameter Bernoulli arrays. Section~\ref{sec:lattice} proves the
lattice span--transport inequality (Theorem~\ref{thm:span}), and
Section~\ref{sec:block} combines it with Theorem~\ref{thm:array-equality} to
prove Theorem~\ref{thm:block}.

\section{Zero biasing and the independent-sum transport chain}\label{sec:basic}

We recall three basic properties of the zero-bias transformation from
\citet[Lemma~2.1]{GoldsteinReinert1997}. First, for a centered random variable
$X$ of variance $v>0$, the distribution of $X^*$ is absolutely continuous
with respect to Lebesgue measure, with density
\begin{equation}\label{eq:density}
             d_X(u)=\frac1v\E\bigl[X\mathbf1_{\{X>u\}}\bigr]
                   =-\frac1v\E\bigl[X\mathbf1_{\{X\le u\}}\bigr].
\end{equation}
Second, substituting $f(x)=x^2/2$ and $f(x)=x|x|/2$ into \eqref{eq:zero-bias}
gives the first moments
\begin{equation}\label{eq:bias-moments}
 \E X^*=\frac{\E X^3}{2v},\qquad
 \E|X^*|=\frac{\E|X|^3}{2v}.
\end{equation}
Third, for any two probability laws $\mu,\nu$ on $\R$ with finite first
moments, the optimal quantile coupling \citep[eqs.~(15)--(16)]{Goldstein2010}
expresses $W_1$ as the $L^1$ distance between their distribution functions:
\begin{equation}\label{eq:cdf-distance}
       W_1(\mu,\nu)=\int_\R|F_\mu(x)-F_\nu(x)|\,dx.
\end{equation}

Now consider the independent array \eqref{eq:array-data}. Omitting any
summands of variance zero, write $v_i=\E X_i^2$, $\rho_i=\E|X_i|^3$, and
$T_i=S-X_i$. Let $I\in\{1,\ldots,n\}$ be a random index chosen independently
of $(X_1,\ldots,X_n)$ with $\Pp(I=i)=v_i/V$, and replace $X_I$ by an
independent random variable $X_I^*$ having the zero-biased distribution of
$X_I$. Conditioning on $T_i$ in \eqref{eq:zero-bias} gives
$\E[Sf(S)]=\sum_{i=1}^n v_i\E f'(T_i+X_i^*)$, so that
\begin{equation}\label{eq:replacement}
                  S^*\ \stackrel{d}=\ T_I+X_I^*.
\end{equation}
Applying the representation \eqref{eq:cdf-distance} and the triangle
inequality to the mixture \eqref{eq:replacement}, followed by translation
invariance of $W_1$ under convolution with $T_i$ and the single-summand bound
\eqref{eq:single-bound}, gives the three-step inequality chain
\begin{equation}\label{eq:bound-chain}
 \begin{split}
 W_1(S,S^*)
 &\le\sum_{i=1}^n\frac{v_i}{V}W_1(S,T_i+X_i^*)\\
 &\le\sum_{i=1}^n\frac{v_i}{V}W_1(X_i,X_i^*)
 \le\frac R{2V}.
 \end{split}
\end{equation}
Equality in \eqref{eq:additive-bound} holds if and only if all three
inequalities in \eqref{eq:bound-chain} are equalities: every summand must
attain equality in \eqref{eq:single-bound}, convolution with $T_i$ must
preserve $W_1(X_i,X_i^*)$ for every $i$, and the signed CDF differences
$F_{T_i+X_i}-F_{T_i+X_i^*}$ must have a common sign across $i$ almost
everywhere. The next two sections analyze these conditions in turn.

\section{Equality for a single distribution}\label{sec:single-equality}

For a centered probability measure $\mu$ on $\R$ with variance $v>0$ and
third absolute moment $\rho=\int|x|^3\,\mu(dx)<\infty$, define the scale- and
reflection-invariant transport ratio
\[
                 B(\mu)=\frac{2v W_1(\mu,\mu^*)}{\rho}\in(0,1].
\]

\subsection{Two-point distributions and strictness for three-point distributions}

Any centered two-point distribution takes the form $\Law(h(B-p))$ with $h>0$,
$0<p<1$, $q=1-p$, and $B\sim\Bern(p)$. Its variance is $v=h^2pq$, its third
absolute moment is $\rho=h^3pq(p^2+q^2)$, and by \eqref{eq:density} its
zero-biased distribution $\mu^*$ is uniform on $[-hp,hq]$. Direct integration
of $|F_\mu-F_{\mu^*}|$ on $[-hp,0)$ and $[0,hq]$ gives
\begin{equation}\label{eq:binary-transport}
                 W_1(\mu,\mu^*)=\frac h2(p^2+q^2)=\frac{\rho}{2v},
\end{equation}
so $B(\mu)=1$ \citep[Lemma~1.2]{Goldstein2010}.

\begin{lemma}\label{lem:strict-ternary}
Every centered distribution $\mu$ supported on three distinct points of
positive mass satisfies $B(\mu)<1$.
\end{lemma}
\begin{proof}
First suppose $0\notin\supp\mu$. After reflection if necessary, we may write
$\supp\mu=\{x,y,z\}$ with $x<y<0<z$. Let $\mu_1$ and $\mu_0$ be the unique
centered two-point distributions supported on $\{x,z\}$ and $\{y,z\}$,
respectively. Then $\mu=\alpha\mu_1+(1-\alpha)\mu_0$ for a unique
$\alpha\in(0,1)$. Since $\mu_1$ and $\mu_0$ have variances $-xz$ and $-yz$,
the zero-biased distribution $\mu^*$ is the variance-reweighted mixture
\citep[Theorem~2.1]{Goldstein2010}
\[
 \mu^*=\beta\mu_1^*+(1-\beta)\mu_0^*,\qquad
 \beta=\frac{\alpha x}{\alpha x+(1-\alpha)y}>\alpha.
\]
Writing $D_j=W_1(\mu_j,\mu_j^*)$ and $D_{10}=W_1(\mu_0,\mu_1^*)$, the
triangle inequality gives
\begin{equation}\label{eq:ternary-coupling}
 W_1(\mu,\mu^*)
 \le\alpha D_1+(1-\beta)D_0+(\beta-\alpha)D_{10},
\end{equation}
whereas the two-point identity \eqref{eq:binary-transport} for $\mu_1,\mu_0$
gives $\rho/(2v)=\beta D_1+(1-\beta)D_0$. Integrating the affine pieces of
$F_{\mu_1^*}-F_{\mu_0}$ explicitly (compare \citet[Lemma~3.1 and eq.~(28)]{Goldstein2010})
yields
\[
 D_1-D_{10}=
 \begin{cases}
 \displaystyle
 \frac{-y(y-x)\{y^2+2z^2-y(x+2z)\}}{(z-x)(z-y)^2},
       &y(x+z)\le y^2+z^2,\\[2ex]
 \displaystyle\frac{z^2}{z-x},&y(x+z)>y^2+z^2.
 \end{cases}
\]
In the first case, $y<0<z$ and $x<y$ imply $y^2+2z^2-y(x+2z)>z(z-y)>0$, so
in both cases $D_1-D_{10}>0$. Subtracting \eqref{eq:ternary-coupling} from
$\rho/(2v)$ therefore gives $\rho/(2v)-W_1(\mu,\mu^*)\ge(\beta-\alpha)(D_1-D_{10})>0$.

Second, suppose $0\in\supp\mu$. Write $\supp\mu=\{-ph,0,qh\}$ with $h>0$,
$p+q=1$, $0<p\le q$ (after reflection), middle atom mass $1-\tau\in(0,1)$,
and endpoint masses $\tau q$ and $\tau p$. Here $\mu^*$ is uniform on
$[-ph,qh]$, and direct integration of $|F_\mu-F_{\mu^*}|$ on $[-ph,0)$ and
$[0,qh]$ gives, with $r=p^2+q^2$,
\begin{equation}\label{eq:zero-middle}
 B(\mu)=
 \begin{cases}
 (q-p+2\tau^2p^2)/r,&\tau q\ge p,\\
 1+2\tau^2-4\tau pq/r,&\tau q<p.
 \end{cases}
\end{equation}
Hence $1-B(\mu)$ equals $2p^2(1-\tau^2)/r>0$ in the first case and
$2\tau(2pq/r-\tau)>0$ in the second (since $2pq/r\ge p/q>\tau$), proving
$B(\mu)<1$ in all cases.
\end{proof}

\subsection{Choquet representation at fixed unit variance}\label{sec:choquet}

To reduce an arbitrary centered distribution $\mu$ with $v=1$ and $\rho<\infty$
to laws supported on at most three points without changing the variance of the
components, consider the convex set of probability measures
\[
 \mathcal K=\left\{\nu:\nu\text{ is a probability measure on }\R,
         \ \int_\R x\,\nu(dx)=0,\ \int_\R x^2\,\nu(dx)\le1\right\}.
\]
The second-moment bound $\int x^2\,\nu(dx)\le1$ implies tightness and uniform
integrability of $x$, so $\mathcal K$ is compact and metrizable in the weak
topology induced by $C_b(\R)$. By Choquet's theorem
\citep[pp.~333--337, Corollary~2 and Lemma~5]{Choquet1960}, there exists a
probability measure $\pi$ supported on the set of extreme points of
$\mathcal K$ with barycenter $\mu$: that is, $\mu(A)=\int_{\mathcal K}\nu(A)\,\pi(d\nu)$
for every Borel set $A\subset\R$.

Every extreme point $\nu$ of $\mathcal K$ is supported on at most three
points: if $\nu$ charged four disjoint Borel sets $A_1,\ldots,A_4$, their
four moment vectors $(\nu(A_j),\int_{A_j}x\,d\nu,\int_{A_j}x^2\,d\nu)\in\R^3$
would be linearly dependent, so a nonzero linear combination
$g=\sum_{j=1}^4c_j\mathbf1_{A_j}$ would give two distinct measures
$(1\pm\varepsilon g)\nu\in\mathcal K$ with average $\nu$. Moreover, by
Tonelli's theorem,
\begin{equation}\label{eq:exact-variance-mixture}
 1=\int_\R x^2\,\mu(dx)=\int_{\mathcal K}\left(\int_\R x^2\,\nu(dx)\right)\pi(d\nu),\qquad
 \rho(\mu)=\int_{\mathcal K}\rho(\nu)\,\pi(d\nu).
\end{equation}
Because $\int x^2\,\nu(dx)\le1$ for every $\nu\in\mathcal K$, the first
identity in \eqref{eq:exact-variance-mixture} forces $\int x^2\,\nu(dx)=1$
for $\pi$-almost every $\nu$. Consequently, $\pi$-almost every component $\nu$
is a centered probability law of unit variance, so its zero-biased law is
given by the measurable kernel
\begin{equation}\label{eq:bias-kernel}
             \nu^*(A)=\int_\R x\left(\int_0^x\mathbf1_A(u)\,du\right)\nu(dx),
\end{equation}
and Tonelli's theorem yields the unweighted mixture identity
\begin{equation}\label{eq:same-weight}
                        \mu^*=\int_{\mathcal K}\nu^*\,\pi(d\nu).
\end{equation}
(As noted in \citet[Theorem~2.2 and Remark~2.2]{Goldstein2010}, equal component
variances are essential so that $\mu^*$ and $\mu$ share the same mixing
measure $\pi$.) Convexity of $W_1$ under mixtures and \eqref{eq:single-bound}
then give
\begin{equation}\label{eq:mixture-bound}
 W_1(\mu,\mu^*)
 \le\int_{\mathcal K} W_1(\nu,\nu^*)\,\pi(d\nu)
 \le\frac12\int_{\mathcal K}\rho(\nu)\,\pi(d\nu)=\frac{\rho(\mu)}2.
\end{equation}
If $B(\mu)=1$, then both inequalities in \eqref{eq:mixture-bound} must be
equalities, so $W_1(\nu,\nu^*)=\rho(\nu)/2$ for $\pi$-almost every $\nu$. By
Lemma~\ref{lem:strict-ternary}, $\pi$-almost every component $\nu$ must be a
unit-variance two-point distribution.

\subsection{Cancellation between distinct two-point components}

Every centered unit-variance two-point distribution is uniquely parameterized
by $a>0$ as
\[
 \nu_a=\frac1{1+a^2}\delta_{-a}
              +\frac{a^2}{1+a^2}\delta_{1/a},
\]
with signed third moment $c(a)=\int x^3\,\nu_a(dx)=1/a-a$, which is strictly
decreasing from $+\infty$ to $-\infty$ on $(0,\infty)$. On $(-a,1/a)$, the
signed CDF difference between $\nu_a$ and $\nu_a^*=\operatorname{Unif}[-a,1/a]$
is
\begin{equation}\label{eq:binary-cdf}
 f_a(x)=F_{\nu_a}(x)-F_{\nu_a^*}(x)
 =\frac1{1+a^2}-\frac{x+a}{a+1/a}\qquad(-a<x<1/a),
\end{equation}
and $f_a(x)=0$ for $x\notin[-a,1/a]$. Observe that $f_a(x)>0$ on $(-a,c(a))$
and $f_a(x)<0$ on $(c(a),1/a)$. For any $0<a<b$, we have
$\max\{-a,c(b)\}<\min\{1/b,c(a)\}$, so on the nonempty open interval
\begin{equation}\label{eq:opposite-interval}
          I_{a,b}=\bigl(\max\{-a,c(b)\},\ \min\{1/b,c(a)\}\bigr),
\end{equation}
we have $f_a(x)>0$ and $f_b(x)<0$. If the induced mixing measure $\pi$ on
$(0,\infty)$ had two distinct points $a<b$ in its support, then for every $x$
in a compact subinterval of $I_{a,b}$ both $\E_\pi(f_a(x))_+$ and
$\E_\pi(f_a(x))_-$ would be strictly positive, giving
$\E_\pi|f_a(x)|-|\E_\pi f_a(x)|=2\min\{\E_\pi(f_a(x))_+,\E_\pi(f_a(x))_-\}>0$
and making the first inequality in \eqref{eq:mixture-bound} strict. Thus
$\pi$ is a point mass at a single $a>0$, proving the single-variable case of
Theorem~\ref{thm:array-equality}.

\section{Equality for independent arrays}\label{sec:array}

Suppose now that $(X_1,\ldots,X_n)$ attains equality in
\eqref{eq:additive-bound}. By the last inequality in \eqref{eq:bound-chain}
and Section~\ref{sec:single-equality}, every nonconstant summand must be a
centered two-point variable:
\[
                  X_i=h_i(B_i-p_i),\qquad h_i>0,\quad 0<p_i<1,
\]
with independent $B_i\sim\Bern(p_i)$.

\subsection{Strict convolution loss for unequal support diameters}

Suppose that $h=\max_i h_i$ and some nonconstant summand has diameter
$0<\delta<h$. Relabel so that $X_1$ has diameter $h$ and lower-atom
probability $q=1-p_1\in(0,1)$. In uncentered coordinates,
$f(x)=F_{X_1+hp_1}(x)-F_{X_1^*+hp_1}(x)=q-x/h$ on $(0,h)$ and $0$ outside
$[0,h]$. Because $T_1=\sum_{j\ge2}X_j$ contains a summand of diameter
$\delta$, the distribution of $T_1$ assigns positive probability to two atoms
separated by $\delta$. The two corresponding translates of $f$ have opposite
signs on a translate of the nonempty open interval
\[
                    \bigl(\max\{qh,\delta\},\ \min\{qh+\delta,h\}\bigr),
\]
so $|\E f(x-T_1)|<\E|f(x-T_1)|$ on a set of positive Lebesgue measure, making
the second inequality $W_1(S,T_1+X_1^*)\le W_1(X_1,X_1^*)$ in
\eqref{eq:bound-chain} strict. Thus equality in \eqref{eq:additive-bound}
forces all nonconstant summands to share a common diameter $h>0$.

\subsection{Exact conditional-variance identity for a common diameter}\label{sec:posterior}

Assume now that all $n$ nonconstant summands have common support diameter
$h>0$. Put
\[
 K=\sum_{i=1}^n B_i,\qquad \mu=\sum_{i=1}^n p_i,\qquad
 w_i=\frac{h^2p_i(1-p_i)}V,\qquad
 c_{i,k}=\Pp\left\{\sum_{j\ne i}B_j=k\right\},
\]
and for each cell $0\le k\le n-1$, define the cell weight, posterior index
distribution, and cell mean
\begin{equation}\label{eq:posterior-weights}
 b_k=\sum_{i=1}^n w_i c_{i,k}>0,\qquad
 \pi_{i,k}=\frac{w_i c_{i,k}}{b_k},\qquad
 Q_k=\sum_{i=1}^n\pi_{i,k}(1-p_i),
\end{equation}
where $\sum_{k=0}^{n-1}b_k=1$ and $\sum_{i=1}^n\pi_{i,k}=1$.

\begin{proposition}[Conditional-variance identity for common-diameter Bernoulli arrays]\label{prop:posterior}
For any common-diameter Bernoulli array,
\begin{equation}\label{eq:posterior-deficit}
 \frac R{2V}-W_1(S,S^*)
       =h\sum_{k=0}^{n-1}b_k\Var_{\pi_k}(p_i)
       =\frac h2\sum_{k=0}^{n-1}
          \frac{\sum_{i,j=1}^nw_iw_jc_{i,k}c_{j,k}(p_i-p_j)^2}{b_k}.
\end{equation}
\end{proposition}
\begin{proof}
On the $k$th lattice cell $[h(k-\mu),h(k+1-\mu)]$, write $x=h(k-\mu+s)$ with
$s\in(0,1)$. Conditioning on $\sum_{j\ne i}B_j$ shows that only the event
$\{\sum_{j\ne i}B_j=k\}$ contributes to $F_S(x)-F_{T_i+X_i^*}(x)$ on this
cell, giving $F_S(x)-F_{T_i+X_i^*}(x)=c_{i,k}(1-p_i-s)$. Averaging with the
variance weights $w_i$ via \eqref{eq:replacement} yields
$F_S(x)-F_{S^*}(x)=b_k(Q_k-s)$, whose absolute value has integral on the $k$th
cell equal to
\[
 h b_k\int_0^1|Q_k-s|\,ds=\frac{h b_k}2\bigl\{Q_k^2+(1-Q_k)^2\bigr\}.
\]
On the other hand, \eqref{eq:binary-transport} and $\sum_k c_{i,k}=1$ give
\[
 \frac R{2V}=\frac h2\sum_{i=1}^nw_i\bigl\{p_i^2+(1-p_i)^2\bigr\}
            =\frac h2\sum_{k=0}^{n-1}b_k\E_{\pi_k}\bigl\{p_i^2+(1-p_i)^2\bigr\}.
\]
Subtracting the two expressions and noting that
$\E_{\pi_k}\{p_i^2+(1-p_i)^2\}-\{Q_k^2+(1-Q_k)^2\}=2\Var_{\pi_k}(p_i)$
proves \eqref{eq:posterior-deficit}.
\end{proof}

At $k=0$, $c_{i,0}=\prod_{j\ne i}(1-p_j)>0$ for every $i$, so $\pi_{i,0}>0$
for all $1\le i\le n$. Thus the right-hand side of \eqref{eq:posterior-deficit}
vanishes if and only if $p_1=\cdots=p_n$, in which case $Q_k=1-p$ for all $k$
and
\begin{equation}\label{eq:iid-transport}
                      W_1(S,S^*)=\frac h2\bigl\{p^2+(1-p)^2\bigr\}=\frac{R}{2V}.
\end{equation}
This completes the proof of Theorem~\ref{thm:array-equality}.

\section{Lattice span, skewness, and zero-bias transport}\label{sec:lattice}

Let $X$ be a centered lattice random variable of variance $v>0$ and finite
third absolute moment, supported on a coset $a+h\Z$ ($h>0$), with signed
third moment $\kappa=\E X^3$.

\begin{theorem}[Sharp lattice span--transport inequality]\label{thm:span}
For every centered random variable $X$ supported on $a+h\Z$ ($h>0$) with
variance $v>0$ and finite third absolute moment,
\begin{equation}\label{eq:span-transport}
              |\kappa|+3hv\le(3+\sqrt{10})\,2v W_1(X,X^*).
\end{equation}
Equality holds in \eqref{eq:span-transport} if and only if $X\stackrel d=h(N-np)$,
where $N\sim\Bin(n,p)$ for some integer $n\ge1$ and
$p\in\{p_{\mathrm E},1-p_{\mathrm E}\}$.
\end{theorem}

\subsection{Proof of the inequality by cellwise Jensen convexification}

By \eqref{eq:density}, the density $d_X$ of $X^*$ is constant on each open
lattice cell $(a+kh,a+(k+1)h)$; denote its value there by $d_k\ge0$. On cells
with $d_k=0$, $F_X-F_{X^*}=0$. On cells with $d_k>0$, set
\begin{gather*}
 \omega_k=hd_k,\qquad
 r_k=\frac{F_X(a+kh)-F_{X^*}(a+kh)}{hd_k},\\
 \ell(r)=\int_0^1|r-s|\,ds=
 \begin{cases}
 r^2-r+1/2,&0\le r\le1,\\
 |r-1/2|,&r\notin[0,1].
 \end{cases}
\end{gather*}
Then $\sum_k\omega_k=1$, and within the $k$th cell $F_X(a+(k+s)h)-F_{X^*}(a+(k+s)h)=hd_k(r_k-s)$,
so integrating $|F_X-F_{X^*}|$ and $F_{X^*}-F_X$ over $\R$ and using
$\E X^*-\E X=\kappa/(2v)$ gives
\begin{equation}\label{eq:cell-identities}
 \frac{W_1(X,X^*)}h=\sum_k\omega_k\ell(r_k),\qquad
 \sum_k\omega_k\left(r_k-\frac12\right)=\frac\kappa{2vh}.
\end{equation}
Because $\ell(r)$ is convex on $\R$, Jensen's inequality applied to
\eqref{eq:cell-identities} with $t=|\kappa|/(vh)=\bigl|\sum_k\omega_k(2r_k-1)\bigr|$
yields
\begin{equation}\label{eq:jensen-bound}
 2vW_1(X,X^*)=2vh\sum_k\omega_k\ell(r_k)\ge
 \begin{cases}
 vh(1+t^2)/2,&0\le t\le1,\\
 vht,&t\ge1.
 \end{cases}
\end{equation}
For $0\le t\le1$, the ratio $(|\kappa|+3hv)/[vh(1+t^2)/2]=2(3+t)/(1+t^2)$
attains its unique maximum $3+\sqrt{10}$ at $t=\sqrt{10}-3<1$; for $t\ge1$,
$(|\kappa|+3hv)/(vht)=(3+t)/t\le4<3+\sqrt{10}$. This proves
\eqref{eq:span-transport}. (Note that combining \eqref{eq:span-transport}
with $2vW_1(X,X^*)\le\E|X|^3$ immediately recovers Esseen's moment inequality
$|\kappa|+3hv\le(3+\sqrt{10})\E|X|^3$ \citep[Lemma~1]{Esseen1956}.)

\subsection{Characterization of the binomial equality cases}

Equality in \eqref{eq:span-transport} requires $t=|\kappa|/(vh)=\sqrt{10}-3\in(0,1)$,
where $\ell$ is strictly convex, so all cells with $\omega_k>0$ must share the
same ratio $r_k=r$ satisfying
\begin{equation}\label{eq:r-optimal}
                 0<r<1,\qquad |2r-1|=\sqrt{10}-3.
\end{equation}
At any lattice point $x\in a+h\Z$ with atom mass $p_x=\Pp(X=x)$ and adjacent
cell densities $d_-$ (to its left) and $d_+$ (to its right), the jump
relations $p_x=h\{rd_++(1-r)d_-\}$ and $v(d_--d_+)=xp_x$ solve to
\begin{equation}\label{eq:binomial-density}
 hd_-=p_x\left(1+\frac{rhx}{v}\right),\qquad
 hd_+=p_x\left(1-\frac{(1-r)hx}{v}\right).
\end{equation}
No interior lattice point between the support extremes can have $p_x=0$,
since across a zero-mass grid point $d_-=d_+>0$ and $r_k$ would decrease by
$1$, contradicting constancy of $r_k$. Nonnegativity of $d_\pm$ in
\eqref{eq:binomial-density} confines the support to $[-v/(rh),v/((1-r)h)]$,
with extreme atoms at $x_-=-v/(rh)$ and $x_+=v/((1-r)h)$ separated by
$x_+-x_-=nh$ for some integer $n\ge1$. Setting $p=1-r\in\{p_{\mathrm E},1-p_{\mathrm E}\}$,
we have $v=np(1-p)h^2$ and $x_j=h(j-np)$ for $0\le j\le n$. Equating the two
expressions in \eqref{eq:binomial-density} for the density on the cell
$(x_j,x_{j+1})$ gives the binomial recursion
\[
             \frac{p_{x_{j+1}}}{p_{x_j}}
                    =\frac{(n-j)p}{(j+1)(1-p)},
\]
which identifies $\Law(X)$ as $h(\Bin(n,p)-np)$. Conversely,
\eqref{eq:iid-transport} verifies that every such law attains equality in
\eqref{eq:span-transport}, completing the proof of Theorem~\ref{thm:span}.

\section{Proof of Theorem~\ref{thm:block}: repeated independent blocks}\label{sec:block}

Let $S=\sum_{i=1}^nX_i$ be the sum of the fixed block in \eqref{eq:array-data},
with variance $V=\sum_{i=1}^n\E X_i^2$, third signed moment
$\kappa=\E S^3=\sum_{i=1}^n\E X_i^3$, and maximal lattice span $h\ge0$ (where
$h=0$ if the law of $S$ is nonlattice). By Esseen's asymptotic expansion for
sums of independent copies of a fixed distribution with finite third absolute
moment \citep[pp.~160--162, eqs.~(6)--(7)]{Esseen1956},
\begin{equation}\label{eq:esseen-limit}
 \lim_{m\to\infty}\sqrt m\,
 d_{\mathrm K}\!\left(\frac{S_1+\cdots+S_m}{\sqrt{mV}},\,Z\right)
                  =\frac{\phz}{6V^{3/2}}(|\kappa|+3hV).
\end{equation}
Multiplying \eqref{eq:esseen-limit} by $V^{3/2}/R$ gives
\begin{equation}\label{eq:original-price-limit}
                    \lim_{m\to\infty}C_m
                      =\frac{\phz}{6R}(|\kappa|+3hV).
\end{equation}

If $h>0$, applying Theorem~\ref{thm:span} to $S$ and dividing by $R$ gives
the first inequality in \eqref{eq:block-bound}, and Theorem~\ref{thm:array-equality}
gives the second. If $h=0$, then $|\kappa|=2V|\E S^*|\le2VW_1(S,S^*)$ by
\eqref{eq:bias-moments}, so
\[
             \lim_{m\to\infty}C_m
             \le\frac{\phz}{6}\frac{2V W_1(S,S^*)}{R}
             \le\frac{\phz}{6}<\ce.
\]

Finally, if $\lim_{m\to\infty}C_m=\ce$, then $h>0$ and both inequalities in
\eqref{eq:block-bound} must hold with equality. By Theorem~\ref{thm:array-equality},
all $k\ge1$ nonconstant summands $X_i$ share a common centered two-point law
$g(B-p)$ ($g>0$, $0<p<1$), so $S\stackrel d=g(\Bin(k,p)-kp)$ has maximal
lattice span $h=g$ and $\kappa/(Vg)=1-2p$. Equality in Theorem~\ref{thm:span}
then forces $|1-2p|=\sqrt{10}-3$, i.e., $p\in\{p_{\mathrm E},1-p_{\mathrm E}\}$.
Conversely, any such block attains equality in both steps of
\eqref{eq:block-bound}. This completes the proof of Theorem~\ref{thm:block}.
